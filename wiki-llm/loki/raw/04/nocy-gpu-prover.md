static void derive\_seed(uint8\_t out[32], const uint8\_t seed[32], const char \*label) {

Blake2b hasher;

hasher.init\_keyed(reinterpret\_cast<const uint8\_t \*>(label), std::strlen(label), 32);

struct TimingEntry {

const char \*label;

std::chrono::steady\_clock::duration duration;

};

static void log\_timings(const std::vector<TimingEntry> &timings) {

if (!timing\_enabled()) {

return;

}

for (const auto &entry : timings) {

auto us = std::chrono::duration\_cast<std::chrono::microseconds>(entry.duration).count();

std::fprintf(stderr, "timing: %s=%lldus\n", entry.label, static\_cast<long long>(us));

}

}

static bool fr\_equal(const Fr &a, const Fr &b) {

return a.limbs[0] == b.limbs[0] && a.limbs[1] == b.limbs[1] && a.limbs[2] == b.limbs[2] &&

a.limbs[3] == b.limbs[3];

}

static void debug\_log\_h\_values(const char \*stage, const std::vector<Fr> &values) {

if (!h\_debug\_enabled()) {

return;

}

const size\_t size = values.size();

if (size == 0) {

return;

}

std::vector<size\_t> idxs;

idxs.push\_back(0);

if (size > 1) {

idxs.push\_back(1);

idxs.push\_back(size / 2);

idxs.push\_back(size - 1);

}

std::sort(idxs.begin(), idxs.end());

idxs.erase(std::unique(idxs.begin(), idxs.end()), idxs.end());

std::fprintf(stderr, "h debug: stage=%s size=%zu\n", stage, size);

for (size\_t idx : idxs) {

char hex[65];

fr\_to\_hex\_be(values[idx], hex);

std::fprintf(stderr, "h debug: stage=%s idx=%zu value=%s\n", stage, idx, hex);

}

}

static void debug\_log\_rows(const char \*label, const std::vector<Fr> &values, size\_t n,

size\_t unusable\_rows\_start) {

if (!h\_debug\_enabled() || values.empty()) {

return;

}

std::vector<size\_t> rows;

rows.push\_back(0);

if (n > 1) {

rows.push\_back(1);

rows.push\_back(unusable\_rows\_start);

rows.push\_back(n - 1);

}

std::sort(rows.begin(), rows.end());

rows.erase(std::unique(rows.begin(), rows.end()), rows.end());

for (size\_t row : rows) {

if (row >= values.size()) {

continue;

}

char hex[65];

fr\_to\_hex\_be(values[row], hex);

std::fprintf(stderr, "h debug: %s row=%zu value=%s\n", label, row, hex);

}

}

Fr \*h\_values = nullptr;

size\_t h\_values\_len = 0;

};

\_\_device\_\_ \_\_forceinline\_\_ uint32\_t rotation\_index\_device(uint32\_t idx, int32\_t rot,

int32\_t rot\_scale, uint32\_t isize) {

uint32\_t mask = isize - 1;

int64\_t delta = static\_cast<int64\_t>(rot) \* static\_cast<int64\_t>(rot\_scale);

uint32\_t offset = static\_cast<uint32\_t>(idx) + static\_cast<uint32\_t>(delta);

return offset & mask;

}

\_\_device\_\_ \_\_forceinline\_\_ Fr value\_from\_source\_device(

const DeviceGraphValueSource &source, const uint32\_t \*rotation\_indices,

const Fr \*constants, const Fr \*intermediates, const Fr \*fixed\_values,

const Fr \*advice\_values, const Fr \*instance\_values, const Fr \*challenges,

const Fr &beta, const Fr &gamma, const Fr &theta, const Fr &y,

const Fr &previous\_value, uint32\_t stride) {

switch (source.tag) {

case 0:

return constants[source.a];

case 1:

return intermediates[source.a];

case 2: {

uint32\_t row = rotation\_indices[source.b];

return fixed\_values[source.a \* stride + row];

}

case 3: {

uint32\_t row = rotation\_indices[source.b];

return advice\_values[source.a \* stride + row];

}

case 4: {

uint32\_t row = rotation\_indices[source.b];

return instance\_values[source.a \* stride + row];

}

case 5:

return challenges[source.a];

case 6:

return beta;

case 7:

return gamma;

case 8:

return theta;

case 9:

return y;

case 10:

return previous\_value;

default: {

Fr zero{};

return zero;

}

}

}

\_\_device\_\_ \_\_forceinline\_\_ Fr evaluate\_graph\_value\_device(

const DeviceGraph &graph, const Fr \*fixed\_values, const Fr \*advice\_values,

const Fr \*instance\_values, const Fr \*challenges, const Fr &beta, const Fr &gamma,

const Fr &theta, const Fr &y, const Fr &previous\_value, uint32\_t idx, int32\_t rot\_scale,

uint32\_t stride, uint32\_t isize) {

if (graph.num\_rotations > kGraphMaxRotations ||

graph.num\_intermediates > kGraphMaxIntermediates) {

return Fr{};

}

uint32\_t rotation\_indices[kGraphMaxRotations];

for (uint32\_t i = 0; i < graph.num\_rotations; ++i) {

rotation\_indices[i] = rotation\_index\_device(idx, graph.rotations[i], rot\_scale, isize);

}

Fr intermediates[kGraphMaxIntermediates];

for (uint32\_t i = 0; i < graph.num\_intermediates; ++i) {

intermediates[i] = Fr{};

}

for (uint32\_t i = 0; i < graph.num\_calculations; ++i) {

const auto &calc = graph.calculations[i];

Fr result{};

switch (calc.op) {

case 0: {

Fr a = value\_from\_source\_device(calc.lhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

Fr b = value\_from\_source\_device(calc.rhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

fr\_add(result, a, b);

break;

}

case 1: {

Fr a = value\_from\_source\_device(calc.lhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

Fr b = value\_from\_source\_device(calc.rhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

fr\_sub(result, a, b);

break;

}

case 2: {

Fr a = value\_from\_source\_device(calc.lhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

Fr b = value\_from\_source\_device(calc.rhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

fr\_mul(result, a, b);

break;

}

case 3: {

Fr a = value\_from\_source\_device(calc.lhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

fr\_mul(result, a, a);

break;

}

case 4: {

Fr a = value\_from\_source\_device(calc.lhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

fr\_add(result, a, a);

break;

}

case 5: {

Fr a = value\_from\_source\_device(calc.lhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

Fr zero{};

fr\_sub(result, zero, a);

break;

}

case 6: {

Fr value = value\_from\_source\_device(calc.lhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

Fr factor = value\_from\_source\_device(calc.factor, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

for (uint32\_t p = 0; p < calc.parts\_len; ++p) {

const auto &part = graph.parts[calc.parts\_offset + p];

Fr part\_value = value\_from\_source\_device(part, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

Fr tmp;

fr\_mul(tmp, value, factor);

fr\_add(value, tmp, part\_value);

}

result = value;

break;

}

case 7: {

result = value\_from\_source\_device(calc.lhs, rotation\_indices, graph.constants,

intermediates, fixed\_values, advice\_values,

instance\_values, challenges, beta, gamma,

theta, y, previous\_value, stride);

break;

}

default: {

Fr zero{};

result = zero;

break;

}

}

if (calc.target < graph.num\_intermediates) {

intermediates[calc.target] = result;

}

}

if (graph.num\_calculations > 0 && graph.last\_target < graph.num\_intermediates) {

return intermediates[graph.last\_target];

}

return Fr{};

}

\_\_global\_\_ void graph\_eval\_kernel(DeviceGraph graph, const Fr \*fixed\_values,

const Fr \*advice\_values, const Fr \*instance\_values,

const Fr \*challenges, Fr beta, Fr gamma, Fr theta, Fr y,

Fr previous\_value, Fr \*out, uint32\_t stride,

int32\_t rot\_scale, uint32\_t isize) {

uint32\_t idx = blockIdx.x \* blockDim.x + threadIdx.x;

if (idx >= stride) {

return;

}

out[idx] = evaluate\_graph\_value\_device(graph, fixed\_values, advice\_values, instance\_values,

challenges, beta, gamma, theta, y, previous\_value, idx,

rot\_scale, stride, isize);

}

\_\_global\_\_ void permutation\_kernel(Fr \*values, const Fr \*perm\_product\_cosets, uint32\_t perm\_sets,

const Fr \*perm\_cosets, const DevicePermutationColumn \*columns,

uint32\_t columns\_len, const Fr \*advice\_cosets,

const Fr \*fixed\_cosets, const Fr \*instance\_cosets,

const Fr \*beta\_terms, Fr delta\_start, Fr delta\_const,

const Fr \*l0, const Fr \*l\_last, const Fr \*l\_active\_row,

Fr beta, Fr gamma, Fr y, uint32\_t size, int32\_t rot\_scale,

int32\_t last\_rotation, uint32\_t chunk\_len) {

uint32\_t idx = blockIdx.x \* blockDim.x + threadIdx.x;

if (idx >= size) {

return;

}

if (!values || !perm\_product\_cosets || !perm\_cosets || !columns || !beta\_terms || !l0 ||

!l\_last || !l\_active\_row || chunk\_len == 0 || columns\_len == 0 || perm\_sets == 0) {

return;

}

uint32\_t r\_next = rotation\_index\_device(idx, 1, rot\_scale, size);

uint32\_t r\_last = rotation\_index\_device(idx, last\_rotation, rot\_scale, size);

Fr value = values[idx];

Fr one = fr\_one();

const Fr \*first\_set = perm\_product\_cosets;

const Fr \*last\_set = perm\_product\_cosets + (perm\_sets - 1) \* size;

Fr tmp;

fr\_sub(tmp, one, first\_set[idx]);

fr\_mul(tmp, tmp, l0[idx]);

fr\_mul(value, value, y);

fr\_add(value, value, tmp);

Fr last\_eval = last\_set[idx];

Fr last\_sq;

fr\_mul(last\_sq, last\_eval, last\_eval);

fr\_sub(last\_sq, last\_sq, last\_eval);

fr\_mul(last\_sq, last\_sq, l\_last[idx]);

fr\_mul(value, value, y);

fr\_add(value, value, last\_sq);

if (perm\_sets > 1) {

for (uint32\_t set\_idx = 1; set\_idx < perm\_sets; ++set\_idx) {

const Fr \*set\_coset = perm\_product\_cosets + set\_idx \* size;

const Fr \*prev\_coset = perm\_product\_cosets + (set\_idx - 1) \* size;

Fr diff;

fr\_sub(diff, set\_coset[idx], prev\_coset[r\_last]);

fr\_mul(diff, diff, l0[idx]);

fr\_mul(value, value, y);

fr\_add(value, value, diff);

}

}

Fr current\_delta;

fr\_mul(current\_delta, delta\_start, beta\_terms[idx]);

uint32\_t col\_offset = 0;

while (col\_offset < columns\_len) {

uint32\_t chunk\_end = col\_offset + chunk\_len;

if (chunk\_end > columns\_len) {

chunk\_end = columns\_len;

}

uint32\_t set\_idx = col\_offset / chunk\_len;

const Fr \*set\_coset = perm\_product\_cosets + set\_idx \* size;

Fr left = set\_coset[r\_next];

for (uint32\_t col\_idx = col\_offset; col\_idx < chunk\_end; ++col\_idx) {

DevicePermutationColumn col = columns[col\_idx];

const Fr \*values\_col = nullptr;

if (col.column\_type == 0) {

values\_col = advice\_cosets;

} else if (col.column\_type == 1) {

values\_col = fixed\_cosets;

} else {

values\_col = instance\_cosets;

}

if (!values\_col) {

return;

}

Fr perm\_val = perm\_cosets[col\_idx \* size + idx];

Fr term;

fr\_mul(term, perm\_val, beta);

fr\_add(term, term, gamma);

fr\_add(term, term, values\_col[col.index \* size + idx]);

fr\_mul(left, left, term);

}

Fr right = set\_coset[idx];

for (uint32\_t col\_idx = col\_offset; col\_idx < chunk\_end; ++col\_idx) {

DevicePermutationColumn col = columns[col\_idx];

const Fr \*values\_col = nullptr;

if (col.column\_type == 0) {

values\_col = advice\_cosets;

} else if (col.column\_type == 1) {

values\_col = fixed\_cosets;

} else {

values\_col = instance\_cosets;

}

if (!values\_col) {

return;

}

Fr term;

fr\_add(term, current\_delta, gamma);

fr\_add(term, term, values\_col[col.index \* size + idx]);

fr\_mul(right, right, term);

Fr next\_delta;

fr\_mul(next\_delta, current\_delta, delta\_const);

current\_delta = next\_delta;

}

Fr diff;

fr\_sub(diff, left, right);

fr\_mul(diff, diff, l\_active\_row[idx]);

fr\_mul(value, value, y);

fr\_add(value, value, diff);

col\_offset = chunk\_end;

}

values[idx] = value;

}

\_\_global\_\_ void lookup\_kernel(Fr \*values, const Fr \*product\_coset, const Fr \*permuted\_input\_coset,

const Fr \*permuted\_table\_coset, const Fr \*lookup\_values,

const Fr \*l0, const Fr \*l\_last, const Fr \*l\_active\_row, Fr beta,

Fr gamma, Fr y, uint32\_t size, int32\_t rot\_scale) {

uint32\_t idx = blockIdx.x \* blockDim.x + threadIdx.x;

if (idx >= size) {

return;

}

if (!values || !product\_coset || !permuted\_input\_coset || !permuted\_table\_coset || !lookup\_values ||

!l0 || !l\_last || !l\_active\_row) {

return;

}

uint32\_t r\_next = rotation\_index\_device(idx, 1, rot\_scale, size);

uint32\_t r\_prev = rotation\_index\_device(idx, -1, rot\_scale, size);

Fr value = values[idx];

Fr one = fr\_one();

Fr a\_minus\_s;

fr\_sub(a\_minus\_s, permuted\_input\_coset[idx], permuted\_table\_coset[idx]);

Fr term;

fr\_sub(term, one, product\_coset[idx]);

fr\_mul(term, term, l0[idx]);

fr\_mul(value, value, y);

fr\_add(value, value, term);

Fr term2;

fr\_mul(term2, product\_coset[idx], product\_coset[idx]);

fr\_sub(term2, term2, product\_coset[idx]);

fr\_mul(term2, term2, l\_last[idx]);

fr\_mul(value, value, y);

fr\_add(value, value, term2);

Fr left = product\_coset[r\_next];

Fr tmp;

fr\_add(tmp, permuted\_input\_coset[idx], beta);

fr\_mul(left, left, tmp);

fr\_add(tmp, permuted\_table\_coset[idx], gamma);

fr\_mul(left, left, tmp);

Fr right = product\_coset[idx];

fr\_mul(right, right, lookup\_values[idx]);

Fr diff;

fr\_sub(diff, left, right);

fr\_mul(diff, diff, l\_active\_row[idx]);

fr\_mul(value, value, y);

fr\_add(value, value, diff);

Fr l0\_term;

fr\_mul(l0\_term, a\_minus\_s, l0[idx]);

fr\_mul(value, value, y);

fr\_add(value, value, l0\_term);

Fr prev\_diff;

fr\_sub(prev\_diff, permuted\_input\_coset[idx], permuted\_input\_coset[r\_prev]);

fr\_mul(prev\_diff, prev\_diff, a\_minus\_s);

fr\_mul(prev\_diff, prev\_diff, l\_active\_row[idx]);

fr\_mul(value, value, y);

fr\_add(value, value, prev\_diff);

values[idx] = value;

}

\_\_global\_\_ void distribute\_powers\_zeta\_kernel(Fr \*values, uint32\_t base\_len, uint32\_t stride,

uint32\_t batch, Fr g\_coset, Fr g\_coset\_inv,

uint8\_t into\_coset) {

uint32\_t idx = blockIdx.x \* blockDim.x + threadIdx.x;

uint32\_t total = base\_len \* batch;

if (!values || idx >= total || base\_len == 0) {

return;

}

uint32\_t poly\_idx = idx / base\_len;

uint32\_t row = idx - poly\_idx \* base\_len;

uint32\_t mod = row % 3;

if (mod == 0) {

return;

}

Fr factor = (mod == 1)

? (into\_coset ? g\_coset : g\_coset\_inv)

: (into\_coset ? g\_coset\_inv : g\_coset);

Fr tmp;

uint32\_t offset = poly\_idx \* stride + row;

fr\_mul(tmp, values[offset], factor);

values[offset] = tmp;

}

static void log\_fr\_mismatch(const char \*label, size\_t idx, const Fr &expected, const Fr &got) {

char expected\_hex[65];

char got\_hex[65];

fr\_to\_hex\_be(expected, expected\_hex);

fr\_to\_hex\_be(got, got\_hex);

std::fprintf(stderr, "sanity: %s idx=%zu expected=%s got=%s\n", label, idx, expected\_hex,

got\_hex);

}

static void check\_t\_evaluations(const ProverKey &pk) {

if (!sanity\_enabled()) {

return;

}

const auto &t\_evals = pk.domain.t\_evaluations;

if (t\_evals.empty()) {

std::fprintf(stderr, "sanity: t\_evaluations empty\n");

return;

}

size\_t t\_len = t\_evals.size();

size\_t idxs[] = {0, 1, t\_len / 2, t\_len - 1};

for (size\_t idx : idxs) {

if (idx >= t\_len) {

continue;

}

Fr omega\_i = fr\_pow\_u64(pk.domain.extended\_omega, static\_cast<uint64\_t>(idx));

Fr x;

fr\_mul(x, omega\_i, pk.domain.g\_coset);

Fr xn = fr\_pow\_u64(x, static\_cast<uint64\_t>(1ULL << pk.domain.k));

Fr t;

fr\_sub(t, xn, fr\_one());

Fr t\_inv;

fr\_inv(t\_inv, t);

if (!fr\_equal(t\_inv, t\_evals[idx])) {

log\_fr\_mismatch("t\_evaluation", idx, t\_inv, t\_evals[idx]);

}

}

}

static void check\_lagrange\_polys(const ProverKey &pk) {

if (!sanity\_enabled()) {

return;

}

size\_t n = static\_cast<size\_t>(1ULL << pk.domain.k);

size\_t extended\_len = static\_cast<size\_t>(1ULL << pk.domain.extended\_k);

if (pk.l0.values.size() != extended\_len || pk.l\_last.values.size() != extended\_len ||

pk.l\_active\_row.values.size() != extended\_len) {

std::fprintf(stderr, "sanity: lagrange poly size mismatch\n");

return;

}

std::vector<Fr> l0\_vals(n, Fr{});

std::vector<Fr> l\_last\_vals(n, Fr{});

std::vector<Fr> l\_blind\_vals(n, Fr{});

l0\_vals[0] = fr\_one();

size\_t last\_row = n - (pk.blinding\_factors + 1);

if (last\_row < n) {

l\_last\_vals[last\_row] = fr\_one();

}

for (size\_t idx = n - pk.blinding\_factors; idx < n; ++idx) {

l\_blind\_vals[idx] = fr\_one();

}

PolynomialFr l0\_lagrange{std::move(l0\_vals)};

PolynomialFr l\_last\_lagrange{std::move(l\_last\_vals)};

PolynomialFr l\_blind\_lagrange{std::move(l\_blind\_vals)};

PolynomialFr l0\_coeff = lagrange\_to\_coeff(l0\_lagrange, pk.domain);

PolynomialFr l\_last\_coeff = lagrange\_to\_coeff(l\_last\_lagrange, pk.domain);

PolynomialFr l\_blind\_coeff = lagrange\_to\_coeff(l\_blind\_lagrange, pk.domain);

PolynomialFr l0\_ext = coeff\_to\_extended(l0\_coeff, pk.domain);

PolynomialFr l\_last\_ext = coeff\_to\_extended(l\_last\_coeff, pk.domain);

PolynomialFr l\_blind\_ext = coeff\_to\_extended(l\_blind\_coeff, pk.domain);

std::vector<Fr> l\_active\_ext(extended\_len);

for (size\_t i = 0; i < extended\_len; ++i) {

Fr tmp;

fr\_add(tmp, l\_last\_ext.values[i], l\_blind\_ext.values[i]);

fr\_sub(l\_active\_ext[i], fr\_one(), tmp);

}

size\_t idxs[] = {0, 1, extended\_len / 2, extended\_len - 1};

for (size\_t idx : idxs) {

if (idx >= extended\_len) {

continue;

}

if (!fr\_equal(l0\_ext.values[idx], pk.l0.values[idx])) {

log\_fr\_mismatch("l0", idx, l0\_ext.values[idx], pk.l0.values[idx]);

}

if (!fr\_equal(l\_last\_ext.values[idx], pk.l\_last.values[idx])) {

log\_fr\_mismatch("l\_last", idx, l\_last\_ext.values[idx], pk.l\_last.values[idx]);

}

if (!fr\_equal(l\_active\_ext[idx], pk.l\_active\_row.values[idx])) {

log\_fr\_mismatch("l\_active\_row", idx, l\_active\_ext[idx], pk.l\_active\_row.values[idx]);

}

}

}

static void check\_permutation\_data(const ProverKey &pk) {

if (!sanity\_enabled()) {

return;

}

size\_t poly\_count = pk.permutation\_polys.size();

size\_t perm\_count = pk.permutation\_permutations.size();

size\_t coset\_count = pk.permutation\_cosets.size();

if (poly\_count == 0) {

return;

}

if (poly\_count != perm\_count || poly\_count != coset\_count) {

std::fprintf(stderr,

"sanity: permutation size mismatch polys=%zu permutations=%zu cosets=%zu\n",

poly\_count, perm\_count, coset\_count);

return;

}

size\_t n = static\_cast<size\_t>(1ULL << pk.domain.k);

size\_t extended\_len = static\_cast<size\_t>(1ULL << pk.domain.extended\_k);

size\_t col\_idxs[] = {0, 1, poly\_count / 2, poly\_count - 1};

size\_t row\_idxs[] = {0, 1, n / 2, n - 1};

size\_t coset\_idxs[] = {0, 1, extended\_len / 2, extended\_len - 1};

for (size\_t col : col\_idxs) {

if (col >= poly\_count) {

continue;

}

PolynomialFr computed\_coset = coeff\_to\_extended(pk.permutation\_polys[col], pk.domain);

for (size\_t idx : coset\_idxs) {

if (idx >= computed\_coset.values.size() || idx >= pk.permutation\_cosets[col].values.size()) {

continue;

}

if (!fr\_equal(computed\_coset.values[idx], pk.permutation\_cosets[col].values[idx])) {

log\_fr\_mismatch("permutation\_coset", idx, computed\_coset.values[idx],

pk.permutation\_cosets[col].values[idx]);

}

}

for (size\_t row : row\_idxs) {

if (row >= n || row >= pk.permutation\_permutations[col].values.size()) {

continue;

}

Fr point = fr\_pow\_u64(pk.domain.omega, static\_cast<uint64\_t>(row));

Fr eval = poly\_eval(pk.permutation\_polys[col].values, point);

if (!fr\_equal(eval, pk.permutation\_permutations[col].values[row])) {

log\_fr\_mismatch("permutation\_eval", row, eval,

pk.permutation\_permutations[col].values[row]);

}

}

}

}

if (desired == 0) {

return true;

}

if (cudaMalloc(ptr, desired \* sizeof(Fr)) != cudaSuccess) {

error = "cudaMalloc buffer failed";

return false;

}

len = desired;

return true;

}

static bool ensure\_fft\_scratch(GraphGpuCache &cache, size\_t required, std::string &error) {

return ensure\_device\_buffer(&cache.fft\_scratch, cache.fft\_scratch\_len, required, error);

}

static void collect\_graph\_sources(const GraphEvaluator &graph, std::vector<uint32\_t> &fixed\_cols,

std::vector<uint32\_t> &advice\_cols,

std::vector<uint32\_t> &instance\_cols) {

auto record = [&](const GraphValueSource &src) {

if (src.tag == 2) {

fixed\_cols.push\_back(src.a);

} else if (src.tag == 3) {

advice\_cols.push\_back(src.a);

} else if (src.tag == 4) {

instance\_cols.push\_back(src.a);

}

};

for (const auto &calc : graph.calculations) {

switch (calc.op) {

case 0:

case 1:

case 2:

record(calc.lhs);

record(calc.rhs);

break;

case 3:

case 4:

case 5:

case 7:

record(calc.lhs);

break;

case 6:

record(calc.lhs);

for (const auto &part : calc.parts) {

record(part);

}

record(calc.factor);

break;

default:

break;

}

}

}

static bool build\_column\_map(uint32\_t fixed\_count, uint32\_t advice\_count, uint32\_t instance\_count,

const std::vector<const GraphEvaluator \*> &graphs,

const std::vector<PermutationColumn> \*perm\_columns,

GraphColumnMap &out, std::string &error) {

std::vector<uint32\_t> fixed\_cols;

std::vector<uint32\_t> advice\_cols;

std::vector<uint32\_t> instance\_cols;

for (const auto \*graph : graphs) {

if (!graph) {

continue;

}

collect\_graph\_sources(\*graph, fixed\_cols, advice\_cols, instance\_cols);

}

if (perm\_columns) {

for (const auto &col : \*perm\_columns) {

if (col.column\_type == 0) {

advice\_cols.push\_back(col.index);

} else if (col.column\_type == 1) {

fixed\_cols.push\_back(col.index);

} else {

instance\_cols.push\_back(col.index);

}

}

}

auto uniq = [](std::vector<uint32\_t> &vals) {

std::sort(vals.begin(), vals.end());

vals.erase(std::unique(vals.begin(), vals.end()), vals.end());

};

uniq(fixed\_cols);

uniq(advice\_cols);

uniq(instance\_cols);

out.fixed\_map.assign(fixed\_count, -1);

out.advice\_map.assign(advice\_count, -1);

out.instance\_map.assign(instance\_count, -1);

out.fixed\_indices = fixed\_cols;

out.advice\_indices = advice\_cols;

out.instance\_indices = instance\_cols;
