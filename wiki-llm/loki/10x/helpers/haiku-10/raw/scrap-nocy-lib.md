use anyhow::{anyhow, Context, Result};

use libloading::{Library, Symbol};

use midnight\_cuda\_packager::{pack\_params, pack\_prover\_key, packed\_params\_magic, packed\_pk\_magic};

use native\_plonk\_packer::pack\_polys;

use std::borrow::Cow;

use std::collections::hash\_map::DefaultHasher;

use std::hash::Hasher;

use std::os::raw::c\_char;

use std::sync::{Arc, Mutex};

use std::time::{Duration, Instant};

fn timing\_enabled() -> bool {

std::env::var("MIDNIGHT\_CUDA\_TIMING")

.ok()

.as\_deref()

.map(|val| val == "1")

.unwrap\_or(false)

}

fn log\_duration(label: &str, duration: Option<Duration>) {

if let Some(value) = duration {

eprintln!("timing: {label}={value:?}");

} else {

eprintln!("timing: {label}=skipped");

}

}

fn hash\_bytes(data: &[u8]) -> u64 {

let mut hasher = DefaultHasher::new();

hasher.write(data);

hasher.finish()

}

#[repr(C)]

pub struct mn\_prove\_input {

pub params: \*const u8,

pub params\_len: usize,

pub prover\_key: \*const u8,

pub prover\_key\_len: usize,

pub preproc: \*const u8,

pub preproc\_len: usize,

pub rng\_seed: \*const u8,

}

#[repr(C)]

pub struct mn\_prove\_output {

pub out: \*mut u8,

pub out\_len: \*mut usize,

}

#[allow(non\_camel\_case\_types)]

type mn\_prove\_fn = unsafe extern "C" fn(\*const mn\_prove\_input, \*mut mn\_prove\_output) -> i32;

#[allow(non\_camel\_case\_types)]

type mn\_last\_error\_fn = unsafe extern "C" fn() -> \*const c\_char;

pub struct NativePlonkApi {

\_lib: Library,

mn\_prove: Symbol<'static, mn\_prove\_fn>,

mn\_last\_error: Option<Symbol<'static, mn\_last\_error\_fn>>,

packed\_polys\_cache: Mutex<Option<PackedPolysCache>>,

}

struct PackedPolysCache {

params\_ptr: \*const u8,

params\_len: usize,

params\_hash: u64,

prover\_key\_ptr: \*const u8,

prover\_key\_len: usize,

prover\_key\_hash: u64,

preproc\_ptr: \*const u8,

preproc\_len: usize,

preproc\_hash: u64,

rng\_seed: [u8; 32],

packed: Arc<Vec<u8>>,

}

impl PackedPolysCache {

fn matches(

&self,

params: &[u8],

prover\_key: &[u8],

preproc: &[u8],

rng\_seed: &[u8; 32],

) -> bool {

if self.rng\_seed != \*rng\_seed {

return false;

}

if self.params\_len != params.len()

|| self.prover\_key\_len != prover\_key.len()

|| self.preproc\_len != preproc.len()

{

return false;

}

let params\_ptr\_match = self.params\_ptr == params.as\_ptr();

let prover\_key\_ptr\_match = self.prover\_key\_ptr == prover\_key.as\_ptr();

let preproc\_ptr\_match = self.preproc\_ptr == preproc.as\_ptr();

if params\_ptr\_match && prover\_key\_ptr\_match && preproc\_ptr\_match {

return true;

}

if !params\_ptr\_match && self.params\_hash != hash\_bytes(params) {

return false;

}

if !prover\_key\_ptr\_match && self.prover\_key\_hash != hash\_bytes(prover\_key) {

return false;

}

if !preproc\_ptr\_match && self.preproc\_hash != hash\_bytes(preproc) {

return false;

}

true

}

}

impl NativePlonkApi {

pub fn load() -> Result<Self> {

// Default name; can be overridden by LD\_LIBRARY\_PATH.

let lib = unsafe { Library::new("libmidnight\_native\_plonk.so") }

.map\_err(|err| anyhow!("failed to load native PLONK library: {err}"))?;

let mn\_prove = unsafe { lib.get::<mn\_prove\_fn>(b"mn\_prove\0") };

let mn\_last\_error = unsafe { lib.get::<mn\_last\_error\_fn>(b"mn\_last\_error\0") };

// Safety: we extend symbol lifetimes to match the library we own.

let mn\_prove = unsafe { std::mem::transmute::<\_, Symbol<'static, mn\_prove\_fn>>(mn\_prove?) };

let mn\_last\_error = mn\_last\_error

.ok()

.map(|sym| unsafe { std::mem::transmute::<\_, Symbol<'static, mn\_last\_error\_fn>>(sym) });

Ok(Self {

\_lib: lib,

mn\_prove,

mn\_last\_error,

packed\_polys\_cache: Mutex::new(None),

})

}

pub fn prove(

&self,

params: &[u8],

prover\_key: &[u8],

preproc: &[u8],

rng\_seed: &[u8; 32],

out: &mut [u8],

) -> Result<usize> {

if out.is\_empty() {

return Err(anyhow!("output buffer must be non-empty"));

}

let timing = timing\_enabled();

let total\_start = if timing { Some(Instant::now()) } else { None };

let mut pack\_params\_time = None;

let mut pack\_pk\_time = None;

let mut pack\_polys\_time = None;

let mut mn\_prove\_time = None;

let packed\_params = if params.starts\_with(packed\_params\_magic()) {

Cow::Borrowed(params)

} else {

let start = if timing { Some(Instant::now()) } else { None };

let packed = pack\_params(params).context("failed to pack params")?;

if let Some(start) = start {

pack\_params\_time = Some(start.elapsed());

}

Cow::Owned(packed)

};

let packed\_pk = if prover\_key.starts\_with(packed\_pk\_magic()) {

Cow::Borrowed(prover\_key)

} else {

let start = if timing { Some(Instant::now()) } else { None };

let packed = pack\_prover\_key(prover\_key).context("failed to pack prover key")?;

if let Some(start) = start {

pack\_pk\_time = Some(start.elapsed());

}

Cow::Owned(packed)

};

let cached\_polys = self

.packed\_polys\_cache

.lock()

.ok()

.and\_then(|guard| guard.as\_ref().and\_then(|entry| {

if entry.matches(params, prover\_key, preproc, rng\_seed) {

Some(entry.packed.clone())

} else {

None

}

}));

let packed\_polys = if let Some(packed) = cached\_polys {

packed

} else {

let pack\_polys\_start = if timing { Some(Instant::now()) } else { None };

let packed = Arc::new(

pack\_polys(packed\_params.as\_ref(), packed\_pk.as\_ref(), preproc, rng\_seed)

.context("failed to pack witness polynomials")?,

);

if let Some(start) = pack\_polys\_start {

pack\_polys\_time = Some(start.elapsed());

}

if let Ok(mut guard) = self.packed\_polys\_cache.lock() {

\*guard = Some(PackedPolysCache {

params\_ptr: params.as\_ptr(),

params\_len: params.len(),

params\_hash: hash\_bytes(params),

prover\_key\_ptr: prover\_key.as\_ptr(),

prover\_key\_len: prover\_key.len(),

prover\_key\_hash: hash\_bytes(prover\_key),

preproc\_ptr: preproc.as\_ptr(),

preproc\_len: preproc.len(),

preproc\_hash: hash\_bytes(preproc),

rng\_seed: \*rng\_seed,

packed: packed.clone(),

});

}

packed

};

let mut out\_len = out.len();

let input = mn\_prove\_input {

params: packed\_params.as\_ref().as\_ptr(),

params\_len: packed\_params.as\_ref().len(),

prover\_key: packed\_pk.as\_ref().as\_ptr(),

prover\_key\_len: packed\_pk.as\_ref().len(),

preproc: packed\_polys.as\_ptr(),

preproc\_len: packed\_polys.len(),

rng\_seed: rng\_seed.as\_ptr(),

};

let mut output = mn\_prove\_output {

out: out.as\_mut\_ptr(),

out\_len: &mut out\_len,

};

let mn\_prove\_start = if timing { Some(Instant::now()) } else { None };

let rc = unsafe { (self.mn\_prove)(&input, &mut output) };

if let Some(start) = mn\_prove\_start {

mn\_prove\_time = Some(start.elapsed());

}

if rc != 0 {

let msg = self

.mn\_last\_error

.as\_ref()

.and\_then(|f| {

let ptr = unsafe { f() };

if ptr.is\_null() {

None

} else {

Some(unsafe { std::ffi::CStr::from\_ptr(ptr) }.to\_string\_lossy().into\_owned())

}

})

.unwrap\_or\_else(|| "native mn\_prove failed".to\_string());

return Err(anyhow!(msg));

}

if out\_len > out.len() {

return Err(anyhow!("native mn\_prove returned invalid length"));

}

if timing {

log\_duration("pack\_params", pack\_params\_time);

log\_duration("pack\_prover\_key", pack\_pk\_time);

log\_duration("pack\_polys", pack\_polys\_time);

log\_duration("mn\_prove", mn\_prove\_time);

if let Some(start) = total\_start {

log\_duration("total", Some(start.elapsed()));

}

}

Ok(out\_len)

}

}
