#pragma once

#include <stddef.h>
#include <stdint.h>

#ifdef __cplusplus
extern "C" {
#endif

typedef struct {
  const uint8_t *params;
  size_t params_len;
  const uint8_t *prover_key;
  size_t prover_key_len;
  const uint8_t *preproc;
  size_t preproc_len;
  const uint8_t *rng_seed;
} mn_prove_input;

typedef struct {
  uint8_t *out;
  size_t *out_len;
} mn_prove_output;

int mn_prove(const mn_prove_input *in, mn_prove_output *out);
const char *mn_last_error(void);

typedef struct {
  const uint8_t *scalars;
  size_t scalars_len;
  const uint8_t *points;
  size_t points_len;
  uint8_t *out;
  size_t out_len;
} mn_msm_input;

typedef struct {
  uint8_t *values;
  size_t values_len;
  const uint8_t *omega;
  uint32_t log_n;
} mn_fft_input;

int mn_msm(const mn_msm_input *in);
int mn_fft(const mn_fft_input *in);

#ifdef __cplusplus
}
#endif
