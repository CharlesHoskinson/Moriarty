def read_bounded(path, cap):
    """No-follow bounded read of one regular file; any irregularity is an ERROR (STOP)."""
    fd = os.open(os.fsencode(str(path)),
                 os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC | os.O_NOCTTY)
    try:
        if not stat.S_ISREG(os.fstat(fd).st_mode):
            raise RuntimeError("STOP: not a regular file: %s" % path)
        chunks, total = [], 0
        while total <= cap:
            chunk = os.read(fd, min(READ_CHUNK, cap + 1 - total))
            if not chunk:
                break
            chunks.append(chunk)
            total += len(chunk)
    finally:
        os.close(fd)
    if total > cap:
        raise RuntimeError("STOP: file exceeds read cap: %s" % path)
    return b"".join(chunks)

def b2(data):
    return hashlib.blake2b(data, digest_size=32).digest()


def flip1(data):
    return bytes([data[0] ^ 1]) + data[1:]


def swap1(hex_text):
    return ("1" if hex_text[0] == "0" else "0") + hex_text[1:]

def uint(n):
    if n < 24:
        return bytes([n])
    for ai, width in ((24, 1), (25, 2), (26, 4), (27, 8)):
        if n < 1 << 8 * width:
            return bytes([ai]) + n.to_bytes(width, "big")

def walk_strings(obj):
    if type(obj) is str:
        yield obj
    elif type(obj) is dict:
        for key, value in obj.items():
            yield key
            yield from walk_strings(value)
    elif type(obj) is list:
        for item in obj:
            yield from walk_strings(item)


def tree(root):
    """Sorted no-follow listing: directories end in '/', symlinks in '@'."""
    found = []
    for top, dirs, files in os.walk(root):
        for name in dirs + files:
            path = os.path.join(top, name)
            mode = os.lstat(path).st_mode
            mark = "/" if stat.S_ISDIR(mode) else "@" if stat.S_ISLNK(mode) else ""
            found.append(os.path.relpath(path, root) + mark)
    return sorted(found)

class R4TestBase(unittest.TestCase):
    """Shared helpers only; no test methods. Wrappers touch only r2, host_codec, legacy."""

    def r4(self):
        return load_r4(self)

    def plan(self, export=None, setup=None, expiry=K6_EXPIRY, b9=None):
        return self.r4().plan_outputs(RAW[B9] if b9 is None else b9,
                                      RAW[EXP] if export is None else export,
                                      RAW[SET] if setup is None else setup, expiry)

    def refused(self, code, fn, *args, **kwargs):
        with self.assertRaises(ValueError) as caught:
            fn(*args, **kwargs)
        error = caught.exception
        self.assertEqual(len(error.args), 1, repr(error.args)[:300])
        message = error.args[0]
        self.assertIs(type(message), str)
        self.assertTrue(message.startswith(code + ": "), message[:300])
        self.assertFalse("\n" in message or "\r" in message, message[:300])
        return message

    def run_argv(self, argv):
        r4 = self.r4()
        out, err = io.StringIO(), io.StringIO()
        with contextlib.redirect_stdout(out), contextlib.redirect_stderr(err):
            code = r4.main(argv)
        return code, out.getvalue(), err.getvalue()

    def run_main(self, *args):
        return self.run_argv([R4_FILE] + [str(a) for a in args])

    def assert_cli_refusal(self, result, code):
        exit_code, out, err = result
        self.assertEqual((exit_code, out), (2, ""))
        self.assertTrue(err.startswith(code + ": ") and err.endswith("\n")
                        and err.count("\n") == 1, err[:300])

    def assert_cli_success(self, result, output):
        exit_code, out, err = result
        self.assertEqual((exit_code, err, out.count("\n")), (0, "", 1))
        self.assertTrue(out.endswith("\n"))
        self.assertEqual(json.loads(out), {"output": str(output), "cases": 4})

    @contextlib.contextmanager
    def wrapped(self, module, name, replacement):
        if not any(module is m for m in (r2, host_codec, legacy)):
            raise RuntimeError("STOP: wrappers are limited to r2, host_codec and legacy")
        original = getattr(module, name)
        setattr(module, name, replacement)
        try:
            yield original
        finally:
            setattr(module, name, original)

    @contextlib.contextmanager
    def counted(self, module, name):
        calls, original = [], getattr(module, name)

        def counter(*args, **kwargs):
            calls.append((args, kwargs))
            return original(*args, **kwargs)
        with self.wrapped(module, name, counter):
            yield calls

    @contextlib.contextmanager
    def faulted(self, module, name, error):
        calls = []

        def fault(*args, **kwargs):
            calls.append((args, kwargs))
            raise error
        with self.wrapped(module, name, fault):
            yield calls

    @contextlib.contextmanager
    def consumer_wrapper(self, after):
        """Delegate to r2.prepare_c2_issue, then call after(result) before returning it."""
        original = r2.prepare_c2_issue

        def wrapper(*args, **kwargs):
            result = original(*args, **kwargs)
            after(result)
            return result
        with self.wrapped(r2, "prepare_c2_issue", wrapper):
            yield

    def case_scratch(self):
        path = SCRATCH / self.id().rsplit(".", 1)[-1]
        os.mkdir(path, 0o700)
        return path

    def copy_inputs(self, dest):
        paths = []
        for role, rel in INPUT_ROLES:
            path = Path(dest) / (role + ".json")
            fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC,
                         0o644)
            try:
                view = memoryview(RAW[rel])
                while view:
                    view = view[os.write(fd, view):]
            finally:
                os.close(fd)
            paths.append(path)
        return tuple(paths)

    def assert_inputs_unchanged(self):
        for _role, rel in INPUT_ROLES:
            self.assertEqual(read_bounded(HERE / rel, PINS[rel][0]), RAW[rel], rel)
