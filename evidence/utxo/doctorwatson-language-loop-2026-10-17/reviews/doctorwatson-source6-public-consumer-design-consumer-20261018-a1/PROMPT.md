+            diagnostics::shape(
+                checked.ast.span(checked.values.node(value)),
+                &format!("the checked `{name}` field"),
+            )
+        })
+}
+
+fn string(checked: &Checked<'_>, value: ValueIdx) -> Result<String> {
+    if checked.values.tag(value) != ValueTag::String {
+        return Err(diagnostics::shape(
+            checked.ast.span(checked.values.node(value)),
+            "a checked string",
+        ));
+    }
+    let token = TokenIdx::new(usize::from(checked.values.rows.a()[value.index()]));
+    Ok(decode_string(checked.ast.token_text(token)))
+}
+
+fn number(checked: &Checked<'_>, value: ValueIdx, tag: ValueTag) -> Result<u128> {
+    if checked.values.tag(value) != tag {
+        return Err(diagnostics::shape(
+            checked.ast.span(checked.values.node(value)),
+            "a checked scalar or nominal quantity in its declared role",
+        ));
+    }
+    Ok(checked.values.amount(value))
+}
+
+fn id(checked: &Checked<'_>, entity: ValueIdx) -> Result<String> {
+    string(checked, field(checked, entity, "id")?)
+}
+
+fn call_name(checked: &Checked<'_>, value: ValueIdx) -> String {
+    let mut token = checked.ast.main_token(checked.values.node(value));
+    let mut name = checked.ast.token_text(token).to_owned();
+    while checked.ast.tokens.kind(checked.ast.token_after(token, 1)) == TokenKind::Dot {
+        token = checked.ast.token_after(token, 2);
+        name.push('.');
+        name.push_str(checked.ast.token_text(token));
+    }
+    name
+}
+
+fn extract(checked: &Checked<'_>, action: &str) -> Result<(Terms, Span)> {
+    let ast = checked.ast;
+    let index = checked.actions.node().iter().position(|&node| {
+        ast.token_text(ast.token_after(ast.main_token(node), 1)) == action
+    });
+    let index = index.ok_or_else(|| diagnostics::action(ast.span(NodeIdx::ROOT), action))?;
+    let node = checked.actions.node()[index];
+    let span = ast.span(node);
+    if checked.actions.support()[index] != Some(Support::LocalS0) {
+        return Err(diagnostics::unsupported(span));
+    }
+    let intent = checked.actions.intent()[index]
+        .and_then(|decl| checked.value(decl))
+        .ok_or_else(|| diagnostics::shape(span, "a resolved checked intent"))?;
+    let operation = field(checked, intent, "operation")?;
+    if call_name(checked, operation) != "transfer" {
+        return Err(diagnostics::unsupported(span));
+    }
+    let asset = field(checked, intent, "asset")?;
+    let validity = field(checked, intent, "valid")?;
+    let quantity = |record, name| {
+        number(checked, field(checked, record, name)?, ValueTag::Quantity)
+    };
+    let scalar = |record, name| {
+        number(checked, field(checked, record, name)?, ValueTag::Scalar)
+    };
+    let text = |name| string(checked, field(checked, intent, name)?);
+    let value = quantity(operation, "value")?;
+    let fee = quantity(operation, "fee")?;
+    let gross = value
+        .checked_add(fee)
+        .ok_or_else(|| diagnostics::shape(span, "a UInt128 transfer total"))?;
+    Ok((
+        Terms {
+            agreement: ast.token_text(ast.main_token(NodeIdx::ROOT)).to_owned(),
+            domain: id(checked, field(checked, intent, "domain")?)?,
+            asset: id(checked, asset)?,
+            scale: scalar(asset, "scale")?,
+            signer: id(checked, field(checked, intent, "signer")?)?,
+            owner: id(checked, field(checked, operation, "from")?)?,
+            recipient: id(checked, field(checked, operation, "to")?)?,
+            fee_recipient: id(checked, field(checked, operation, "fee_to")?)?,
+            key: text("key")?,
+            nonce: text("nonce")?,
+            pre_head: text("pre_head")?,
+            valid_from: scalar(validity, "from")?,
+            valid_to: scalar(validity, "to")?,
+            gross_cap: quantity(intent, "gross_cap")?,
+            fee_cap: quantity(intent, "fee_cap")?,
+            net_floor: quantity(intent, "net_floor")?,
+            value,
+            fee,
+            gross,
+            source_hash: text("source_hash")?,
+            policy_digest: text("policy_digest")?,
+        },
+        span,
+    ))
+}
+
+fn validate_scenario(terms: &Terms, s: &LocalTransferScenario<'_>, span: Span) -> Result<()> {
+    for (field, actual, expected) in [
+        ("domain", s.domain, terms.domain.as_str()),
+        ("asset", s.asset, terms.asset.as_str()),
+        ("allowance.owner", s.allowance.owner, terms.signer.as_str()),
+    ] {
+        if actual != expected {
+            return Err(diagnostics::scenario(span, field, expected));
+        }
+    }
+    for (index, (balance, expected)) in s.balances.iter().zip([
+        terms.owner.as_str(),
+        terms.recipient.as_str(),
+        terms.fee_recipient.as_str(),
+    ]).enumerate() {
+        if balance.account != expected {
+            return Err(diagnostics::scenario(span, &format!("balances[{index}].account"), expected));
+        }
+    }
+    for (field, text) in [
+        ("head", s.head),
+        ("predecessor", s.predecessor),
+        ("post_head", s.post_head),
+    ] {
+        if text.is_empty() || text.len() > mori_lexer::MAX_STRING_BYTES {
+            return Err(diagnostics::scenario(span, field, "nonempty text of at most 1024 UTF-8 bytes"));
+        }
+    }
+    if s.allowance.remaining.checked_add(s.allowance.spent).is_none()
+        || s.work_remaining.checked_add(s.work_spent).is_none()
+    {
+        return Err(diagnostics::scenario(span, "counters", "opening totals that fit UInt128"));
+    }
+    Ok(())
+}
+
+fn quote(text: &str) -> String {
+    serde_json::to_string(text).expect("serializing a UTF-8 string cannot fail")
+}
+
+fn append_line(out: &mut String, line: String, span: Span) -> Result<()> {
+    if out.len() + line.len() + 1 > mori_lexer::MAX_SOURCE_BYTES {
+        return Err(diagnostics::bound(span));
+    }
+    out.push_str(&line);
+    out.push('\n');
+    Ok(())
+}
+
+fn emit(t: &Terms, s: &LocalTransferScenario<'_>, span: Span) -> Result<String> {
+    let mut out = String::new();
+    macro_rules! line {
+        ($($arg:tt)*) => {
+            append_line(&mut out, format!($($arg)*), span)?
+        };
+    }
+    let statement = format!(
+        "transfer from {} to {} fee_to {} value {} fee {}",
+        t.owner, t.recipient, t.fee_recipient, t.value, t.fee,
+    );
+    line!("profile \"moriarty-financial-agreement-source/6\";");
+    line!("agreement {} {{", t.agreement);
+    line!("domain {};", t.domain);
+    line!("settlement {} scale {};", t.asset, t.scale);
+    line!("selected TransferLiteralFee source_hash {} digest {};", quote(&t.source_hash), quote(&t.policy_digest));
+    line!("intent {{");
+    line!("signer {} key {};", t.signer, quote(&t.key));
+    line!("nonce {};", quote(&t.nonce));
+    line!("pre_head {};", quote(&t.pre_head));
+    line!("valid {}..{};", t.valid_from, t.valid_to);
+    line!("gross_cap {};", t.gross_cap);
+    line!("fee_cap {};", t.fee_cap);
+    line!("net_floor {};", t.net_floor);
+    line!("failure success_only;");
+    line!("signed_action {statement};");
+    for field in ["observations", "disclosures", "retained_effects", "retained_duties"] {
+        line!("{field} empty;");
+    }
+    for field in ["delegation", "recovery"] {
+        line!("{field} none;");
+    }
+    line!("}}");
+    line!("authenticated {{");
+    line!("head {};", quote(s.head));
+    line!("predecessor {};", quote(s.predecessor));
+    line!("round {};", s.round);
+    for balance in &s.balances {
+        line!("balance {} {};", balance.account, balance.amount);
+    }
+    line!("allowance {} remaining {} spent {};", s.allowance.owner, s.allowance.remaining, s.allowance.spent);
+    let replay = match s.replay {
+        Replay::Unused => "unused",
+        Replay::Consumed => "consumed",
+    };
+    line!("replay {replay};");
+    line!("work_remaining {};", s.work_remaining);
+    line!("work_spent {};", s.work_spent);
+    line!("}}");
+    line!("submit {statement};");
+    line!("effects {{");
+    line!("debit {} {};", t.owner, t.gross);
+    line!("credit {} {};", t.recipient, t.value);
+    if t.fee > 0 {
+        line!("credit {} {};", t.fee_recipient, t.fee);
+    }
+    line!("use_allowance {} {};", t.signer, t.gross);
+    line!("use_replay {};", quote(&t.nonce));
+    line!("advance_head {} {};", quote(&t.pre_head), quote(s.post_head));
+    line!("}}");
+    line!("post_head {};", quote(s.post_head));
+    line!("}}");
+    Ok(out)
+}

+# Proposed explicit Rust beta Source/6 transfer expansion
+
+Status: unadopted interface/source proposal. No compile, runtime, native or release qualification is established. The `source6-expansion` Cargo feature enables the explicit local transfer presentation path; it is separate from `native` and does not alter default features.
+
+Existing default command, unchanged:
+
+```sh
+mori expand core.mori --action noop --scenario core.json
+```
+
+This still validates a Core action against its Core scenario and prints its generic Core Program JSON. A Rust beta file supplied without the new option still reaches the existing Core parser and refuses; no default beta output is replaced. The separate Node beta package and its default JSON expansion output are unchanged.
+
+Proposed explicit command, in a build with `source6-expansion` enabled:
+
+```sh
+mori check invoice.mori --json
+mori --quiet expand invoice.mori --action pay --scenario scenario.json --emit-source6-transfer
+```
+
+The explicit command reads a typed local transfer scenario and calls `mori-source6::expand_transfer`, which uses the real Rust beta parser and checker. Stdout is the actual generated Source/6 text with its existing final newline; `--quiet` leaves stderr empty. Without `--quiet`, stderr reports `Expanded Source/6 transfer proposal (local-stipulation-only)`. It does not emit a Core success result. Without the build feature, the explicit option reports `mori::command::source6-unavailable`.
+
+The proposed scenario JSON is the closed **transfer subset** of the existing `moriarty-local-scenario/1` / `local-stipulation` shape. It requires domain, asset, head, predecessor, round, exactly three ordered balances, allowance, replay, work_remaining, work_spent and post_head. Numeric values are canonical decimal strings parsed through the existing Core UInt decoder; no new decimal evaluator is introduced. Balance order is owner, recipient, fee recipient. Replay is `unused` or `consumed`. Obligation and candidate_effects overrides are outside this option. The current strict CLI reader and existing read/JSON bounds apply.
+
+This profile-token reuse, transfer subset, option name, feature name, raw stdout/status output, and JSON-decode-before-helper error ordering are **new unadopted public interface choices**. They require the planned independent design/interface votes; they do not establish full Node beta wire or diagnostic parity. The typed borrowed local scenario passed to the library is an internal input, not proof or authentication.
+
+Boundary: Rust beta/1 → proposed transfer presentation → Source/6 text. The real experimental TypeScript `parseAndLowerSource6` must still consume actual emitted bytes; its Core/5 preparation remains separately unqualified. This option provides no Source/6/Core/5→Core/1/Core/2/Sherlock/native embedding, no handwritten target IR and no target/proof/post-state success. The default Core and separate Node consumers retain their own profiles.

use std::path::PathBuf;
use std::process::ExitCode;
use std::time::Instant;

use mori_diagnostics::{MoriDiagnostic, NamedSource};

use crate::json::CheckReport;
use crate::shell::Shell;

#[derive(Debug, clap::Args)]
pub struct Args {
    /// Source file
    pub file: PathBuf,

    /// Print the result as JSON
    #[arg(long)]
    pub json: bool,
}

/// Checks syntax, names, values, types, declarations, calls and intents.
pub fn run(args: Args, shell: &mut Shell) -> super::Result<ExitCode> {
    let name = args.file.display().to_string();
    let source = super::core_io::read_source(&args.file)?;

    if !args.json {
        shell.status("Checking", &name);
    }
    let started = Instant::now();
    if mori_lexer::first_token_text(&source).is_ok_and(|token| token == "language") {
        let staged = mori_core::language(&source).is_ok_and(|language| language == "moriarty/2");
        let result = if staged {
            mori_core::stage::load(&source).map(|program| program.program().clone())
        } else {
            mori_core::parse(&source)
        };
        if args.json {
            let report = match &result {
                Ok(program) => {
                    serde_json::json!({"status":"accepted", "file":name,"source_hash":mori_core::intent::hash_bytes(source.as_bytes()),"language":program.language,"agreement":program.name,"actions":program.actions,"diagnostics":[]})
                }
                Err(error) => {
                    serde_json::json!({"status":"rejected","file":name,"diagnostics":[error]})
                }
            };
            if staged {
                super::simulate::print_stage_report(report)?;
            } else {
                super::core_io::print(&report)?;
            }
        } else {
            match &result {
                Ok(_) => shell.status("Finished", "bounded core check"),
                Err(error) => shell.error(error),
            }
        }
        return Ok(if result.is_ok() {
            ExitCode::SUCCESS
        } else {
            ExitCode::FAILURE
        });
    }
    let (accepted, report) = match mori_parser::parse(&source) {
        Err(error) => {
            let diagnostics = vec![error];
            let report = args
                .json
                .then(|| CheckReport::new(name.clone(), &source, None, &diagnostics));
            (
                finish(shell, &name, &source, diagnostics, args.json),
                report,
            )
        }
        Ok(ast) => {
            if !args.json {
                shell.verbose_status(
                    "Parsed",
                    format!("{} tokens, {} nodes", ast.tokens.len(), ast.nodes.len()),
                );
            }
            let result = mori_checker::check(&ast);
            let report = args.json.then(|| {
                CheckReport::new(
                    name.clone(),
                    &source,
                    Some(&result.checked),
                    &result.diagnostics,
                )
            });
            (
                finish(shell, &name, &source, result.diagnostics, args.json),
                report,
            )
        }
    };

    if let Some(report) = report {
        super::core_io::print(&report)?;
    } else if accepted {
        shell.status(
            "Finished",
            format!("check in {:.2}s", started.elapsed().as_secs_f64()),
        );
    }
    Ok(if accepted {
        ExitCode::SUCCESS
    } else {
        ExitCode::FAILURE
    })
}

/// Renders the diagnostics, unless the report is JSON. True when there are none.
fn finish(
    shell: &mut Shell,
    name: &str,
    source: &str,
    diagnostics: Vec<MoriDiagnostic>,
    json: bool,
) -> bool {
    if diagnostics.is_empty() {
        return true;
    }
    if !json {
        let count = diagnostics.len();
        for diagnostic in diagnostics {
            shell.report(&diagnostic.with_source_code(NamedSource::new(
                crate::display::text(name),
                source.to_owned(),
            )));
        }
        let plural = if count == 1 { "" } else { "s" };
        shell.error(format!(
            "could not check `{name}` due to {count} previous error{plural}"
        ));
    }
    false
}

# Financial agreement Source/6 and Core/5 S0 contract

**Status:** provisional Sprint 0 specification, 2026-09-29. The grammar is a closed presentation syntax for one S0 stage proposal, and an isolated Source/6 parser and local Core/5 preparer implement this first slice. It is not a signed `/3` encoding or an adopted MIL/4 profile. W-D0–W-D4 remain open. See the [MIL/4 semantic contract](../../formal/mil4/semantics-contract.md) and [S0 implementation contract](../../formal/mil4/s0-implementation-contract.md).

## Formation and version gate

The [EBNF](financial-agreement-source-v6-grammar.ebnf) parses exactly one `profile`, one `agreement`, and one stage proposal. Its header must contain the raw token `"moriarty-financial-agreement-source/6"` before parsing the agreement body; a differently escaped spelling is outside this provisional presentation. The fixed declaration and field order is presentation syntax, not a claim about signed bytes. Unknown, repeated, omitted or out-of-order fields reject at Source/6 formation with `SOURCE6_SHAPE`; an unknown action or effect tag rejects with `SOURCE6_UNKNOWN_TAG`. A different header rejects with `SOURCE6_VERSION`. The closed source grammar also rejects a nonempty failure, observation, disclosure, retained-effect or duty form at formation with `SOURCE6_SHAPE`; `S0_FAILURE_UNSUPPORTED` applies only when a typed Core/5 stage reaches the failure judgment. These source rejections have no Core/5 term or K transition. A local Source/5 parser test confirms that a `/6` header rejects with `PROFILE_MISMATCH`; Core/4-to-Core/5 compatibility has not been implemented or verified here.

The grammar borrows only the token definitions from `lexical.md`; all literal words in the Source/6 EBNF are profile-local reserved words and cannot be identifiers. This does not alter Source/5 keywords. Limits are simultaneous: source UTF-8 bytes ≤65536, tokens ≤8192, AST nodes ≤8192, nesting depth ≤64, identifiers ≤64 ASCII characters, decoded strings ≤1024 UTF-8 bytes, and one stage/effect vector per document. Opaque string fields in this S0 presentation must be nonempty. `scale` is 0..18; each nominal amount (`value`, `fee`, `amount`, caps and floor) is 0..`2^127−1`; each balance, allowance counter and work count is 0..`2^128−1`. S0 uses checked UInt128 intermediates, including `value+fee`, `principal+accrued`, credits and spent counters. The current source/kernel nominal bound does not by itself impose the same bound on every lifecycle state field. This S0 proposal additionally caps principal, accrued and outstanding at `2^127−1` and requires `outstanding=principal+accrued`. W-D4 must review that additional narrowing. Invalid bounds, including an inverted validity interval, reject before K admission with `SOURCE6_RANGE`.

Source/6 formation applies an action-dependent shape check after the closed EBNF parses: transfer requires three balance rows in owner, recipient, fee-recipient order and no obligation; repay requires two rows in payer, bound-creditor order and exactly one obligation. All endpoint identifiers are pairwise distinct for transfer; payer and creditor differ for repay. Duplicate or missing authenticated cells reject `SOURCE6_CELL_SHAPE`; an absent receiver balance is not silently initialized. The allowance owner equals signer. The replay row states whether the selected signed key is unused or consumed; a consumed row rejects at history. The authenticated block is a *claim* until a snapshot-to-head premise establishes each cell and the current head. The strings used for key, nonce, digest and heads are opaque typed identifiers in this syntax, not hash byte definitions.

## Source fields to Core/5

One elaboration proposes `Core5Stage(pre, action, signedScope, submittedEffects, premises)`. The current direct parser preserves agreement ID, asset scale and authenticated predecessor on its AST, but the local Core preparer does not bind those three fields or an exact `/3` digest. It reports the candidate as `PreparedUnqualified`. The Core/5 tags below are proposed typed constructors. No caller Boolean can establish authentication or signature validity.

The local intent object carries the proposed version label `moriarty-intent/3`. That label is a type discriminator only. Source/6 lowering does not encode canonical `/3` bytes or calculate their digest, and no verifier consumes this object as a signed authorization.

For transfer, prepare `Debit(owner,v+f)`, `Credit(recipient,v)`, optional `Credit(feeRecipient,f)` when `f>0`, `UseAllowance(owner,v+f)`, `UseReplay(key)`, `AdvanceHead(pre,post)` in exactly that order. The optional line is present iff `f>0`; a zero-valued fee line rejects. Require `v>0`, `f≤feeCap`, `v+f≤grossCap`, `v≥netFloor`, sufficient owner balance and remaining allowance, and no overflow in any receiver or spent counter. Debit and credits conserve the same nominal asset. Preserve gross effects even though balance changes could be netted. The local wrapper checks Stage first, then requires the transfer in `submit` to equal `signed_action`; it withholds any candidate prepared during that local check when they differ.

For repay, require one Outstanding obligation, `0<n≤outstanding`, matching asset, identity conversion `(mantissa=1,scale=0,rounding=none)`, `feeCap=0`, `netFloor=0`, sufficient payer balance and allowance, and no overflow in the creditor or spent counter. Let `da=min(n,accrued)` and `dp=n−da`. Prepare `Debit(payer,n)`, `Credit(boundCreditor,n)`, `SetObligation(p−dp,a−da,p+a−n,status')`, `UseAllowance(payer,n)`, `UseReplay(key)`, `AdvanceHead(pre,post)`. `status'` is Settled iff outstanding becomes zero. Every unlisted authenticated cell is unchanged. A debt reduction without the creditor credit rejects. The local wrapper checks Stage first, then requires the repayment in `submit` to equal `signed_action`; it withholds any candidate prepared during that local check when they differ.

## Admission and observation

Core/5 applies `stage → intent → effect → authority → history → failure`. `stage` checks version, selected program, typed cells and authentication premises. `intent` checks exact signed scope, validity, endpoints, alias policy and all signed nominal caps and floors. `effect` computes and compares the complete ordered vector and post cells. `authority` checks signer, allowance and work budget. `history` compares the current head and unused replay key and verifies successor binding. `failure` accepts only terminal success with empty retained effects and duties. The first failing judgment returns `(judgment,code,diagnosticWork)` without a published post-state or effects. Within-judgment code spelling and precedence remain W-D3 choices; provisional codes are in `s0-implementation-contract.md`.

The result shape is `Core5Observation(pre,action,signedScope,preparedEffects,post,remainingDuty,remainingWork,replay,preHead,postHead,phase,judgment,code)`. Accepted S0 success contains complete effects, consumption, writes, one post-head and empty duty. Atomic rejection contains the first judgment/code and diagnostic work, with `post`, published effects and post-head absent. Snapshot authentication, exact signature verification, head extension and ledger compare-and-consume are typed external premises; failure or absence rejects. Source/6 grammar acceptance alone does not imply admission.

## Boundary with earlier and later forms

Source/5 `profile`, declarations, `action`, expression and `emit` forms have no automatic injection into this S0 stage. A future migration must map every legacy field, selected Core/4 program, authenticated state, signed digest and complete effect obligation, then prove the old and new observations equivalent on the stated domain. `Repay` in Core/4 consumes a separate Transfer; Core/5's `RepayAccrualFirst` is one funded stage. All old Source/5 and Core/4 behavior remains historical until such a mapping is demonstrated.

The eight MIL/4 first families and all later profiles reject from this S0 grammar with `SOURCE6_PROFILE_UNSUPPORTED` or `SOURCE6_UNKNOWN_TAG` at formation. A later full Source/6 grammar must add typed productions and Core/5 constructors per family. General Φ₁, uncertified Ω, division, rounding, mint, reserve, foreign evidence, accepted failures and recovery are outside S0. This document does not turn those forms into generic records or strings.
