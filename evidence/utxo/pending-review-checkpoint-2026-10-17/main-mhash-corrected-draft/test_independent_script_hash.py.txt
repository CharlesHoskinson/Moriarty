def _read(name):

def _strict_json(raw):
    def hook(pairs):
        keys = [k for k, _ in pairs]
        if len(keys) != len(set(keys)):
            raise ValueError("duplicate JSON key in fixture")
        return dict(pairs)
    return json.loads(raw.decode("utf-8"), object_pairs_hook=hook)

def _bstr(payload):
    """Minimal definite-length CBOR major-type-2 header + payload; builds test inputs only."""
    n = len(payload)
    if n < 24:
        return bytes([0x40 + n]) + payload
    if n < 256:
        return bytes([0x58, n]) + payload
    if n < 65536:
        return bytes([0x59]) + n.to_bytes(2, "big") + payload
    return bytes([0x5a]) + n.to_bytes(4, "big") + payload


def _flip_byte(code_hex, index):
    """Replace one payload byte (hex pair at byte index) keeping CBOR framing intact."""
    i = 2 * index
    return code_hex[:i] + "%02x" % (int(code_hex[i:i + 2], 16) ^ 0x01) + code_hex[i + 2:]


def _flip_nibble(h):
    return ("1" if h[0] != "1" else "2") + h[1:]


def _impl_source():
    if not os.path.isfile(IMPL_PATH):
        raise AssertionError(RED)
    with open(IMPL_PATH, encoding="utf-8") as f:

def _static_problems(src):
    problems = []
    for node in ast.walk(ast.parse(src)):
        if isinstance(node, ast.Import):
            problems += ["import " + a.name for a in node.names if a.name.split(".")[0] not in ALLOWED_IMPORTS]
        elif isinstance(node, ast.ImportFrom):
            if node.level != 0 or (node.module or "").split(".")[0] not in ALLOWED_IMPORTS:
                problems.append("from-import %r level %d" % (node.module, node.level))
        elif isinstance(node, ast.Name) and node.id in BANNED_NAMES:
            problems.append("name " + node.id)
    return problems


def _impl():
    if "impl" not in _cache:
        src = _impl_source()
        problems = _static_problems(src)
        if problems:
            raise AssertionError("M-HASH static guard refused to execute implementation: %s" % problems)
        spec = importlib.util.spec_from_file_location("independent_script_hash", IMPL_PATH)
        mod = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(mod)
        _cache["impl"] = mod
    return _cache["impl"]


class Base(unittest.TestCase):
    def assertInputError(self, fn, *args):
        m = _impl()
        with self.assertRaises(m.ScriptHashInputError) as cm:
            fn(*args)
        self.assertNotIsInstance(cm.exception, m.ScriptHashMismatchError)

    def assertMismatch(self, raw):
        m = _impl()
        with self.assertRaises(m.ScriptHashMismatchError) as cm:
            m.check_blueprint(raw)
        self.assertNotIsInstance(cm.exception, m.ScriptHashInputError)

    def assertHex56(self, h):
        self.assertIs(type(h), str)
        self.assertEqual(len(h), 56)
        self.assertTrue(all(c in "0123456789abcdef" for c in h), h)

    def test_hex_and_type_inputs(self):
        m = _impl()
        # CORRECTION (ROOT-CAUSE.md): the frozen literal CIP57_CODE[10:20] is the decimal-digit run
        # "3232322225", so CIP57_CODE[10:20].upper() was a no-op and the former "mixed-case" variant
        # equalled CIP57_CODE verbatim -- the accepted canonical input, not a mixed-case string. The
        # slice [40:50] of the same frozen literal is "2e646e6400", which upper-cases to "2E646E6400".
        # The construction below is thus a genuine mixed-case hex string of the identical bytes. The
        # three independent literal assertions pin that it really contains uppercase A-F and differs
        # from the admitted lowercase input, so this negative fixture can never silently degenerate
        # again. No implementation, framing, tag, vector, bound or oracle value is touched.
        mixed_case = CIP57_CODE[:40] + CIP57_CODE[40:50].upper() + CIP57_CODE[50:]
        self.assertNotEqual(mixed_case, CIP57_CODE)  # genuinely altered, not a digit-only no-op
        self.assertTrue(any(c in "ABCDEF" for c in mixed_case))  # contains uppercase hex A-F
        self.assertEqual(mixed_case.lower(), CIP57_CODE)  # same bytes; only letter case differs
        for label, variant in (
            ("uppercase", CIP57_CODE.upper()),
            ("mixed-case", mixed_case),
            ("leading-space", " " + CIP57_CODE),
            ("trailing-newline", CIP57_CODE + "\n"),
            ("inner-space", CIP57_CODE[:4] + " " + CIP57_CODE[4:]),
            ("0x-prefix", "0x" + CIP57_CODE),
            ("odd-length", CIP57_CODE[:-1]),
            ("non-hex", CIP57_CODE[:-2] + "zz"),
            ("fullwidth-digits", CIP57_CODE[:-2] + "１１"),
            ("empty", ""),
            ("bytes", bytes.fromhex(CIP57_CODE)),
            ("bytearray", bytearray.fromhex(CIP57_CODE)),
            ("none", None),
            ("int", 123),
            ("list", [CIP57_CODE]),

            ("non-minimal-58", "5817" + "01" * 23),
            ("non-minimal-59", "5900ff" + "01" * 255),
            ("empty-bstr", "40"),
        ):
            with self.subTest(case=label):
                self.assertInputError(m.script_hash, "v3", variant)

    def test_single_byte_substitution(self):
        m = _impl()
        _, _, codes = _b9()
        for name, ver, code, want in (("cip57", "v2", CIP57_CODE, CIP57_HASH),
                                      ("c2", "v3", codes["c2"], RECEIPTS["c2"][1]),
                                      ("s2p", "v3", codes["s2p"], RECEIPTS["s2p"][1])):
            n = len(code) // 2
            for idx in (n - 1, n // 2, 4):
                with self.subTest(case=name, byte=idx):
                    h = m.script_hash(ver, _flip_byte(code, idx))
                    self.assertHex56(h)
                    self.assertNotEqual(h, want)

    def test_code_hash_substitution(self):
        m = _impl()
        _, doc0, codes = _b9()

        def mutate(fn):
            d = copy.deepcopy(doc0)
            fn(d)
            return json.dumps(d).encode("utf-8")

        def each(d, titles, key, value):
            for v in d["validators"]:
                if v["title"] in titles:
                    v[key] = value(v[key])

        cases = (
            ("c2-code-from-vi", lambda d: each(d, GROUPS["c2"][0], "compiledCode", lambda _: codes["vi"])),
            ("s2p-hash-from-vs3", lambda d: each(d, GROUPS["s2p"][0], "hash", lambda _: RECEIPTS["vs3"][1])),
            ("one-handler-code-byte", lambda d: each(d, {"c2.c2.spend"}, "compiledCode",
                                                     lambda c: _flip_byte(c, len(c) // 2 - 1))),
            ("one-handler-hash-nibble", lambda d: each(d, {"vs4.vs4.else"}, "hash", _flip_nibble)),
            ("preamble-v2", lambda d: d["preamble"].__setitem__("plutusVersion", "v2")),
        )
        for label, fn in cases:
            with self.subTest(case=label):
                self.assertMismatch(mutate(fn))
        cip = _strict_json(_cip57_doc_bytes())
        cip["preamble"]["plutusVersion"] = "v3"
        self.assertMismatch(json.dumps(cip).encode())

        def consistent_swap(d):
            for v in d["validators"]:
                g = "vi" if v["title"] in GROUPS["c2"][0] else "c2" if v["title"] in GROUPS["vi"][0] else None
                if g:
                    v["compiledCode"], v["hash"] = codes[g], RECEIPTS[g][1]
        got = m.check_blueprint(mutate(consistent_swap))
        self.assertEqual(got["c2.c2.mint"], RECEIPTS["vi"][1])
        self.assertNotEqual(got["c2.c2.mint"], RECEIPTS["c2"][1])  # only pinned receipts detect this (C5)

    def test_blueprint_malformed(self):
        m = _impl()
        entry = {"title": "a", "compiledCode": CIP57_CODE, "hash": CIP57_HASH}
        pre = {"title": "t", "plutusVersion": "v2"}

        def doc(preamble=pre, validators=None):
            d = {"preamble": preamble, "validators": [dict(entry)] if validators is None else validators}
            return json.dumps(d).encode()

        def ent(**kw):
            e = dict(entry)
            for k, v in kw.items():
                if v is KeyError:
                    e.pop(k)
                else:
                    e[k] = v
            return [e]

        dup_entry = ('{"preamble":{"title":"t","plutusVersion":"v2"},"validators":[{"title":"a",'
                     '"compiledCode":"%s","hash":"%s","hash":"%s"}]}'
                     % (CIP57_CODE, _flip_nibble(CIP57_HASH), CIP57_HASH)).encode()
        dup_pre = ('{"preamble":{"title":"t","plutusVersion":"v3","plutusVersion":"v2"},"validators":'
                   '[{"title":"a","compiledCode":"%s","hash":"%s"}]}' % (CIP57_CODE, CIP57_HASH)).encode()
        ok = doc()
        cases = (
            ("str-input", ok.decode()),
            ("invalid-utf8", b"\xff\xfe" + ok),
            ("invalid-json", b"{"),
            ("top-level-array", b"[]"),
            ("missing-preamble", json.dumps({"validators": [entry]}).encode()),
            ("preamble-null", doc(preamble=None)),
            ("preamble-not-object", doc(preamble="v2")),
            ("missing-plutusVersion", doc(preamble={"title": "t"})),
            ("plutusVersion-v4", doc(preamble={"title": "t", "plutusVersion": "v4"})),
            ("plutusVersion-int", doc(preamble={"title": "t", "plutusVersion": 2})),
            ("missing-validators", json.dumps({"preamble": pre}).encode()),
            ("validators-object", doc(validators={"a": entry})),
            ("validators-empty", doc(validators=[])),
            ("entry-not-object", doc(validators=["a"])),
            ("missing-title", doc(validators=ent(title=KeyError))),
            ("empty-title", doc(validators=ent(title=""))),
            ("int-title", doc(validators=ent(title=1))),
            ("duplicate-title", doc(validators=[dict(entry), dict(entry)])),
            ("missing-compiledCode", doc(validators=ent(compiledCode=KeyError))),
            ("missing-hash", doc(validators=ent(hash=KeyError))),
            ("hash-width-32-bytes", doc(validators=ent(hash=CIP57_HASH + "00000000"))),
            ("hash-width-27-bytes", doc(validators=ent(hash=CIP57_HASH[:54]))),
            ("hash-uppercase", doc(validators=ent(hash=CIP57_HASH.upper()))),
            ("hash-int", doc(validators=ent(hash=5))),
            ("code-uppercase", doc(validators=ent(compiledCode=CIP57_CODE.upper()))),
            ("code-stripped", doc(validators=ent(compiledCode=CIP57_CODE[4:]))),
            ("parameters-remaining", doc(validators=ent(parameters=[{"title": "p", "schema": {}}]))),
            ("parameters-not-list", doc(validators=ent(parameters="p"))),
            ("duplicate-key-entry", dup_entry),
            ("duplicate-key-preamble", dup_pre),
            ("over-size", ok + b" " * (MAX_BLUEPRINT_BYTES + 1 - len(ok))),
            ("too-many-validators", doc(validators=[dict(entry, title="h%d" % i) for i in range(MAX_VALIDATORS + 1)])),
        )
        for label, raw in cases:
            with self.subTest(case=label):
                self.assertInputError(m.check_blueprint, raw)
