class ControlError(Exception):
    """A test-side control failed its own construction check; never raised by the implementation."""


class TripwireError(Exception):
    """Raised by a hook tripwire; deliberately not a ValueError, so a consulted hook cannot pass as refusal."""


class _Str(str):
    """Plain str subclass with no overridden behaviour."""


class _Bytes(bytes):

def _read(name):
    with open(os.path.join(PACKET, name), "rb") as f:
        return f.read()


def _strict_json_text(text):
    def pairs(items):
        keys = [k for k, _ in items]
        if len(keys) != len(set(keys)):
            raise ControlError("duplicate JSON key in control")
        return dict(items)

    def constant(name):
        raise ControlError("non-JSON constant %s in control" % name)
    return json.loads(text, object_pairs_hook=pairs, parse_constant=constant)


def _strict_json(raw):
    return _strict_json_text(raw.decode("utf-8"))


def _cip57_doc_bytes():
    """The CIP-0057 'Aiken's Hello World' JSON example, sliced verbatim from the pinned document."""
    if "base" not in _cache:
        text = _read("CIP-0057.md").decode("utf-8")
        start = text.index("```json\n", text.index("<summary>Aiken's Hello World</summary>")) + len("```json\n")
        _cache["base"] = text[start:text.index("\n```", start)].encode("utf-8")
    return _cache["base"]


def _cip57_doc():
    return _strict_json(_cip57_doc_bytes())


def _entry(**fields):
    """A fresh copy of the CIP hello_world validator entry with the given fields replaced."""
    e = _cip57_doc()["validators"][0]
    e.update(fields)
    return e


def _doc(entries, **preamble):
    d = _cip57_doc()
    d["preamble"].update(preamble)
    d["validators"] = list(entries)
    return json.dumps(d).encode("utf-8")


def _splice(anchor, replacement):
    base = _cip57_doc_bytes()
    if base.count(anchor) != ANCHOR_COUNTS[anchor]:
        raise ControlError("anchor %r occurrence count changed" % anchor)
    return base.replace(anchor, replacement, 1)


def _top(value):
    """Base document with one extra unread top-level member (D8 leaves extra fields open)."""
    base = _cip57_doc_bytes()
    if not base.startswith(TOP_ANCHOR):
        raise ControlError("CIP example no longer starts with an object")
    return TOP_ANCHOR + b'  "x-d7-unread": ' + value + b",\n" + base[len(TOP_ANCHOR):]


def _preamble_extra(value):
    return _splice(PREAMBLE_ANCHOR, PREAMBLE_ANCHOR + b'    "x-d7-unread": ' + value + b",\n")


def _datum_index(value):
    """Replace the value of the first (datum schema) "index" member, which the tool never reads."""
    return _splice(INDEX_ANCHOR, b'"index": ' + value + b",")


def _description(bad):
    return _splice(DESC_ANCHOR, b"Aiken" + bad + b" contracts for project")

def _premise(label, raw, exc_type):
    """Interpreter premise: stdlib json.loads raises exactly exc_type on this control. Fails, never skips."""
    try:
        json.loads(raw.decode("utf-8"))
    except Exception as e:
        if type(e) is exc_type:
            return
        got = type(e).__name__
    else:
        got = "no exception"
    raise AssertionError("M-HASH-D7-PREMISE %s not met: stdlib json.loads gave %s, expected exactly %s; "
                         "BLOCKED, do not adapt the control or the expectation" % (label, got, exc_type.__name__))


def _flip_byte(code_hex, index):
    """Replace one payload byte (hex pair at byte index) keeping CBOR framing intact."""
    i = 2 * index
    return code_hex[:i] + "%02x" % (int(code_hex[i:i + 2], 16) ^ 0x01) + code_hex[i + 2:]


def _flip_nibble(h):
    return ("1" if h[0] != "1" else "2") + h[1:]


def _trip(name, log):
    def hook(self, *args, **kwargs):
        log.append(name)
        raise TripwireError(name)
    return hook


def _tripwires(target, value, log):
    """A target subclass instance whose content-consuming hooks record any use and raise."""
    ns = {n: _trip(n, log) for n in TRIP_HOOKS if n == "__bytes__" or hasattr(target, n)}
    return type("Trip" + target.__name__.title(), (target,), ns)(value)


def _impostor(target, value, log):
    """Not a target subclass, but reports target via __class__: isinstance() accepts it, type() does not."""
    ns = {n: _trip(n, log) for n in TRIP_HOOKS}
    ns["__class__"] = property(lambda self: target)
    ns["__len__"] = lambda self: len(value)
    return type("Impostor", (object,), ns)()


def _static_problems(tree):
    problems = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            problems += ["import " + a.name for a in node.names if a.name.split(".")[0] not in ALLOWED_IMPORTS]
        elif isinstance(node, ast.ImportFrom):
            if node.level != 0 or (node.module or "").split(".")[0] not in ALLOWED_IMPORTS:
                problems.append("from-import %r level %d" % (node.module, node.level))
        elif isinstance(node, ast.Name) and node.id in BANNED_NAMES:
            problems.append("name " + node.id)
    return problems


def _impl():
    if "impl" not in _cache:
        if os.path.islink(IMPL_PATH):
            raise AssertionError("M-HASH-D7 loader refused: independent_script_hash.py is a symlink")
        if not os.path.isfile(IMPL_PATH):
            raise AssertionError(RED)
        with open(IMPL_PATH, "rb") as f:
            raw = f.read(MAX_IMPL_BYTES + 1)
        if len(raw) > MAX_IMPL_BYTES:
            raise AssertionError("M-HASH-D7 loader refused: implementation exceeds %d bytes" % MAX_IMPL_BYTES)
        tree = ast.parse(raw.decode("utf-8"), IMPL_PATH)
        problems = _static_problems(tree)
        if problems:
            raise AssertionError("M-HASH-D7 static lint refused to execute implementation: %s" % problems)
        # Execute exactly the linted tree: no second read, no bytecode cache. Lint only, not a sandbox.
        mod = types.ModuleType("independent_script_hash")
        exec(compile(tree, IMPL_PATH, "exec", dont_inherit=True), mod.__dict__)
        _cache["impl"] = mod
    return _cache["impl"]


class Base(unittest.TestCase):
    def api(self):
        m = _impl()
        self.assertTrue(issubclass(m.ScriptHashInputError, ValueError))
        self.assertTrue(issubclass(m.ScriptHashMismatchError, ValueError))
        self.assertFalse(issubclass(m.ScriptHashInputError, m.ScriptHashMismatchError))
        self.assertFalse(issubclass(m.ScriptHashMismatchError, m.ScriptHashInputError))
        self.m = m
        return m

    def assertInputError(self, fn, *args):
        with self.assertRaises(self.m.ScriptHashInputError) as cm:
            fn(*args)
        self.assertIs(type(cm.exception), self.m.ScriptHashInputError)

    def assertMismatch(self, raw):
        with self.assertRaises(self.m.ScriptHashMismatchError) as cm:
            self.m.check_blueprint(raw)
        self.assertIs(type(cm.exception), self.m.ScriptHashMismatchError)

    def assertResult(self, raw, want):
        got = self.m.check_blueprint(raw)
        self.assertIsInstance(got, dict)
        self.assertEqual(got, want)
        for title, h in got.items():
            self.assertIs(type(title), str)
            self.assertIs(type(h), str)
        return got

    def assertHello(self, doc):
        self.assertEqual(doc["preamble"]["plutusVersion"], "v2")
        self.assertEqual([(v["title"], v["compiledCode"], v["hash"]) for v in doc["validators"]],
                         [("hello_world", CIP57_CODE, CIP57_HASH)])


class T00AddendumProvenance(Base):
    """Passes without any implementation: pins, the CIP-57 positive base, control construction, premises."""

    def test_t00_provenance_controls_and_premises(self):
        for name, (size, sha) in PINS.items():
            with self.subTest(pin=name):
                raw = _read(name)
                self.assertEqual(len(raw), size)
                self.assertEqual(hashlib.sha256(raw).hexdigest(), sha)
        src = _strict_json(_read("SOURCES.json"))
        self.assertEqual(src["cipsCommit"], CIPS_COMMIT)
        status = {e["path"]: (e["status"], e["bytes"], e["sha256"]) for e in src["files"]}
        self.assertEqual(status["CIP-0057.md"], (200,) + PINS["CIP-0057.md"])
        cip = _read("CIP-0057.md").decode("utf-8")
        for anchor in ("The full compiled and cbor-encoded serialized flat script",
                       "A blake2b-224 hash digest of the validator script, as found in addresses",
                       '"plutusVersion": "v2"',
                       '"compiledCode": "%s"' % CIP57_CODE,
                       '"hash": "%s"' % CIP57_HASH):
            self.assertIn(anchor, cip)
        code = bytes.fromhex(CIP57_CODE)
        self.assertEqual((code[0], code[1], len(code)), (0x58, 0xAD, 2 + 0xAD))

        base = _cip57_doc_bytes()
        self.assertTrue(base.isascii())
        self.assertTrue(base.startswith(TOP_ANCHOR))
        for anchor, n in ANCHOR_COUNTS.items():
            self.assertEqual(base.count(anchor), n)
        for token in (b"NaN", b"Infinity", b"\xef\xbb\xbf", b"parameters", b"x-d7-unread"):
            self.assertNotIn(token, base)
        doc = _strict_json(base)
        self.assertHello(doc)
        self.assertNotIn("parameters", doc["validators"][0])
        self.assertEqual(_strict_json(_doc([_entry()])), doc)

        flipped_hash = _flip_nibble(CIP57_HASH)
        self.assertNotEqual(flipped_hash, CIP57_HASH)
        self.assertEqual(len(flipped_hash), 56)
        self.assertTrue(all(c in "0123456789abcdef" for c in flipped_hash))
        flipped_code = _flip_byte(CIP57_CODE, FLIP_INDEX)
        self.assertEqual((len(flipped_code), flipped_code[:4]), (len(CIP57_CODE), "58ad"))
        self.assertNotEqual(flipped_code, CIP57_CODE)

        ctl = _controls()
        self.assertEqual(set(ctl), set(POSITIVE) | set(BOM_CODECS) | set(NO_BOM_CODECS) | set(INVALID_UTF8)
                         | set(CONSTANTS) | set(PARAMETER_VALUES) | set(DEEP_NESTING) | {INT_CONTROL})
        for name, raw in ctl.items():
            with self.subTest(bound=name):
                self.assertIs(type(raw), bytes)
                self.assertLess(len(raw), MAX_BLUEPRINT_BYTES)
        for name in POSITIVE:
            with self.subTest(positive=name):
                d = _strict_json(ctl[name])
                self.assertHello(d)
                self.assertIn(d["validators"][0].get("parameters", "absent"), ("absent", []))
        text = base.decode("utf-8")
        for name, codec in BOM_CODECS.items():
            with self.subTest(encoding=name):
                self.assertEqual(ctl[name].decode(codec), text)
                if name == "utf8-bom":
                    self.assertEqual(ctl[name].decode("utf-8"), "﻿" + text)
                else:
                    with self.assertRaises(UnicodeDecodeError):
                        ctl[name].decode("utf-8")
        for name, codec in NO_BOM_CODECS.items():
            with self.subTest(encoding=name):
                self.assertEqual(ctl[name].decode(codec), text)
                self.assertIn("\x00", ctl[name].decode("utf-8"))
        for name in INVALID_UTF8:
            with self.subTest(invalid_utf8=name):
                with self.assertRaises(UnicodeDecodeError):
                    ctl[name].decode("utf-8")
                self.assertHello(_strict_json_text(ctl[name].decode("utf-8", "replace")))
        for name in CONSTANTS:
            with self.subTest(constant=name):
                with self.assertRaises(ControlError):
                    _strict_json(ctl[name])
                self.assertHello(json.loads(ctl[name].decode("utf-8")))
        for name, want in PARAMETER_VALUES.items():
            with self.subTest(parameters=name):
                d = _strict_json(ctl[name])
                self.assertHello(d)
                got = d["validators"][0]["parameters"]
                self.assertIs(type(got), type(want))
                self.assertEqual(got, want)
        for name in DEEP_NESTING:
            with self.subTest(premise="P1", control=name):
                _premise("P1", ctl[name], RecursionError)
        with self.subTest(premise="P2", control=INT_CONTROL):
            _premise("P2", ctl[INT_CONTROL], ValueError)

        log = []
        for target, value in ((str, CIP57_CODE), (bytes, base)):
            for obj in (_tripwires(target, value, log), _impostor(target, value, log)):
                with self.subTest(hook_control=type(obj).__name__, target=target.__name__):
                    self.assertIsInstance(obj, target)
                    self.assertIsNot(type(obj), target)
        self.assertEqual(log, [])


class T10Precedence(Base):
    """D2: every entry is validated before any comparison, so entry order never changes the class."""

    def test_d2_malformed_entry_wins_in_every_order(self):
        m = self.api()
        self.assertResult(_cip57_doc_bytes(), WANT)
        ok = ("ok", _entry(title="ok"))
        mismatch = ("mismatch", _entry(title="mismatch", hash=_flip_nibble(CIP57_HASH)))
        for label, fields in MALFORMED:
            bad = ("malformed", _entry(**dict({"title": "malformed"}, **fields)))
            orders = list(itertools.permutations((ok, mismatch, bad))) + list(itertools.permutations((mismatch, bad)))
            for order in orders:
                with self.subTest(malformed=label, order="/".join(n for n, _ in order)):
                    self.assertInputError(m.check_blueprint, _doc([e for _, e in order]))

    def test_d2_mismatch_at_every_position_and_order(self):
        self.api()
        titles = ("first", "second", "third")
        self.assertResult(_doc([_entry(title=t) for t in titles]), {t: CIP57_HASH for t in titles})
        for label, fields in (("hash-nibble", {"hash": _flip_nibble(CIP57_HASH)}),
                              ("code-byte", {"compiledCode": _flip_byte(CIP57_CODE, FLIP_INDEX)})):
            for pos in range(len(titles)):
                entries = [_entry(title=t, **(fields if i == pos else {})) for i, t in enumerate(titles)]
                with self.subTest(case=label, position=pos):
                    self.assertMismatch(_doc(entries))
        ok = _entry(title="ok")
        mismatch = _entry(title="mismatch", hash=_flip_nibble(CIP57_HASH))
        for order in ((ok, mismatch), (mismatch, ok)):
            with self.subTest(case="pair", first=order[0]["title"]):
                self.assertMismatch(_doc(order))


class T11FreshResult(Base):
    """Each call returns a fresh dict; mutating one result never leaks into another call."""

    def test_fresh_detached_result_across_calls(self):
        self.api()
        base = _cip57_doc_bytes()
        titles = ("first", "second")
        pair = _doc([_entry(title=t) for t in titles])
        want_pair = {t: CIP57_HASH for t in titles}
        r1 = self.assertResult(base, WANT)
        r1["hello_world"] = _flip_nibble(CIP57_HASH)
        r1["injected"] = CIP57_HASH
        r2 = self.assertResult(base, WANT)
        self.assertIsNot(r2, r1)
        self.assertEqual(r1, {"hello_world": _flip_nibble(CIP57_HASH), "injected": CIP57_HASH})
        r3 = self.assertResult(pair, want_pair)
        self.assertIsNot(r3, r1)
        self.assertIsNot(r3, r2)
        self.assertEqual(r2, WANT)
        r2.clear()
        r3.pop("first")
        self.assertResult(base, WANT)
        self.assertResult(pair, want_pair)
        self.assertEqual((r2, r3), ({}, {"second": CIP57_HASH}))

    def test_d3_utf16_utf32_with_and_without_bom_refused(self):

        self.assertResult(ctl["base"], WANT)
        for name in UTF16_UTF32:

    def test_d4_script_hash_exact_str_only(self):
        m = self.api()
        self.assertEqual(m.script_hash("v2", CIP57_CODE), CIP57_HASH)
        cases = (
            ("str-subclass-version", lambda log: (_Str("v2"), CIP57_CODE)),
            ("str-subclass-code", lambda log: ("v2", _Str(CIP57_CODE))),
            ("tripwire-version", lambda log: (_tripwires(str, "v2", log), CIP57_CODE)),
            ("tripwire-code", lambda log: ("v2", _tripwires(str, CIP57_CODE, log))),
            ("impostor-version", lambda log: (_impostor(str, "v2", log), CIP57_CODE)),
            ("impostor-code", lambda log: ("v2", _impostor(str, CIP57_CODE, log))),
            ("bool-version", lambda log: (True, CIP57_CODE)),
            ("bool-code", lambda log: ("v2", True)),
            ("memoryview-code", lambda log: ("v2", memoryview(CIP57_CODE.encode("ascii")))),
        )
        for label, make in cases:
            log = []
            args = make(log)
            with self.subTest(case=label):
                self.assertInputError(m.script_hash, *args)
                self.assertEqual(log, [])

            ("bytes-subclass", lambda log: _Bytes(base)),
            ("tripwire-bytes", lambda log: _tripwires(bytes, base, log)),
            ("impostor-bytes", lambda log: _impostor(bytes, base, log)),
            ("none", lambda log: None),
            ("json-plutusVersion-true", lambda log: _doc([_entry()], plutusVersion=True)),
            ("json-title-true", lambda log: _doc([_entry(title=True)])),
            ("json-compiledCode-true", lambda log: _doc([_entry(compiledCode=True)])),
            ("json-hash-true", lambda log: _doc([_entry(hash=True)])),
        )
        for label, make in cases:
