def assert_allowed(test, result, *, context=""):
    test.assertTrue(result["allow"], context or result)
    test.assertEqual(result["reasonCode"], "ALLOWED", context or result)


def assert_denied(test, result, reason, *, context=""):
    test.assertFalse(result["allow"], context or result)
    test.assertEqual(result["reasonCode"], reason, context or result)

            "verify",
            "review",
            "reproduce",
            "implement",
            "repair",
            "admin",
            "report",
        ):
            assert_allowed(self, assess(s, live_action(kind)), context=kind)

        s = reproducer_snapshot()

class IsolatedTempStore(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory()
        self.root = Path(self.temp.name) / "isolated-repo"
        self.root.mkdir()
        # No git metadata: get_db_path must use the in-repo fallback, never a
        # live worktree common directory.
        self.db_path = get_db_path(self.root)
        self.assertEqual(
            self.db_path, self.root / ".moriarty-dev" / "state.sqlite3"
        )
        self.assertFalse((self.root / ".git").exists())
        init_db(self.db_path).close()

    def tearDown(self):
        self.temp.cleanup()

    def history(self):
        return get_history(self.db_path, str(self.root), REQ, CAP)

    def seed_findings(self, *finding_ids):
        for index, finding_id in enumerate(finding_ids):
            record_event(
                self.db_path,
                str(self.root),
                REQ,
                CAP,
                CANDIDATE_HASH,
                "seed-" + str(index),
                "defect_failure",
                {
                    "findingId": finding_id,
                    "defectClass": "retained-loan-executor-defect",
                },
            )

    def approved(self, scope, resolved=None):
        receipt = {
            "author": "grok-4.6",
            "reviewer": "claude-opus-5",
            "candidateHash": CANDIDATE_HASH,
            "scope": scope,
            "verdict": "APPROVED",
            "requirement": REQ,
            "capability": CAP,
        }
        if resolved is not None:
            receipt["resolvedFindings"] = list(resolved)
        return receipt

    def reserve_kind(self, kind, *, profile=LIVE_PROFILE, action_id=None):
        action = live_action(kind, profile=profile)
        if action_id is not None:
            action["id"] = action_id
        return reserve(self.db_path, str(self.root), action, snapshot())

        conflicts = []

            try:
                winners.append(self.reserve_kind("verify", action_id=action_id))
            except ReservationConflictError:
                conflicts.append(action_id)

        t1 = threading.Thread(target=worker, args=("live-loan-verify-a",))
        t2 = threading.Thread(target=worker, args=("live-loan-verify-b",))

def get_db_path(repo_root: str | Path) -> Path:
    repo_path = Path(repo_root).resolve()
    git_entry = repo_path / ".git"
    try:
        if git_entry.is_file():
            text = git_entry.read_text(encoding="utf-8").strip()
            if text.startswith("gitdir:"):
                gitdir = Path(text.split("gitdir:", 1)[1].strip())
                if not gitdir.is_absolute():
                    gitdir = (repo_path / gitdir).resolve()
                commondir_file = gitdir / "commondir"
                if commondir_file.is_file():
                    cd_text = commondir_file.read_text(encoding="utf-8").strip()
                    common = Path(cd_text)
                    if not common.is_absolute():
                        common = (gitdir / common).resolve()
                    target = common / "moriarty-dev" / "state.sqlite3"
                    return target
                return gitdir / "moriarty-dev" / "state.sqlite3"
        elif git_entry.is_dir():
            return git_entry / "moriarty-dev" / "state.sqlite3"
    except Exception:
        pass
    fallback = repo_path / ".moriarty-dev" / "state.sqlite3"
    return fallback


def _canonical_repo_ids(repo: str | Path) -> list[str]:
    repo_path = Path(repo).resolve()
    ids = [str(repo), str(repo_path)]
    git_entry = repo_path / ".git"
    try:
        if git_entry.is_file():
            text = git_entry.read_text(encoding="utf-8").strip()
            if text.startswith("gitdir:"):
                gitdir = Path(text.split("gitdir:", 1)[1].strip())
                if not gitdir.is_absolute():
                    gitdir = (repo_path / gitdir).resolve()
                ids.append(str(gitdir))
                commondir_file = gitdir / "commondir"
                if commondir_file.is_file():
                    cd_text = commondir_file.read_text(encoding="utf-8").strip()
                    common = Path(cd_text)
                    if not common.is_absolute():
                        common = (gitdir / common).resolve()
                    ids.append(str(common))
        elif git_entry.is_dir():
            ids.append(str(git_entry.resolve()))
    except Exception:
        pass
    seen = set()
    result = []
    for i in ids:
        if i not in seen:
            seen.add(i)
            result.append(i)
    return result


def init_db(db_path: Path) -> sqlite3.Connection:
    if db_path.is_file():
        # Safeguard: verify existing file is a valid SQLite DB before executing statements.
        # If corrupt, raise StoreError and preserve original corrupt bytes for diagnosis.
        if db_path.stat().st_size > 0:
            try:
                with open(db_path, "rb") as f:
                    header = f.read(16)
                if header != b"SQLite format 3\x00":
                    raise StoreError(f"Database file is corrupt: {db_path}; invalid file header, preserving original bytes for diagnosis")
                with closing(sqlite3.connect(str(db_path), timeout=5.0)) as conn_test:
                    conn_test.execute("PRAGMA schema_version;")
            except StoreError:
                raise
            except Exception as exc:
                raise StoreError(f"Database file is corrupt: {db_path}; preserving original bytes for diagnosis: {exc}")

    db_path.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(str(db_path), timeout=10.0, isolation_level=None)
    try:
        conn.execute("PRAGMA journal_mode=WAL;")
        conn.execute("PRAGMA busy_timeout=5000;")
        conn.executescript(SCHEMA_SQL)
    except Exception:
        conn.close()
        raise
    return conn


def _now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def make_history_reader(db_path: Path, *, timeout: float = 10.0):
    def history_reader(lineage: dict) -> dict | None:
        repo = lineage.get("repository", "")
        req = lineage.get("requirement", "")
        cap = lineage.get("capability", "")
        return get_history(db_path, repo, req, cap, timeout=timeout)
    return history_reader


def _profile_occupies_slot(profile, action_kind):
    if action_kind in LEGACY_PRIMARY_KINDS:
        return True
    if type(profile) is not str:
        return True
    if profile.strip() == "":
        return True
    if profile == LIVE_EVIDENCE_PROFILE:
        return True
    if profile in KNOWN_NONLIVE_PROFILES:
        return False
    return True


def _extract_start_profile(payload):
    if not isinstance(payload, dict):
        return None
    action = payload.get("action")
    if not isinstance(action, dict) or "evidenceProfile" not in action:
        return None
    return action.get("evidenceProfile")


def _action_start_profiles(conn, repo, requirement, capability, candidate, action_id, reservation_id=None):
    try:
        cur = conn.execute(
            """
            SELECT payload_json FROM events
            WHERE event_kind = 'action_start'
              AND repository = ?
              AND requirement = ?
              AND capability = ?
              AND candidate = ?
              AND action_id = ?
            ORDER BY id DESC
            """,
            (repo, requirement, capability, candidate, action_id),
        )
        matched = []
        fallback = []
        for (raw,) in cur.fetchall():
            try:
                payload = json.loads(raw) if isinstance(raw, str) else (raw or {})
            except Exception:
                fallback.append(None)
                continue
            profile = _extract_start_profile(payload)
            if (
                reservation_id is not None
                and isinstance(payload, dict)
                and payload.get("reservationId") == reservation_id
            ):
                matched.append(profile)
            fallback.append(profile)
        if reservation_id is not None and matched:
            return matched
        return fallback
    except Exception:
        return None


def _profiles_occupy(profiles, action_kind):
    if action_kind in LEGACY_PRIMARY_KINDS:
        return True
    if not profiles:
        return True
    flags = [_profile_occupies_slot(profile, action_kind) for profile in profiles]
    if any(flags) and not all(flags):
        return True
    return flags[0]


def _reservation_occupies_primary(conn, reservation_id, action_kind, repo, requirement, capability, candidate, action_id):
    if action_kind in LEGACY_PRIMARY_KINDS:
        return True
    profiles = _action_start_profiles(
        conn,
        repo,
        requirement,
        capability,
        candidate,
        action_id,
        reservation_id=reservation_id,
    )
    if profiles is None:
        return True
    return _profiles_occupy(profiles, action_kind)


def _any_other_primary(conn, exclude_id=None):
    rows = conn.execute(
        """
        SELECT id, repository, requirement, capability, candidate, action_id, action_kind
        FROM reservations WHERE status = 'active'
        """
    ).fetchall()
    for rid, repo, req, cap, cand, act_id, kind in rows:
        if exclude_id is not None and rid == exclude_id:
            continue
        if _reservation_occupies_primary(conn, rid, kind, repo, req, cap, cand, act_id):
            return True
    return False


def _expand_related_repo_ids(conn, requested_ids, requirement, capability):
    requested_set = set(requested_ids)
    expanded = list(requested_ids)
    seen = set(requested_ids)
    try:
        cur = conn.execute(
            "SELECT DISTINCT repository FROM events WHERE requirement = ? AND capability = ?",
            (requirement, capability),
        )
        for (stored,) in cur.fetchall():
            if stored in seen:
                continue
            stored_ids = set(_canonical_repo_ids(stored))
            if stored_ids & requested_set:
                expanded.append(stored)
                seen.add(stored)
    except Exception:
        return requested_ids
    return expanded


def get_history(db_path: Path, repo: str, requirement: str, capability: str, *, timeout: float = 10.0, expand_related: bool = False) -> dict | None:
    if not db_path.is_file():
        return None

    try:
        if db_path.stat().st_size > 0:
            with open(db_path, "rb") as f:
                header = f.read(16)
            if header != b"SQLite format 3\x00":
                return None
        conn = sqlite3.connect(str(db_path), timeout=timeout)
    except Exception:
        return None

    try:
        cur = conn.execute("SELECT name FROM sqlite_master WHERE type='table' AND name='events'")
        if not cur.fetchone():
            return None

        repo_ids = _canonical_repo_ids(repo)
        if expand_related:
            repo_ids = _expand_related_repo_ids(conn, repo_ids, requirement, capability)
        placeholders = ",".join("?" for _ in repo_ids)

        # Total events for repository: schema-only or unrecorded database returns None
        cur = conn.execute(
            f"SELECT count(*) FROM events WHERE repository IN ({placeholders})",
            repo_ids,
        )
        total_events = cur.fetchone()[0]
        if total_events == 0:
            return None

        # This database belongs to one Git common directory. All linked
        # worktrees share its primary slot. A dead parent is ambiguous: its
        # child and charge can survive, so only recorded completion releases it.
        primary_active = _any_other_primary(conn)

        cur = conn.execute(
            f"""
            SELECT event_kind, payload_json FROM events
            WHERE repository IN ({placeholders}) AND requirement = ? AND capability = ?
            ORDER BY id ASC
            """,
            repo_ids + [requirement, capability],
        )
        rows = cur.fetchall()

        unresolved_findings = set()
        reproducer_verified = False
        approach_changed = False

        for kind, payload_raw in rows:
            try:
                payload = json.loads(payload_raw) if isinstance(payload_raw, str) else (payload_raw or {})
            except Exception:
                payload = {}

            if kind == "defect_failure":
                fid = (
                    payload.get("findingId")
                    or payload.get("defectId")
                    or payload.get("id")
                    or payload.get("defectClass")
                )
                if fid:
                    unresolved_findings.add(str(fid))
                else:
                    unresolved_findings.add(f"defect-{len(unresolved_findings)+1}")
                reproducer_verified = False
                approach_changed = False

            elif kind == "result_review_failure":
                review_findings = payload.get("findings") or payload.get("blockers") or []
                if not review_findings:
                    fid = payload.get("findingId") or f"REVIEW-FAIL-{payload.get('candidateHash', '')[:8]}"
                    unresolved_findings.add(str(fid))
                    reproducer_verified = False
                    approach_changed = False

            elif kind == "approach_changed":
                approach_changed = True

            elif kind == "reproducer_verified":
                reproducer_verified = True

            elif kind == "result_review":
                verdict = payload.get("verdict") or payload.get("scopedVerdict")
                if verdict == "APPROVED":
                    resolved = payload.get("resolvedFindings") or payload.get("resolved")
                    if resolved and isinstance(resolved, list):
                        for rf in resolved:
                            unresolved_findings.discard(str(rf))
            # Notice: kind == "defect_resolved" is ignored; author events cannot erase blockers

        # Admin cycles
        cur = conn.execute(
            f"""
            SELECT count(*) FROM events
            WHERE repository IN ({placeholders}) AND event_kind = 'admin_cycle'
            """,
            repo_ids,
        )
        admin_cycle_count = cur.fetchone()[0]

        # Admin intervals
        cur = conn.execute(
            f"SELECT count(*) FROM admin_intervals WHERE repository IN ({placeholders})",
            repo_ids,
        )
        total_intervals = cur.fetchone()[0]

        if total_intervals == 0:
            admin_seconds = None
            admin_cycles = admin_cycle_count if admin_cycle_count > 0 else None
        else:
            cur = conn.execute(
                f"""
                SELECT start_time, end_time FROM admin_intervals
                WHERE repository IN ({placeholders})
                  AND activity_type NOT IN ('test', 'testing', 'afk', 'idle')
                ORDER BY start_time ASC
                """,
                repo_ids,
            )
            intervals = cur.fetchall()
            admin_seconds = int(_merge_intervals(intervals))
            admin_cycles = admin_cycle_count

        return {
            "sameDefectFailures": len(unresolved_findings),
            "adminCycles": admin_cycles,
            "adminSeconds": admin_seconds,
            "primaryActive": primary_active,
            "reproducerVerified": reproducer_verified,
            "approachChanged": approach_changed,
        }
    except Exception:
        return None
    finally:
        try:
            conn.close()
        except Exception:
            pass


def read_store_identity(db_path: Path, *, timeout: float = 10.0) -> str | None:
    """Read a persisted identity token if present. Never creates schema or rows."""
    if not db_path.is_file():
        return None
    try:
        uri = "file:" + str(Path(db_path).resolve()) + "?mode=ro"
        conn = sqlite3.connect(uri, uri=True, timeout=timeout)
    except Exception:
        return None
    try:
        cur = conn.execute(
            "SELECT name FROM sqlite_master WHERE type='table' AND name='store_identity'"
        )
        if not cur.fetchone():
            return None
        rows = conn.execute("SELECT token FROM store_identity LIMIT 2").fetchall()
        if len(rows) != 1:
            return None
        token = rows[0][0]
        if type(token) is not str or token.strip() == "":
            return None
        return token
    except Exception:
        return None
    finally:
        try:
            conn.close()
        except Exception:
            pass


def _exact_lineage_recorded(db_path: Path, repo: str, requirement: str, capability: str, *, timeout: float = 10.0) -> bool:
    if not db_path.is_file():
        return False
    try:
        conn = sqlite3.connect(str(db_path), timeout=timeout)
    except Exception:
        return False
    try:
        repo_ids = _canonical_repo_ids(repo)
        repo_ids = _expand_related_repo_ids(conn, repo_ids, requirement, capability)
        placeholders = ",".join("?" for _ in repo_ids)
        cur = conn.execute(
            f"SELECT count(*) FROM events WHERE repository IN ({placeholders}) AND requirement = ? AND capability = ?",
            repo_ids + [requirement, capability],
        )
        return cur.fetchone()[0] > 0
    except Exception:
        return False
    finally:
        try:
            conn.close()
        except Exception:
            pass


def get_live_history(db_path: Path, repo: str, requirement: str, capability: str, *, identity_anchor=None, timeout: float = 10.0) -> dict | None:
    """Live-profile history: available only on a positive in-store identity match.

    Production callers pass no anchor, so history stays unavailable until a
    separately approved identity adoption. Tests may pass an in-process
    ``identity_anchor`` that must equal the persisted token.
    """
    if type(identity_anchor) is not str or identity_anchor.strip() == "":
        return None
    token = read_store_identity(db_path, timeout=timeout)
    if token is None or token != identity_anchor:
        return None
    if not _exact_lineage_recorded(db_path, repo, requirement, capability, timeout=timeout):
        return None
    return get_history(
        db_path, repo, requirement, capability, timeout=timeout, expand_related=True
    )


def make_guarded_history_reader(db_path: Path, *, live_lineage=None, identity_anchor=None, timeout: float = 10.0):
    def history_reader(lineage: dict) -> dict | None:
        repo = lineage.get("repository", "")
        req = lineage.get("requirement", "")
        cap = lineage.get("capability", "")
        is_live = False
        if live_lineage is not None:
            try:
                is_live = bool(live_lineage(lineage))
            except Exception:
                is_live = True
        if is_live:
            return get_live_history(
                db_path, repo, req, cap, identity_anchor=identity_anchor, timeout=timeout
            )
        return get_history(db_path, repo, req, cap, timeout=timeout)
    return history_reader

def _merge_intervals(intervals: list[tuple[float, float]]) -> float:
    if not intervals:
        return 0.0
    valid = []
    for pair in intervals:
        try:
            s, e = float(pair[0]), float(pair[1])
            if e >= s:
                valid.append((s, e))
        except (ValueError, TypeError, IndexError):
            continue
    if not valid:
        return 0.0
    valid.sort(key=lambda x: (x[0], x[1]))
    merged = [list(valid[0])]
    for s, e in valid[1:]:
        prev = merged[-1]
        if s <= prev[1]:
            prev[1] = max(prev[1], e)
        else:
            merged.append([s, e])
    return sum(e - s for s, e in merged)


def reserve(db_path: Path, repo: str, action: dict, snapshot: dict, charge_id: str | None = None) -> str:
    conn = init_db(db_path)
    try:
        conn.execute("BEGIN IMMEDIATE")
        action_kind = action.get("kind", "")
        incoming_occupies = _profile_occupies_slot(
            action.get("evidenceProfile"), action_kind
        )
        if incoming_occupies and _any_other_primary(conn):
            raise ReservationConflictError(
                "Unresolved primary reservation already exists"
            )

        res_id = str(uuid.uuid4())
        if charge_id is not None:
            try:
                conn.execute("INSERT INTO runner_claims (charge_id, reservation_id) VALUES (?, ?)", (charge_id, res_id))
            except sqlite3.IntegrityError as exc:
                raise ReservationConflictError("Existing charge already claimed: " + charge_id) from exc
        pid = os.getpid()
        now = _now_iso()
        conn.execute(
            """
            INSERT INTO reservations (
                id, repository, requirement, capability, candidate,
                action_id, action_kind, pid, status, reserved_at
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, 'active', ?)
            """,
            (
                res_id,
                repo,
                action.get("requirement", ""),
                action.get("capability", ""),
                action.get("candidate", ""),
                action.get("id", ""),
                action_kind,
                pid,
                now,
            ),
        )

        conn.execute(
            """
            INSERT INTO events (
                repository, requirement, capability, candidate,
                action_id, event_kind, observed_at, payload_json
            ) VALUES (?, ?, ?, ?, ?, 'action_start', ?, ?)
            """,
            (
                repo,
                action.get("requirement", ""),
                action.get("capability", ""),
                action.get("candidate", ""),
                action.get("id", ""),
                now,
                json.dumps({"action": action, "snapshot": snapshot, "reservationId": res_id}),
            ),
        )
        conn.commit()
        return res_id
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def has_other_primary(db_path: Path, reservation_id: str) -> bool:
    with closing(sqlite3.connect(db_path)) as conn:
        return _any_other_primary(conn, exclude_id=reservation_id)


def finish(db_path: Path, reservation_id: str, receipt: dict | None = None, status: str = "finished"):
    conn = init_db(db_path)
    try:
        conn.execute("BEGIN IMMEDIATE")
        now = _now_iso()
        cur = conn.execute(
            "SELECT repository, requirement, capability, candidate, action_id FROM reservations WHERE id = ?",
            (reservation_id,),
        )
        row = cur.fetchone()
        receipt_str = json.dumps(receipt or {})
        conn.execute(
            """
            UPDATE reservations
            SET status = ?, finished_at = ?, receipt_json = ?
            WHERE id = ?
            """,
            (status, now, receipt_str, reservation_id),
        )
        if row:
            repo, req, cap, cand, act_id = row
            conn.execute(
                """
                INSERT INTO events (
                    repository, requirement, capability, candidate,
                    action_id, event_kind, observed_at, payload_json
                ) VALUES (?, ?, ?, ?, ?, 'action_end', ?, ?)
                """,
                (repo, req, cap, cand, act_id, now, receipt_str),
            )
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()

def record_review(db_path: Path, repo: str, receipt_data: dict, action: dict | None = None):
    if not isinstance(receipt_data, dict):
        raise StoreError("Receipt must be a JSON object")

    author = receipt_data.get("author")
    reviewer = (
        receipt_data.get("reviewer")
        or receipt_data.get("reviewerObserved")
        or receipt_data.get("reviewerRequested")
    )
    if not reviewer or not isinstance(reviewer, str):
        raise StoreError("Reviewer identity required in review receipt")
    if author and isinstance(author, str):
        a_low = author.lower().strip()
        r_low = reviewer.lower().strip()
        if a_low in r_low or r_low in a_low:
            raise StoreError("Author self-review is forbidden")

    cand = receipt_data.get("candidateHash", receipt_data.get("candidate", ""))
    if not cand or not isinstance(cand, str) or len(cand) < 16:
        raise StoreError("Valid candidateHash required in review receipt")

    scope = receipt_data.get("scope", receipt_data.get("auditScope", ""))
    if not scope or not isinstance(scope, str) or scope.strip() == "" or scope.strip().lower() == "none":
        raise StoreError("Valid non-empty scope required in review receipt")

    if action:
        action_cand = action.get("candidate", cand)
        if cand != action_cand:
            raise StoreError(f"Candidate mismatch: receipt candidate {cand} does not match action candidate {action_cand}")
        req = action.get("requirement", receipt_data.get("requirement", ""))
        cap = action.get("capability", receipt_data.get("capability", ""))
        act_id = action.get("id", "review")
    else:
        req = receipt_data.get("requirement", "")
        cap = receipt_data.get("capability", "")
        act_id = receipt_data.get("actionId", "review")

    verdict = receipt_data.get("verdict", receipt_data.get("scopedVerdict", ""))
    findings = receipt_data.get("findings")
    if findings is None:
        findings = receipt_data.get("blockers", [])

    conn = init_db(db_path)
    now = _now_iso()
    try:
        with conn:
            event_kind = "result_review" if verdict == "APPROVED" else "result_review_failure"
            conn.execute(
                """
                INSERT INTO events (
                    repository, requirement, capability, candidate,
                    action_id, event_kind, observed_at, payload_json
                ) VALUES (?, ?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    repo,
                    req,
                    cap,
                    cand,
                    act_id,
                    event_kind,
                    now,
                    json.dumps(receipt_data),
                ),
            )
            if verdict != "APPROVED":
                if not findings:
                    fid = receipt_data.get("findingId") or f"REVIEW-FAIL-{cand[:8]}"
                    findings = [{"findingId": fid, "defectClass": "review_rejected", "description": "Review not approved"}]
                for finding in findings:
                    if isinstance(finding, str):
                        finding = {"findingId": finding, "defectClass": finding}
                    conn.execute(
                        """
                        INSERT INTO events (
                            repository, requirement, capability, candidate,
                            action_id, event_kind, observed_at, payload_json
                        ) VALUES (?, ?, ?, ?, ?, 'defect_failure', ?, ?)
                        """,
                        (
                            repo,
                            req,
                            cap,
                            cand,
                            act_id,
                            now,
                            json.dumps(finding),
                        ),
                    )
    finally:
        conn.close()

def enqueue_tx(db_path: Path, repo: str, tx_id: str, status: str, details: dict):
    if status != "submitted":
        raise StoreError("Only a selected public submitted transaction may enter the outbox")
    if not isinstance(tx_id, str) or not re.fullmatch(r"[A-Za-z0-9:_-]{1,256}", tx_id) or not isinstance(details, dict):
        raise StoreError("Transaction id and public details are required")
    forbidden_subs = ("seed", "sk", "secret", "private", "witness", "password", "token", "auth", "spending_key", "signing_key")
    def _check_forbidden(d):
        if isinstance(d, dict):
            for k, v in d.items():
                k_low = k.lower()
                for sub in forbidden_subs:
                    if sub in k_low and k_low not in ("publicsink", "public_sink"):
                        raise ValueError(f"Forbidden private field in transaction details: {k}")
                _check_forbidden(v)
        elif isinstance(d, list):
            for item in d:
                _check_forbidden(item)

    _check_forbidden(details)
    # Closed public fields: arbitrary notes/objects can contain witness data
    # even when their keys do not look private. The CLI supplies only network.
    if set(details) - {"network", "blockNumber", "contractAddress"}:
        raise ValueError("Only network, blockNumber and contractAddress are public outbox fields")
    if "network" in details and details["network"] != "preview":
        raise ValueError("Only Preview public observations are supported")
    if "blockNumber" in details and (type(details["blockNumber"]) is not int or details["blockNumber"] < 0):
        raise ValueError("Public blockNumber must be a nonnegative integer")
    if "contractAddress" in details and (not isinstance(details["contractAddress"], str)
            or not re.fullmatch(r"[A-Za-z0-9:_-]{1,256}", details["contractAddress"])):
        raise ValueError("Invalid public contractAddress")

    conn = init_db(db_path)
    try:
        conn.execute("BEGIN IMMEDIATE")
        try:
            existing = conn.execute("SELECT repository FROM outbox WHERE tx_id = ?", (tx_id,)).fetchone()
            if existing:
                if get_db_path(existing[0]).resolve() != get_db_path(repo).resolve():
                    raise StoreError("Transaction belongs to another repository")
                conn.commit()
                return
            conn.execute(
                """
                INSERT INTO outbox (repository, tx_id, status, enqueued_at, details_json)
                VALUES (?, ?, ?, ?, ?)
                """,
                (repo, tx_id, status, _now_iso(), json.dumps(details)),
            )
            conn.execute(
                """
                INSERT INTO events (
                    repository, requirement, capability, candidate,
                    action_id, event_kind, observed_at, payload_json
                ) VALUES (?, '', '', '', 'tx', 'tx_notification', ?, ?)
                """,
                (
                    repo,
                    _now_iso(),
                    json.dumps({"txId": tx_id, "status": status, "details": details}),
                ),
            )
            conn.commit()
        except Exception:
            conn.rollback()
            raise
    finally:
        conn.close()


VALID_TX_STATUSES = {"submitted", "unknown-finality", "failed", "confirmed"}
TX_TRANSITIONS = {
    "submitted": {"unknown-finality", "failed", "confirmed"},
    "unknown-finality": {"failed", "confirmed"},
    "failed": set(),
    "confirmed": set(),
    "delivered": set(),
}


def update_tx_status(db_path: Path, tx_id: str, new_status: str, note: str | None = None):
    if new_status not in VALID_TX_STATUSES:
        raise StoreError(f"Invalid transaction status: {new_status}")
    if note is not None:
        raise StoreError("Free-form notes are not public transaction fields")
    conn = init_db(db_path)
    try:
        conn.execute("BEGIN IMMEDIATE")
        try:
            cur = conn.execute("SELECT repository, status, details_json FROM outbox WHERE tx_id = ?", (tx_id,))
            row = cur.fetchone()
            if not row:
                raise StoreError(f"Transaction not found in outbox: {tx_id}")
            repo, old_status, details_str = row
            if new_status == old_status:
                conn.commit()
                return
            if new_status not in TX_TRANSITIONS.get(old_status, set()):
                raise StoreError(f"Invalid transaction transition: {old_status} -> {new_status}")
            try:
                details = json.loads(details_str)
            except Exception:
                details = {}
            conn.execute(
                "UPDATE outbox SET status = ?, details_json = ?, delivered_at = NULL, enqueued_at = ? WHERE tx_id = ?",
                (new_status, json.dumps(details), _now_iso(), tx_id),
            )
            conn.execute(
                """
                INSERT INTO events (
                    repository, requirement, capability, candidate,
                    action_id, event_kind, observed_at, payload_json
                ) VALUES (?, '', '', '', 'tx', 'tx_status_change', ?, ?)
                """,
                (repo, _now_iso(), json.dumps({"txId": tx_id, "newStatus": new_status, "note": note})),
            )
            conn.commit()
        except Exception:
            conn.rollback()
            raise
    finally:
        conn.close()


def get_undelivered_txs(db_path: Path, repo: str | None = None, *, timeout: float = 5.0, limit: int | None = None) -> list[dict]:
    if not db_path.is_file():
        return []
    conn = sqlite3.connect(str(db_path), timeout=timeout)
    try:
        cur = conn.execute(
            "SELECT tx_id, status, details_json, enqueued_at, repository FROM outbox WHERE delivered_at IS NULL ORDER BY id ASC LIMIT ?",
            (-1 if limit is None else limit,),
        )
        rows = cur.fetchall()
        result = []
        for tx_id, status, details_str, enq, owner in rows:
            if repo and get_db_path(owner).resolve() != get_db_path(repo).resolve():
                continue
            try:
                details = json.loads(details_str)
            except Exception:
                details = {}
            result.append({
                "txId": tx_id,
                "status": status,
                "enqueuedAt": enq,
                "details": details,
            })
        return result
    finally:
        conn.close()


def mark_delivered(db_path: Path, tx_id: str, acknowledgement: dict | None = None):
    if not isinstance(acknowledgement, dict) or acknowledgement.get("txId") != tx_id:
        raise StoreError("Actual delivery acknowledgement for this transaction is required")
    delivery_id = acknowledgement.get("deliveryId")
    if not isinstance(delivery_id, str) or not delivery_id:
        raise StoreError("Delivery acknowledgement id is required")
    if acknowledgement.get("source") != "codex-session-message" or not acknowledgement.get("contentSha256"):
        raise StoreError("Host-observed message acknowledgement required")
    conn = init_db(db_path)
    now = _now_iso()
    try:
        conn.execute("BEGIN IMMEDIATE")
        try:
            row = conn.execute("SELECT repository, status, enqueued_at, delivered_at FROM outbox WHERE tx_id = ?", (tx_id,)).fetchone()
            if not row:
                raise StoreError("Transaction not found in outbox")
            repo, status, enqueued_at, delivered_at = row
            if delivered_at is not None:
                raise StoreError("Transaction is already delivered")
            if acknowledgement.get("status") != status or acknowledgement.get("enqueuedAt") != enqueued_at:
                raise StoreError("Delivery observation is stale for current transaction status")
            conn.execute(
                "UPDATE outbox SET delivered_at = ? WHERE tx_id = ?",
                (now, tx_id),
            )
            conn.execute(
                """
                INSERT INTO events (
                    repository, requirement, capability, candidate,
                    action_id, event_kind, observed_at, payload_json
                ) VALUES (?, '', '', '', 'tx', 'tx_delivered', ?, ?)
                """,
                (repo, now, json.dumps({**acknowledgement, "deliveredAt": now})),
            )
            conn.commit()
        except Exception:
            conn.rollback()
            raise
    finally:
        conn.close()
