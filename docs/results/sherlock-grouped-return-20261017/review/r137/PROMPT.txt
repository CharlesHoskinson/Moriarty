            body: expr(Expr::If(

    Snapshot {
        program: program.clone(), control, environment, frames,
        stack_limit: 8, remaining_fuel: fuel, initial_fuel: 82, outcome,
    }
}
fn expected_tail(program: &Arc<Program>) -> Vec<Snapshot> {
    vec![
        // State75: base case returned1; caller multiplications1 and2 pending.
        expected_snapshot(program, 7, env(0), vec![ret(env(1)), mul(1), ret(env(2)), mul(2), ret(empty())], returning(1), Outcome::Pending),
        // State76: caller n=1 restored, its pending multiplication retained.
        expected_snapshot(program, 6, env(1), vec![mul(1), ret(env(2)), mul(2), ret(empty())], returning(1), Outcome::Pending),
        expected_snapshot(program, 5, env(1), vec![ret(env(2)), mul(2), ret(empty())], primitive(1), Outcome::Pending),
        expected_snapshot(program, 4, env(1), vec![ret(env(2)), mul(2), ret(empty())], returning(1), Outcome::Pending),
        // State79: caller n=2 restored, outer multiplication2 still pending.
        expected_snapshot(program, 3, env(2), vec![mul(2), ret(empty())], returning(1), Outcome::Pending),
        expected_snapshot(program, 2, env(2), vec![ret(empty())], primitive(2), Outcome::Pending),
        expected_snapshot(program, 1, env(2), vec![ret(empty())], returning(2), Outcome::Pending),

        expected_snapshot(program, 0, empty(), vec![], returning(2), Outcome::Pending),

        expected_snapshot(program, 0, empty(), vec![], None, Outcome::Done(Value::UInt(2))),
    ]

fn expected_records(states: &[Snapshot]) -> Vec<StepRecord> {
    let rules = [185, 153, 223, 185, 153, 223, 185, 80];
    let charges = [1, 1, 1, 1, 1, 1, 1, 0];
    rules.into_iter().zip(charges).enumerate().map(|(i, (rule_start_line, charge))| {
        StepRecord {
            rule_start_line, charge,
            before: states[i].clone(), after: states[i + 1].clone(),
        }
    }).collect()
}
fn assert_group(
    actual: &ReferenceSlice, states: &[Snapshot], records: &[StepRecord],
    first: usize, past: usize,
) {
    assert_eq!(actual.profile_id, PROFILE);
    assert_eq!(actual.start, states[first]);
    // Whole typed record equality covers every before/after field and order,
    // not merely the final result, line list or cross-group self-consistency.
    assert_eq!(actual.records, records[first..past]);
    assert_eq!(actual.end, states[past]);
    let mut prefix = vec![0u128];
    for expected in &records[first..past] {
        prefix.push(prefix.last().copied().unwrap() + u128::from(expected.charge));
    }
    assert_eq!(actual.prefix_work, prefix);
    let terminal = past == records.len();
    let (stop, kind) = if terminal {
        (SliceStop::Outcome(Outcome::Done(Value::UInt(2))), SliceBoundaryKind::Terminal)
    } else {
        (SliceStop::TechnicalCut, SliceBoundaryKind::TechnicalCut)
    };
    assert_eq!(actual.stop, stop);
    assert_eq!(actual.boundaries, vec![SliceBoundary { position: past - first, kind }]);
    for event in &actual.records {
        assert!(event.before.depth() <= event.before.stack_limit);
        assert!(event.after.depth() <= event.after.stack_limit);
    }
}

#[test]
fn grouped_recursive_callee_return_preserves_complete_records_for_widths_1_2_4_8() {
    // All starts are independently frozen boundaries around the same callee
    // return: before base return, just after base return, and before next return.
    // The latter starts make width8 include fewer than8 events before first stop.
    for start_boundary in [75usize, 76, 78] {
        for width in [1usize, 2, 4, 8] {
            let source = program();
            let states = expected_tail(&source);
            let expected = expected_records(&states);
            let mut machine = Machine::new(source, 82, 8);
            // Warm-up observations are discarded, not used as expectations.
            // Full independently frozen start equality below checks the cut.
            for _ in 0..start_boundary { machine.step().expect("pending warm-up event"); }
            let mut at = start_boundary - 75;
            assert_eq!(machine.snapshot(), states[at]);
            while at < expected.len() {
                let past = (at + width).min(expected.len());
                let group = advance_m(&mut machine, width);
                assert_group(&group, &states, &expected, at, past);
                assert_eq!(machine.snapshot(), states[past]);
                at = past;
            }
            let terminal = states.last().unwrap();
            let empty_group = advance_m(&mut machine, width);
            assert_eq!(empty_group.start, *terminal);
            assert_eq!(empty_group.end, *terminal);
            assert!(empty_group.records.is_empty());
            assert_eq!(empty_group.prefix_work, vec![0]);
            assert_eq!(empty_group.stop, SliceStop::Outcome(Outcome::Done(Value::UInt(2))));
            assert_eq!(empty_group.boundaries, vec![SliceBoundary {
                position: 0, kind: SliceBoundaryKind::Terminal,
            }]);
            assert!(machine.step().is_none(), "no event after the first published outcome");
            assert_eq!(machine.snapshot(), *terminal);
        }
    }
}

use super::reference_types::*;
use std::sync::Arc;

/// Host reference evaluator for the frozen `sherlock-k-a/1` profile on the typed common domain.
///
/// # Security Warning (H1 — Unmetered Host Evaluator)
/// This machine is a host diagnostic reference with zero native authority and no host time,
/// heap, or native stack admission bounds. Do not expose it to untrusted inputs.
pub struct Machine {
    pub(super) state: Snapshot,
}

impl Machine {
    pub fn new(program: Arc<Program>, fuel: u128, stack_limit: usize) -> Self {
        Self {
            state: Snapshot {
                control: Some(Control::Eval(program.entry.clone())),
                program,
                environment: Environment::new(),
                frames: vec![],
                stack_limit,
                remaining_fuel: fuel,
                initial_fuel: fuel,
                outcome: Outcome::Pending,
            },
        }
    }

    pub fn snapshot(&self) -> Snapshot {
        self.state.clone()
    }

    /// Executes at most one `sherlock-k-a/1` transition or free terminal publication.
    ///
    /// Clones the full [`Snapshot`] twice (`before` and `after`), including environments and
    /// lists. Unmetered on host resources; not safe for untrusted input (H1).
    pub fn step(&mut self) -> Option<StepRecord> {
        if self.state.outcome != Outcome::Pending {
            return None;
        }
        let control = self.state.control.clone()?;
        let before = self.snapshot();

        // K terminal publication precedes exhaustion, even at zero fuel.
        let (rule_start_line, charge) = match control {
            Control::Return(value) if self.state.frames.is_empty() => {
                self.state.control = None;
                self.state.outcome = Outcome::Done(value);
                (80, 0)
            }
            Control::Error(code) => {
                self.state.control = None;
                self.state.outcome = Outcome::Rejected(code);
                (82, 0)
            }
            _ if self.state.remaining_fuel == 0 => {
                self.state.control = None;
                self.state.outcome = Outcome::Exhausted;
                (84, 0)
            }
            control => {
                let (line, next) = self.transition(control);
                self.state.control = Some(next);
                self.state.remaining_fuel -= 1;
                (line, 1)
            }
        };
        Some(StepRecord {
            rule_start_line,
            charge,
            before,
            after: self.snapshot(),
        })
    }

    /// Retains at most `max_events` successive [`Self::step`] records from the current state.
    ///
    /// Each event preserves its complete before/after snapshots, source-rule line and charge.
    /// Free terminal publication counts as one event; collection stops immediately when its
    /// outcome is published. A zero budget leaves the machine unchanged and returns no records.
    /// Later calls resume this same machine, including any pending publication.
    ///
    /// This is a host observation grouping limit, not another fuel or resource policy. Pure
    /// helpers, snapshot cloning and retained allocations remain unmetered as in [`Self::step`].
    pub fn run_steps(&mut self, max_events: usize) -> Vec<StepRecord> {
        let mut records = Vec::new();
        for _ in 0..max_events {
            let Some(record) = self.step() else {
                break;
            };
            let terminal = record.after.outcome != Outcome::Pending;
            records.push(record);
            if terminal {
                break;
            }
        }
        records
    }

    // One charged machine rule from the pinned main.k. Pure helpers below do
    // not generate steps or consume fuel in the A profile.
    fn transition(&mut self, control: Control) -> (usize, Control) {
        match control {
            Control::Eval(expr) => self.evaluate(expr),
            Control::Push(frame, next, environment) => {
                if self.state.frames.len() < self.state.stack_limit {
                    self.state.frames.insert(0, frame);
                    self.state.environment = environment;
                    (135, *next)
                } else {
                    (140, Control::Error(4))
                }
            }
            Control::Return(value) => self.return_value(value),
            Control::Invoke(name, values) => {
                let function = self
                    .state
                    .program
                    .functions
                    .iter()
                    .find(|function| function.name == name)
                    .cloned();
                (200, Control::Enter(function, values))
            }
            Control::Enter(None, _) => (202, Control::Error(7)),
            Control::Enter(Some(function), values) => {
                let bindings = bind(&function.parameters, &values);
                (
                    204,
                    Control::Bound(bindings, function.return_type, function.body),
                )
            }
            Control::Bound(None, _, _) => (206, Control::Error(8)),
            Control::Bound(Some(environment), ty, body) => {
                let frame = Frame::Return(ty, self.state.environment.clone());
                (208, push(frame, Control::Eval(body), environment))
            }
            Control::Cases(_, arms, _) if arms.is_empty() => (212, Control::Error(9)),
            Control::Cases(value, mut arms, environment) => {
                let arm = arms.remove(0);
                let bindings = matching(&arm.pattern, &value, &environment);
                (
                    214,
                    Control::CaseResult(bindings, value, arm.body, arms, environment),
                )
            }
            Control::CaseResult(None, value, _, arms, environment) => {
                (217, Control::Cases(value, arms, environment))
            }
            Control::CaseResult(Some(bindings), _, body, _, environment) => (
                220,
                push(Frame::Scope(environment), Control::Eval(body), bindings),
            ),
            Control::PrimitiveResult(Ok(value)) => (223, Control::Return(value)),
            Control::PrimitiveResult(Err(code)) => (225, Control::Error(code)),
            // Error publication is handled by step without a fuel charge.
            Control::Error(_) => unreachable!("error publication is administrative"),
        }
    }

    fn evaluate(&self, expr: ExprRef) -> (usize, Control) {
        let environment = &self.state.environment;
        match expr.as_ref() {
            // Every value representable by this typed API satisfies uiValid.
            Expr::Literal(value) => (102, Control::Return(value.clone())),
            Expr::Var(name) => match environment.get(name) {
                Some(value) => (108, Control::Return(value.clone())),
                None => (111, Control::Error(3)),
            },
            Expr::Let(name, value, body) => {
                let frame = Frame::Let(name.clone(), body.clone(), environment.clone());
                (
                    114,
                    push(frame, Control::Eval(value.clone()), environment.clone()),
                )
            }
            Expr::Binary(op, left, right) => {
                let frame = Frame::Left(*op, right.clone(), environment.clone());
                (
                    117,
                    push(frame, Control::Eval(left.clone()), environment.clone()),
                )
            }
            Expr::Unary(op, value) => (
                120,
                push(
                    Frame::Unary(*op),
                    Control::Eval(value.clone()),
                    environment.clone(),
                ),
            ),
            Expr::If(condition, yes, no) => {
                let frame = Frame::If(yes.clone(), no.clone(), environment.clone());
                (
                    123,
                    push(frame, Control::Eval(condition.clone()), environment.clone()),
                )
            }
            Expr::Match(value, arms) => {
                let frame = Frame::Match(arms.clone(), environment.clone());
                (
                    126,
                    push(frame, Control::Eval(value.clone()), environment.clone()),
                )
            }
            Expr::Call(name, args) if args.is_empty() => {
                (129, Control::Invoke(name.clone(), vec![]))
            }
            Expr::Call(name, args) => {
                let frame = Frame::Args(
                    name.clone(),
                    vec![],
                    args[1..].to_vec(),
                    environment.clone(),
                );
                (
                    131,
                    push(frame, Control::Eval(args[0].clone()), environment.clone()),
                )
            }
        }
    }

    fn return_value(&mut self, value: Value) -> (usize, Control) {
        // The empty-stack return was already published by step. Replacements
        // retain depth; only source rules with a stack rewrite to S pop it.
        match self.state.frames[0].clone() {
            Frame::Let(name, body, environment) => {
                self.state.frames[0] = Frame::Scope(environment.clone());
                self.state.environment = environment;
                self.state.environment.insert(name, value);
                (145, Control::Eval(body))
            }
            Frame::Left(op, right, environment) => {
                self.state.frames[0] = Frame::Right(op, value);
                self.state.environment = environment;
                (149, Control::Eval(right))
            }
            Frame::Right(op, left) => {
                self.state.frames.remove(0);
                (
                    153,
                    Control::PrimitiveResult(binary_result(op, left, value)),
                )
            }
            Frame::Unary(op) => {
                self.state.frames.remove(0);
                (157, Control::PrimitiveResult(unary_result(op, value)))
            }
            Frame::If(yes, no, environment) => match value {
                Value::Bool(condition) => {
                    self.state.frames.remove(0);
                    self.state.environment = environment;
                    if condition {
                        (161, Control::Eval(yes))
                    } else {
                        (165, Control::Eval(no))
                    }
                }
                _ => (169, Control::Error(5)),
            },
            Frame::Match(arms, environment) => {
                self.state.frames.remove(0);
                self.state.environment = environment.clone();
                (173, Control::Cases(value, arms, environment))
            }
            Frame::Args(name, mut values, mut rest, environment) => {
                values.push(value);
                self.state.environment = environment.clone();
                if rest.is_empty() {
                    self.state.frames.remove(0);
                    (181, Control::Invoke(name, values))
                } else {
                    let next = rest.remove(0);
                    self.state.frames[0] = Frame::Args(name, values, rest, environment);
                    (177, Control::Eval(next))
                }
            }
            Frame::Return(ty, environment) => {
                if has_type(&value, ty) {
                    self.state.frames.remove(0);
                    self.state.environment = environment;
                    (185, Control::Return(value))
                } else {
                    (190, Control::Error(6))
                }
            }
            Frame::Scope(environment) => {
                self.state.frames.remove(0);
                self.state.environment = environment;
                (195, Control::Return(value))
            }
        }
    }

    /// Runs the machine until a terminal [`Outcome`] (`Done`, `Rejected`, or `Exhausted`) is
    /// published.
    ///
    /// # Security Warning (H1 — DoS and Native Stack Overflow Risks)
    /// This method is an **unmetered** host reference evaluator and **must not** be connected to
    /// the CLI, LSP, MCP, the `/1` or `/2` source compiler adapter, or any untrusted/network-reachable
    /// input path until host time, memory, and nesting bounds are enforced:
    /// - `fuel: u128` and `stack_limit: usize` have no host ceiling.
    /// - Every step clones the full [`Snapshot`] twice (`before`/`after`) and `run()` discards
    ///   both copies; every pushed [`Frame`] clones its [`Environment`]; `BinaryOp::Cons` is $O(n)$.
    /// - Derived `Drop`, `Clone`, and `PartialEq` impls (as well as diagnostic K rendering)
    ///   recurse over `Arc<Expr>` and `Arc<Tree>`. Deep caller-supplied ASTs or runtime-built
    ///   `Branch`/`Leaf` trees can overflow the native call stack and abort the process.
    pub fn run(&mut self) -> Snapshot {
        while self.step().is_some() {}
        self.snapshot()
    }
}

fn push(frame: Frame, control: Control, environment: Environment) -> Control {
    Control::Push(frame, Box::new(control), environment)
}

// main.k:238-245. Type tests concern the value's sort, not its contents.
fn has_type(value: &Value, ty: Type) -> bool {
    matches!(
        (value, ty),
        (Value::UInt(_), Type::UInt)
            | (Value::Bool(_), Type::Bool)
            | (Value::List(_), Type::List)
            | (Value::Tree(_), Type::Tree)
    )
}

// main.k:250-272. Translate uiChecked's range failure to error 10 using
// checked arithmetic on the typed U128 input domain.
fn binary_result(op: BinaryOp, left: Value, right: Value) -> Result<Value, u8> {
    match (op, left, right) {
        (BinaryOp::Add, Value::UInt(a), Value::UInt(b)) => {
            a.checked_add(b).map(Value::UInt).ok_or(10)
        }
        (BinaryOp::Sub, Value::UInt(a), Value::UInt(b)) => {
            a.checked_sub(b).map(Value::UInt).ok_or(10)
        }
        (BinaryOp::Mul, Value::UInt(a), Value::UInt(b)) => {
            a.checked_mul(b).map(Value::UInt).ok_or(10)
        }
        (BinaryOp::Lt, Value::UInt(a), Value::UInt(b)) => Ok(Value::Bool(a < b)),
        (BinaryOp::Eq, a, b) => Ok(Value::Bool(a == b)),
        (BinaryOp::Cons, Value::UInt(head), Value::List(mut tail)) => {
            tail.insert(0, head);
            Ok(Value::List(tail))
        }
        (BinaryOp::Branch, Value::Tree(left), Value::Tree(right)) => {
            Ok(Value::Tree(Tree::Branch(Arc::new(left), Arc::new(right))))
        }
        _ => Err(11),
    }
}

// main.k:273-278.
fn unary_result(op: UnaryOp, value: Value) -> Result<Value, u8> {
    match (op, value) {
        (UnaryOp::Leaf, Value::UInt(value)) => Ok(Value::Tree(Tree::Leaf(value))),
        (UnaryOp::Not, Value::Bool(value)) => Ok(Value::Bool(!value)),
        _ => Err(11),
    }
}

// main.k:290-298. uiBind builds its tail first, and duplicate names or
// argument count/type mismatches make the complete binding fail.
fn bind(parameters: &[Parameter], values: &[Value]) -> Option<Environment> {
    if parameters.len() != values.len() {
        return None;
    }
    let mut environment = Environment::new();
    for (parameter, value) in parameters.iter().zip(values).rev() {
        if !has_type(value, parameter.ty) || environment.contains_key(&parameter.name) {
            return None;
        }
        environment.insert(parameter.name.clone(), value.clone());
    }
    Some(environment)
}

// main.k:300-319. Each successful pattern extends the saved environment;
// equal names in Cons/Branch are a miss, including on matching value kinds.
fn matching(pattern: &Pattern, value: &Value, saved: &Environment) -> Option<Environment> {
    let mut environment = saved.clone();
    match (pattern, value) {
        (Pattern::Var(name), value) => {
            environment.insert(name.clone(), value.clone());
        }
        (Pattern::Nil, Value::List(values)) if values.is_empty() => {}
        (Pattern::Cons(head, tail), Value::List(values)) if head != tail && !values.is_empty() => {
            environment.insert(head.clone(), Value::UInt(values[0]));
            environment.insert(tail.clone(), Value::List(values[1..].to_vec()));
        }
        (Pattern::Leaf(name), Value::Tree(Tree::Leaf(value))) => {
            environment.insert(name.clone(), Value::UInt(*value));
        }
        (Pattern::Branch(left, right), Value::Tree(Tree::Branch(a, b))) if left != right => {
            environment.insert(left.clone(), Value::Tree(a.as_ref().clone()));
            environment.insert(right.clone(), Value::Tree(b.as_ref().clone()));
        }
        _ => return None,
    }
    Some(environment)
}

    Rejected(u8),
    Exhausted,

impl Outcome {
    /// Host-diagnostic classifier for the frozen `sherlock-k-a/1` A1 resource-limit partition.
    ///
    /// Returns `true` iff `self` is [`Outcome::Exhausted`] (zero-fuel termination) or
    /// [`Outcome::Rejected(4)`] (K-A frame-cap overflow at `D >=Int L`). All other outcomes—
    /// including [`Outcome::Pending`], [`Outcome::Done`], and all other [`Outcome::Rejected`]
    /// codes (such as semantic, typing, binding, pattern, or arithmetic errors `3, 5..=11` and
    /// unreachable literal code `2`)—return `false`.
    ///
    /// Scope: host diagnostic only on the `sherlock-k-a/1` typed common domain. This predicate
    /// carries zero native or cryptographic authority, does not transfer across Quint
    /// (`sherlock-quint-a/1`) or Candidate B (`sherlock-helper-microstep/1`), and does not make
    /// the unmetered host evaluator safe for untrusted inputs. No consumer may interpret a
    /// resource-limit outcome as a semantic, typing, or financial rejection.
    pub const fn is_resource_limit(&self) -> bool {
        matches!(self, Self::Exhausted | Self::Rejected(4))
    }
}

/// Complete post-startup K boundary state on the typed host common domain.
/// Frames are stored top first. `None` control represents K's `.K`.
/// Snapshots are observations; no arbitrary-state import is exposed.
///
/// Machine invariant: every `Snapshot` produced by [`super::reference::Machine`] satisfies
/// `remaining_fuel <= initial_fuel`.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Snapshot {
    pub program: Arc<Program>,
    pub control: Option<Control>,
    pub environment: Environment,
    pub frames: Vec<Frame>,
    pub stack_limit: usize,
    pub remaining_fuel: u128,
    pub initial_fuel: u128,
    pub outcome: Outcome,
}

impl Snapshot {
    pub fn depth(&self) -> usize {
        self.frames.len()
    }

    /// Returns `initial_fuel - remaining_fuel` for machine-valid observations.
    ///
    /// # Preconditions
    /// Requires the machine invariant `self.remaining_fuel <= self.initial_fuel`.
    /// For hand-constructed snapshots where this invariant is not guaranteed, use
    /// [`Snapshot::checked_charged_steps`] so invalid counters are reported as `None`
    /// rather than silently clamped, wrapped, or panicked.
    pub fn charged_steps(&self) -> u128 {
        self.initial_fuel - self.remaining_fuel
    }

    /// Checked host-diagnostic step counter returning `None` if `remaining_fuel > initial_fuel`.
    pub const fn checked_charged_steps(&self) -> Option<u128> {
        self.initial_fuel.checked_sub(self.remaining_fuel)
    }
}

    pub rule_start_line: usize,
    pub charge: u8,
    pub before: Snapshot,
    pub after: Snapshot,
}

//! Public host grouping baseline for the existing frozen K-A reference events.
//!
//! This ordinary host baseline groups genuine Machine::step records; it does
//! not implement a new transition relation, helper clock or optimized machine.
//! Full snapshots and record order remain observable. A technical cut records
//! a grouping boundary only: it creates no financial commit, effect, suspension
//! or new machine Outcome. Preparation and full observation cloning are unmetered
//! host work. This is no bounded B schema, native payload or capacity claim.

use super::reference_types::*;
use super::Machine;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum SliceBoundaryKind {
    Terminal,
    TechnicalCut,
}

/// Position counts retained records, so zero names the slice's start state.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct SliceBoundary {
    pub position: usize,
    pub kind: SliceBoundaryKind,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SliceStop {
    Outcome(Outcome),
    TechnicalCut,
}

/// Host-only K-A ReferenceSlice observation. It carries only fields available
/// in the existing K-A machine. Source suspension and proposed B identities,
/// layouts and helpers remain separate from this host API.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ReferenceSlice {
    pub profile_id: &'static str,
    pub start: Snapshot,
    pub records: Vec<StepRecord>,
    /// Length records.len()+1, starts zero, accumulates each actual record.charge.
    pub prefix_work: Vec<u128>,
    pub boundaries: Vec<SliceBoundary>,
    pub end: Snapshot,
    pub stop: SliceStop,
}

/// Retain at most m reference events, including free publication as an event.
/// Normal comparison budgets are 1/2/4/8. Other host usize budgets merely choose
/// grouping length; no native width or resource policy is selected here.
///
/// Zero is an explicit observation-only cut: pending input yields identical
/// start/end, no records, prefix_work=[0], and a TechnicalCut at position zero.
/// Already published outcomes, including with m=0, yield an empty terminal
/// slice with the unchanged Outcome and a Terminal boundary at position zero.
///
/// Each next step is attempted in order on the live ordinary Machine, with its
/// own fuel/cap/publication behavior. There is no stepping ahead or replay.
/// Reaching a pending terminal control at the event budget leaves publication
/// for the next group; when the budget includes publication, its zero charge
/// is retained and the group stops immediately at that first published outcome.
pub fn advance_m(machine: &mut Machine, m: usize) -> ReferenceSlice {
    let start = machine.snapshot();
    let mut records = Vec::new();
    let mut prefix_work = vec![0];

    if start.outcome == Outcome::Pending {
        for _ in 0..m {
            let Some(record) = machine.step() else {
                break;
            };
            let cumulative = prefix_work.last().copied().expect("initial zero work")
                + u128::from(record.charge);
            let terminal = record.after.outcome != Outcome::Pending;
            prefix_work.push(cumulative);
            records.push(record);
            if terminal {
                break;
            }
        }
    }

    let end = machine.snapshot();
    let (stop, kind) = if end.outcome == Outcome::Pending {
        (SliceStop::TechnicalCut, SliceBoundaryKind::TechnicalCut)
    } else {
        (SliceStop::Outcome(end.outcome.clone()), SliceBoundaryKind::Terminal)
    };
    let boundary = SliceBoundary { position: records.len(), kind };
    ReferenceSlice {
        profile_id: PROFILE,
        start,
        records,
        prefix_work,
        boundaries: vec![boundary],
        end,
        stop,
    }
}
