def _sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def _git_blob_sha1(data):
    h = hashlib.sha1()
    h.update(('blob %d\0' % len(data)).encode('ascii'))
    h.update(data)
    return h.hexdigest()
