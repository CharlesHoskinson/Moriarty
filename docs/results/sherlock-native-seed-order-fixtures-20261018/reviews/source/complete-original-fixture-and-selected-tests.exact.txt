    def _write(self, path, data=b'data', mode=0o644):
        p = Path(path)
        p.parent.mkdir(parents=True, exist_ok=True)
        raw = data.encode('utf-8') if isinstance(data, str) else data
        with open(p, 'wb') as f:
            f.write(raw)
        os.chmod(p, mode)
        return p
