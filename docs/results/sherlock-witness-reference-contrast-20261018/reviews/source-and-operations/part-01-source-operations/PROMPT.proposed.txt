/// Convert the existing flat, fully resolved witness value; no AST evaluation.
pub fn reference_value(value: &DecodedValue) -> Result<a::Value> {
    let mut nodes: Vec<a::Value> = Vec::with_capacity(value.nodes.len());
    for row in &value.nodes {
        let child = |i: usize| nodes.get(i).cloned().ok_or(ProjectionGap::UnrepresentableValue);
        let result = match *row {
            ValueNode::UInt(n) => a::Value::UInt(n), ValueNode::Bool(b) => a::Value::Bool(b),
            ValueNode::Nil => a::Value::List(Vec::new()),
            ValueNode::Cons { head, tail } => {
                let (a::Value::UInt(n), a::Value::List(tail)) = (child(head)?, child(tail)?) else { return Err(ProjectionGap::UnrepresentableValue); };
                let mut list = Vec::with_capacity(tail.len() + 1); list.push(n); list.extend(tail); a::Value::List(list)
            }
            ValueNode::Leaf { value } => {
                let a::Value::UInt(n) = child(value)? else { return Err(ProjectionGap::UnrepresentableValue); };
                a::Value::Tree(a::Tree::Leaf(n))
            }
            ValueNode::Branch { left, right } => {
                let (a::Value::Tree(l), a::Value::Tree(r)) = (child(left)?, child(right)?) else { return Err(ProjectionGap::UnrepresentableValue); };
                a::Value::Tree(a::Tree::Branch(Arc::new(l), Arc::new(r)))
            }
            ValueNode::U128OutOfRange => return Err(ProjectionGap::UnrepresentableValue),
        };
        nodes.push(result);
    }
    nodes.get(value.root).cloned().ok_or(ProjectionGap::UnrepresentableValue)
}

fn expression(value: &ExpressionRowC) -> ExpressionSnapshot {
    match value {
        ExpressionRowC::Literal { constant } => ExpressionSnapshot::Literal { constant:*constant },
        ExpressionRowC::Var { binder } => ExpressionSnapshot::Var { binder:*binder },
        ExpressionRowC::Let { binder, value, body } => ExpressionSnapshot::Let { binder:*binder, value:*value, body:*body },
        ExpressionRowC::Binary { op, left, right } => ExpressionSnapshot::Binary { op:*op, left:*left, right:*right },
        ExpressionRowC::Unary { op, operand } => ExpressionSnapshot::Unary { op:*op, operand:*operand },
        ExpressionRowC::If { cond, then_expr, else_expr } => ExpressionSnapshot::If { cond:*cond, then_expr:*then_expr, else_expr:*else_expr },
        ExpressionRowC::Call { function_name, args } => ExpressionSnapshot::Call { function_name:*function_name, args:*args },
        ExpressionRowC::Match { scrutinee, arms } => ExpressionSnapshot::Match { scrutinee:*scrutinee, arms:*arms },
    }

            layout:LayoutSnapshot {

            },
        },
        image:ImageSnapshot {
            run_id:(*image.run_id).clone(), control:*image.control, control_origin:image.control_origin,
            env_root:image.env_root, stack_limit:image.stack_limit, outcome:*image.outcome,
            initial_fuel:image.initial_fuel, remaining_work:image.remaining_work,
            consumed_work:image.consumed_work, helper_work:image.helper_work,

pub enum CaptureStepError {
    BeforeObservation(HostViolation),

    Step {

/// Capture owned pre-state before mutation, call the existing fixed owner once,
/// then observe/copy actual post-state. This explicit stepping API consumes the
/// caller's existing host-execution dependency; source integration does not admit it.
/// No sink, expected data, new dispatcher, clock conversion or run loop is involved.
pub fn record_step(admitted: &Admitted, image: &mut MachineImage)
    -> Result<ObservedStep, CaptureStepError>
{
    let before = capture(admitted, image).map_err(CaptureStepError::BeforeObservation)?;
    let engine = host_owner_engine();
    match engine.step(admitted, image) {
        Ok(StepResult::Event { event }) => {
            match capture(admitted, image) {
                Ok(after) => Ok(ObservedStep::Event(CompleteStepRecord { before, event, after })),
                Err(violation) => Err(CaptureStepError::AfterObservation {
                    violation, before:Box::new(before), event:Some(event),
                }),
            }
        }
        Ok(StepResult::Terminal) => {
            match capture(admitted, image) {
                Ok(after) => Ok(ObservedStep::Terminal { before, after }),
                Err(violation) => Err(CaptureStepError::AfterObservation {
                    violation, before:Box::new(before), event:None,
                }),
            }
        }
        Err(violation) => Err(CaptureStepError::Step {
            violation, before:Box::new(before), after:capture(admitted, image).map(Box::new),
        }),
    }
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ContinuityBoundary {
    EmptyReturned,
    InitialBefore,
    Between { previous_record: usize, next_record: usize },
    FinalReturned { final_record: usize },
}
pub struct ContinuityDifference<'a> {
    pub boundary: ContinuityBoundary,
    pub expected: &'a HostSnapshot,
    pub observed: &'a HostSnapshot,
}

/// Full typed boundary equality only. A match does not assert that events are
/// valid, authenticate source, establish terminality or prove semantic refinement.
/// A present empty sequence compares initial and returned state directly.
pub fn first_discontinuity<'a>(initial: &'a HostSnapshot,
    records: &'a [CompleteStepRecord], returned: &'a HostSnapshot)
    -> Option<ContinuityDifference<'a>>
{
    if records.is_empty() {
        return (initial != returned).then_some(ContinuityDifference {
            boundary:ContinuityBoundary::EmptyReturned, expected:initial, observed:returned,
        });
    }
    if initial != &records[0].before {
        return Some(ContinuityDifference {
            boundary:ContinuityBoundary::InitialBefore, expected:initial, observed:&records[0].before,
        });
    }
    for index in 1..records.len() {
        if records[index - 1].after != records[index].before {
            return Some(ContinuityDifference {
                boundary:ContinuityBoundary::Between { previous_record:index - 1, next_record:index },
                expected:&records[index - 1].after, observed:&records[index].before,
            });
        }
    }
    let final_record = records.len() - 1;
    (&records[final_record].after != returned).then_some(ContinuityDifference {
        boundary:ContinuityBoundary::FinalReturned { final_record },
        expected:&records[final_record].after, observed:returned,
    })
}


/// Optional witness materialization around the SAME existing fixed-owner step.
/// The raw event/boundaries remain complete and unmodified. No run loop or
/// transition evaluator is introduced. All plan/decode failures are tooling
/// failures after the real step; raw boundaries survive and no retry is implied.
pub struct PlannedObservedStep {
    pub observed: ObservedStep,
    pub before: super::witness_plan::MaterializedSnapshot,
    pub after: super::witness_plan::MaterializedSnapshot,
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum WitnessSide { Before, After }
pub enum PlannedStepError {
    Owner(CaptureStepError),
    Materialization {
        side: WitnessSide,
        failure: super::witness_plan::WitnessFailure,
        observed: Box<ObservedStep>,
        completed_before: Option<Box<super::witness_plan::MaterializedSnapshot>>,
    },
}
/// Execute existing record_step exactly once, then materialize its OWN saved
/// boundaries. A tooling failure does not roll back or erase the actual step.
/// Original capture/record_step signatures, raw semantics and owner errors stay
/// unchanged. Cloning/decoding retains complete values; host costs remain H1 gaps.
pub fn record_step_with_plan(admitted: &Admitted, image: &mut MachineImage)
    -> Result<PlannedObservedStep, PlannedStepError>
{
    use super::witness_plan::{materialize_for, LifetimeUnit};
    let observed = record_step(admitted, image).map_err(PlannedStepError::Owner)?;
    let (before_raw, after_raw) = match &observed {
        ObservedStep::Event(record) => (&record.before, &record.after),
        ObservedStep::Terminal { before, after } => (before, after),
    };
    let before = match materialize_for(before_raw, LifetimeUnit::RecordBefore) {
        Ok(value) => value,
        Err(failure) => return Err(PlannedStepError::Materialization {
            side: WitnessSide::Before, failure, observed: Box::new(observed),
            completed_before: None,
        }),
    };
    let after = match materialize_for(after_raw, LifetimeUnit::RecordAfter) {
        Ok(value) => value,
        Err(failure) => return Err(PlannedStepError::Materialization {
            side: WitnessSide::After, failure, observed: Box::new(observed),
            completed_before: Some(Box::new(before)),
        }),
    };
    Ok(PlannedObservedStep { observed, before, after })
}

/// Preserve the baseline's complete validation, order and operation semantics.
/// Last-use only releases scratch values after their final ordered consumer;
/// boundary live-outs remain through publication. Cached copies are accounted
/// separately and are not reported as reclaimed scratch or semantic savings.
pub fn evaluate_with_storage(plan: &WitnessPlan, input: &HostSnapshot,
    storage: SlotStorage, mut cache: Option<&mut WitnessCache>)
    -> Result<StrategyEvaluation, WitnessFailure>
{
    let at = Location::new("plan", None, "structure");
    let schedule = dag::validate(plan)?;
    let mut scheduling = StrategyDiagnostics::default();
    if let Some(memo) = cache.as_deref_mut() {
        scheduling.scope_reset = memo.bind(plan, input);
    }
    let mut release = vec![Vec::new(); schedule.len()];
    if storage == SlotStorage::LastUse {
        for definition in &plan.definitions {
            // validate() compared these exact definitions to the existing
            // recomputation; every non-live-out last-use is a schedule position.
            if !definition.live_out {
                release[definition.last_use_order].push(definition.value);
            }
        }
    }
    let mut slots: Vec<Option<SlotValue>> = vec![None; plan.nodes.len()];
    let mut diagnostics = HostDiagnostics::default();
    let mut live = 0;
    for (position, id) in schedule.into_iter().enumerate() {
        let node = &plan.nodes[id.0];
        let result = {
            let mut args = Vec::new();
            for dependency in &node.dependencies {
                let value = slots.get(dependency.node.0).and_then(Option::as_ref)
                    .ok_or_else(|| dag::node_failure(node, FailureKind::MissingDependency {
                        dependency: dependency.node,
                    }))?;
                if value.value_type() != dependency.value_type {
                    return Err(dag::node_failure(node, FailureKind::WrongType {
                        expected: dependency.value_type, actual: value.value_type(),
                    }));
                }
                args.push(value);
            }
            // Reads are never bypassed: InputChanged/type/location behavior is
            // the existing operation's behavior, including warm evaluations.
            let memoizable = !matches!(node.operation, Operation::Read(_));
            let cached = if memoizable {
                cache.as_deref().and_then(|memo| memo.entries[id.0].as_ref())
                    .filter(|entry| entry.arguments.len() == args.len()
                        && entry.arguments.iter().zip(&args).all(|(a, b)| a == *b))
            } else { None };
            if let Some(entry) = cached {
                scheduling.cache_hits += 1;
                scheduling.copied_cached_slot_values += 1;
                entry.value.clone()
            } else {
                let value = dag::operation(plan, input, node, &args, &mut diagnostics)?;
                diagnostics.evaluated_nodes += 1;
                if memoizable {
                    if let Some(memo) = cache.as_deref_mut() {
                        scheduling.cache_misses += 1;
                        memo.entries[id.0] = Some(MemoEntry {
                            arguments: args.iter().map(|value| (**value).clone()).collect(),
                            value: value.clone(),
                        });
                    }
                }
                value
            }
        };
        if result.value_type() != node.output_type {
            return Err(dag::node_failure(node, FailureKind::WrongType {
                expected: node.output_type, actual: result.value_type(),
            }));
        }
        slots[id.0] = Some(result);
        live += 1;
        scheduling.scheduled_nodes += 1;
        scheduling.peak_live_slots = scheduling.peak_live_slots.max(live);
        for dead in &release[position] {
            if slots[dead.0].take().is_some() {
                live -= 1;
                scheduling.reclaimed_slots += 1;
            }
        }
    }
    diagnostics.retained_slots = live;
    scheduling.retained_slots_at_publication = live;
    if let Some(memo) = cache.as_deref() {
        scheduling.cached_value_slots = memo.value_slots();
        scheduling.cached_argument_slots = memo.argument_slots();
    }
    let Some(SlotValue::Snapshot(decoded)) = slots.get_mut(plan.output.0).and_then(Option::take) else {
        return Err(fail(at, FailureKind::IncompleteOutputs));
    };
    Ok(StrategyEvaluation {
        witness: WitnessEvaluation { decoded, diagnostics }, scheduling,
    })
}

/// One real owner step, followed solely by reconstruction of its saved raw
/// boundaries. On any tooling failure, the actual step and available boundaries
/// are returned unchanged; no rollback, retry or extra transition is performed.
pub fn record_step_with_storage(admitted: &Admitted, image: &mut MachineImage,
    storage: SlotStorage, mut cache: Option<&mut WitnessRecordCache>)
    -> Result<StrategyObservedStep, StrategyStepError>
{
    let observed = observation::record_step(admitted, image).map_err(StrategyStepError::Owner)?;
    let (before_raw, after_raw) = match &observed {
        ObservedStep::Event(record) => (&record.before, &record.after),
        ObservedStep::Terminal { before, after } => (before, after),
    };
    let before_cache = cache.as_deref_mut().map(|memo| &mut memo.before);
    let before = match materialize_with_storage(before_raw, LifetimeUnit::RecordBefore, storage, before_cache) {
        Ok(value) => value,
        Err(failure) => return Err(StrategyStepError::Materialization {
            side: WitnessSide::Before, failure, observed: Box::new(observed),
            completed_before: None,
        }),
    };
    let after_cache = cache.as_deref_mut().map(|memo| &mut memo.after);
    let after = match materialize_with_storage(after_raw, LifetimeUnit::RecordAfter, storage, after_cache) {
        Ok(value) => value,
        Err(failure) => return Err(StrategyStepError::Materialization {
            side: WitnessSide::After, failure, observed: Box::new(observed),
            completed_before: Some(Box::new(before)),
        }),
    };
    Ok(StrategyObservedStep { observed, before, after })
}

/// Yield every complete observed record/terminal boundary in order. The sink
/// may stop after a delivered boundary, preserving that exact caller-owned
/// MachineImage. No inferred work/event, hidden step, image substitution or
/// application exhaustion is introduced. Execution classification remains the
/// existing caller's obligation, independently of this source-only interface.
pub fn stream_owner_steps<F>(admitted: &Admitted, image: &mut MachineImage,
    mut cache: Option<&mut WitnessRecordCache>, mut sink: F)
    -> Result<StreamStop, StrategyStepError>
where F: FnMut(&StrategyObservedStep) -> StreamAction
{
    loop {
        let step = record_step_with_storage(admitted, image, SlotStorage::LastUse, cache.as_deref_mut())?;
        let terminal = matches!(&step.observed, ObservedStep::Terminal { .. });
        let action = sink(&step);
        // No decoded/raw record history is held across loop iterations.
        drop(step);
        if terminal { return Ok(StreamStop::Terminal); }
        if action == StreamAction::Stop { return Ok(StreamStop::Sink); }
    }
}
