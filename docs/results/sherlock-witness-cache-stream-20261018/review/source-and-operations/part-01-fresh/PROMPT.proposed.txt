/// Kahn's algorithm, choosing the smallest ready ID, retains duplicate edges.
/// This order is independent of hash-map iteration or previous evaluations.
pub(super) fn order(nodes: &[Node]) -> Result<Vec<NodeId>, WitnessFailure> {
    let at = Location::new("plan", None, "dependencies");
    let mut counts = vec![0usize; nodes.len()];
    let mut uses = vec![Vec::new(); nodes.len()];
    for (index, node) in nodes.iter().enumerate() {
        if node.id != NodeId(index) { return Err(fail(node.location, FailureKind::InvalidNodeId)); }
        for dep in &node.dependencies {
            let source = nodes.get(dep.node.0).ok_or_else(|| fail(node.location,
                FailureKind::MissingDependency { dependency: dep.node }))?;
            if source.output_type != dep.value_type {
                return Err(fail(node.location, FailureKind::WrongType {
                    expected: dep.value_type, actual: source.output_type,
                }));
            }
            counts[index] = counts[index].checked_add(1).ok_or_else(|| fail(at, FailureKind::IndexOverflow))?;
            uses[dep.node.0].push(NodeId(index));
        }
    }
    let mut ready = BTreeSet::new();
    for (index, count) in counts.iter().enumerate() { if *count == 0 { ready.insert(NodeId(index)); } }
    let mut result = Vec::new();
    while let Some(id) = ready.iter().next().copied() {
        ready.remove(&id);
        result.push(id);
        for next in &uses[id.0] {
            counts[next.0] -= 1;
            if counts[next.0] == 0 { ready.insert(*next); }
        }
    }
    if result.len() != nodes.len() { return Err(fail(at, FailureKind::CyclicDependencies)); }
    Ok(result)
}

pub(super) fn definitions(nodes: &[Node], order: &[NodeId], live_outs: &[NodeId], lifetime: LifetimeUnit)
    -> Result<Vec<Definition>, WitnessFailure>
{
    let mut positions = vec![0usize; nodes.len()];
    for (position, id) in order.iter().enumerate() {
        let item = positions.get_mut(id.0).ok_or_else(|| fail(Location::new("plan", None, "order"), FailureKind::InvalidNodeId))?;
        *item = position;
    }
    let mut result: Vec<_> = nodes.iter().map(|node| Definition {
        value: node.id, producer: node.id, consumers: Vec::new(),
        last_use_order: positions[node.id.0], lifetime, live_out: live_outs.contains(&node.id),
    }).collect();
    for id in order {
        for dep in &nodes[id.0].dependencies {
            result[dep.node.0].consumers.push(*id);
            result[dep.node.0].last_use_order = result[dep.node.0].last_use_order.max(positions[id.0]);
        }
    }
    for id in live_outs {
        let value = result.get_mut(id.0).ok_or_else(|| fail(Location::new("plan", None, "live_outs"), FailureKind::InvalidNodeId))?;
        // Boundary live-outs remain available through final publication.
        value.last_use_order = order.len();
    }
    Ok(result)
}

fn field_values(node: &Node, fields: &[FieldBinding], args: &[&SlotValue])
    -> Result<Vec<NamedValue>, WitnessFailure>
{
    if node.dependencies.len() != fields.len() + 1 || args.len() != fields.len() + 1 {
        return Err(node_failure(node, FailureKind::IncompleteBindings));
    }
    let mut result = Vec::new();
    for (offset, field) in fields.iter().enumerate() {
        if node.dependencies[offset + 1] != field.dependency {
            return Err(node_failure(node, FailureKind::IncompleteBindings));
        }
        result.push(NamedValue { field: field.field, value: (*args[offset + 1]).clone() });
    }
    Ok(result)
}

pub(super) fn operation(plan: &WitnessPlan, input: &HostSnapshot, node: &Node, args: &[&SlotValue], diagnostics: &mut HostDiagnostics)
    -> Result<SlotValue, WitnessFailure>
{
    let wrong = || node_failure(node, FailureKind::WrongOperation);
    Ok(match &node.operation {
        Operation::Read(id) => {
            if !args.is_empty() { return Err(wrong()); }

            if binding.id != *id { return Err(node_failure(node, FailureKind::IncompleteBindings)); }
            let value = read(input, binding.path, binding.location)?;
            if value.source_type() != binding.value_type {
                return Err(node_failure(node, FailureKind::WrongType {
                    expected: ValueType::Source(binding.value_type), actual: ValueType::Source(value.source_type()),
                }));
            }
            if value != binding.immutable_value { return Err(node_failure(node, FailureKind::InputChanged)); }
            diagnostics.input_reads += 1;
            SlotValue::Source(value)
        }
        Operation::Bind => {
            if args.len() != 2 { return Err(wrong()); }
            let SlotValue::Source(_) = args[0] else { return Err(wrong()); };
            (*args[1]).clone()

        Operation::Value { source, arity } => {
            let value = decode::evaluate_value(plan, node, *source, *arity, args)?;
            diagnostics.value_constructions += 1;
            let copied = value.nodes.len().checked_sub(1).ok_or_else(wrong)?;
            diagnostics.copied_value_nodes = diagnostics.copied_value_nodes.checked_add(copied)
                .ok_or_else(|| node_failure(node, FailureKind::IndexOverflow))?;
            SlotValue::Value(value)
        }
        Operation::EmptyEnvironment => {
            if !args.is_empty() { return Err(wrong()); }
            SlotValue::Environment(Vec::new())
        }
        Operation::Environment => {
            let [SlotValue::Source(InputValue::Environment(cell)), SlotValue::Source(InputValue::Name(name)), SlotValue::Value(value), SlotValue::Environment(previous)] = args else {
                return Err(wrong());

            entries.extend(previous.iter().cloned());
            diagnostics.environment_constructions += 1;
            diagnostics.copied_environment_entries = diagnostics.copied_environment_entries.checked_add(previous.len())
                .ok_or_else(|| node_failure(node, FailureKind::IndexOverflow))?;
            SlotValue::Environment(entries)
        }
        Operation::Sequence { kind, item_type } => {
            for value in args { if value.value_type() != *item_type { return Err(wrong()); } }
            SlotValue::Sequence { kind: *kind, items: args.iter().map(|v| (*v).clone()).collect() }
        }
        Operation::Payload { kind, fields } => {
            let Some(SlotValue::Source(raw)) = args.first().copied() else { return Err(wrong()); };
            let expected = match kind {
                PayloadKind::Frame => SourceType::Frame, PayloadKind::Control => SourceType::Control,
                PayloadKind::Helper => SourceType::Helper, PayloadKind::Task => SourceType::Task,
                PayloadKind::Result => SourceType::Result, PayloadKind::Outcome => SourceType::Outcome,
            };
            if raw.source_type() != expected { return Err(wrong()); }
            diagnostics.payload_constructions += 1;
            SlotValue::Payload(DecodedPayload { kind: *kind, raw: raw.clone(), fields: field_values(node, fields, args)? })
        }
        Operation::Snapshot { fields } => {
            let Some(SlotValue::Source(InputValue::Metadata(metadata))) = args.first().copied() else { return Err(wrong()); };
            if fields.iter().map(|f| f.field).collect::<Vec<_>>() != OUTPUT_FIELDS {
                return Err(node_failure(node, FailureKind::IncompleteOutputs));
            }
            SlotValue::Snapshot(DecodedSnapshot { metadata: metadata.clone(), fields: field_values(node, fields, args)? })
        }
    })
}

pub(super) fn validate(plan: &WitnessPlan)
    -> Result<Vec<NodeId>, WitnessFailure>
{
    let at = Location::new("plan", None, "structure");
    if plan.format != "sherlock-host-witness-plan/1" { return Err(fail(at, FailureKind::WrongOperation)); }
    let schedule = order(&plan.nodes)?;
    if schedule != plan.stable_topological_order { return Err(fail(at, FailureKind::UnstableOrder)); }
    if plan.definitions != definitions(&plan.nodes, &schedule, &plan.live_outs, plan.lifetime)? {
        return Err(fail(at, FailureKind::IncompleteBindings));
    }
    let output = plan.nodes.get(plan.output.0).ok_or_else(|| fail(at, FailureKind::IncompleteOutputs))?;
    let Operation::Snapshot { fields } = &output.operation else { return Err(fail(at, FailureKind::IncompleteOutputs)); };
    let mut expected_live: Vec<_> = fields.iter().map(|f| f.dependency.node).collect();
    expected_live.push(plan.output);
    if plan.live_outs != expected_live { return Err(fail(at, FailureKind::IncompleteOutputs)); }
    // Input declarations must be indexed once and have exactly one Read producer.
    let mut reads = vec![0usize; plan.inputs.len()];
    for node in &plan.nodes {
        if let Operation::Read(id) = node.operation {
            let count = reads.get_mut(id.0).ok_or_else(|| node_failure(node, FailureKind::MissingInput))?;
            *count += 1;
        }
    }
    if reads.iter().any(|count| *count != 1) { return Err(fail(at, FailureKind::IncompleteBindings)); }
    Ok(schedule)
}

pub(super) fn evaluate(plan: &WitnessPlan, input: &HostSnapshot)
    -> Result<WitnessEvaluation, WitnessFailure>
{
    let at = Location::new("plan", None, "structure");
    let schedule = validate(plan)?;
    let mut slots: Vec<Option<SlotValue>> = vec![None; plan.nodes.len()];
    let mut diagnostics = HostDiagnostics::default();
    for id in schedule {
        let node = &plan.nodes[id.0];
        let mut args = Vec::new();
        for dependency in &node.dependencies {
            let value = slots.get(dependency.node.0).and_then(Option::as_ref)
                .ok_or_else(|| node_failure(node, FailureKind::MissingDependency { dependency: dependency.node }))?;
            if value.value_type() != dependency.value_type {
                return Err(node_failure(node, FailureKind::WrongType { expected: dependency.value_type, actual: value.value_type() }));
            }
            args.push(value);
        }
        let result = operation(plan, input, node, &args, &mut diagnostics)?;
        if result.value_type() != node.output_type {
            return Err(node_failure(node, FailureKind::WrongType { expected: node.output_type, actual: result.value_type() }));
        }
        slots[id.0] = Some(result);
        diagnostics.evaluated_nodes += 1;
    }
    diagnostics.retained_slots = slots.len();
    let Some(SlotValue::Snapshot(decoded)) = slots.get_mut(plan.output.0).and_then(Option::take) else {
        return Err(fail(at, FailureKind::IncompleteOutputs));
    };
    Ok(WitnessEvaluation { decoded, diagnostics })
}

/// Preserve the baseline's complete validation, order and operation semantics.
/// Last-use only releases scratch values after their final ordered consumer;
/// boundary live-outs remain through publication. Cached copies are accounted
/// separately and are not reported as reclaimed scratch or semantic savings.
pub fn evaluate_with_storage(plan: &WitnessPlan, input: &HostSnapshot,
    storage: SlotStorage, mut cache: Option<&mut WitnessCache>)
    -> Result<StrategyEvaluation, WitnessFailure>
{
    let at = Location::new("plan", None, "structure");
    let schedule = dag::validate(plan)?;
    let mut scheduling = StrategyDiagnostics::default();
    if let Some(memo) = cache.as_deref_mut() {
        scheduling.scope_reset = memo.bind(plan, input);
    }
    let mut release = vec![Vec::new(); schedule.len()];
    if storage == SlotStorage::LastUse {
        for definition in &plan.definitions {
            // validate() compared these exact definitions to the existing
            // recomputation; every non-live-out last-use is a schedule position.
            if !definition.live_out {
                release[definition.last_use_order].push(definition.value);
            }
        }
    }
    let mut slots: Vec<Option<SlotValue>> = vec![None; plan.nodes.len()];
    let mut diagnostics = HostDiagnostics::default();
    let mut live = 0;
    for (position, id) in schedule.into_iter().enumerate() {
        let node = &plan.nodes[id.0];
        let result = {
            let mut args = Vec::new();
            for dependency in &node.dependencies {
                let value = slots.get(dependency.node.0).and_then(Option::as_ref)
                    .ok_or_else(|| dag::node_failure(node, FailureKind::MissingDependency {
                        dependency: dependency.node,
                    }))?;
                if value.value_type() != dependency.value_type {
                    return Err(dag::node_failure(node, FailureKind::WrongType {
                        expected: dependency.value_type, actual: value.value_type(),
                    }));
                }
                args.push(value);
            }
            // Reads are never bypassed: InputChanged/type/location behavior is
            // the existing operation's behavior, including warm evaluations.
            let memoizable = !matches!(node.operation, Operation::Read(_));
            let cached = if memoizable {
                cache.as_deref().and_then(|memo| memo.entries[id.0].as_ref())
                    .filter(|entry| entry.arguments.len() == args.len()
                        && entry.arguments.iter().zip(&args).all(|(a, b)| a == *b))
            } else { None };
            if let Some(entry) = cached {
                scheduling.cache_hits += 1;
                scheduling.copied_cached_slot_values += 1;
                entry.value.clone()
            } else {
                let value = dag::operation(plan, input, node, &args, &mut diagnostics)?;
                diagnostics.evaluated_nodes += 1;
                if memoizable {
                    if let Some(memo) = cache.as_deref_mut() {
                        scheduling.cache_misses += 1;
                        memo.entries[id.0] = Some(MemoEntry {
                            arguments: args.iter().map(|value| (**value).clone()).collect(),
                            value: value.clone(),
                        });
                    }
                }
                value
            }
        };
        if result.value_type() != node.output_type {
            return Err(dag::node_failure(node, FailureKind::WrongType {
                expected: node.output_type, actual: result.value_type(),
            }));
        }
        slots[id.0] = Some(result);
        live += 1;
        scheduling.scheduled_nodes += 1;
        scheduling.peak_live_slots = scheduling.peak_live_slots.max(live);
        for dead in &release[position] {
            if slots[dead.0].take().is_some() {
                live -= 1;
                scheduling.reclaimed_slots += 1;
            }
        }
    }
    diagnostics.retained_slots = live;
    scheduling.retained_slots_at_publication = live;
    if let Some(memo) = cache.as_deref() {
        scheduling.cached_value_slots = memo.value_slots();
        scheduling.cached_argument_slots = memo.argument_slots();
    }
    let Some(SlotValue::Snapshot(decoded)) = slots.get_mut(plan.output.0).and_then(Option::take) else {
        return Err(fail(at, FailureKind::IncompleteOutputs));
    };
    Ok(StrategyEvaluation {
        witness: WitnessEvaluation { decoded, diagnostics }, scheduling,
    })
}

/// One real owner step, followed solely by reconstruction of its saved raw
/// boundaries. On any tooling failure, the actual step and available boundaries
/// are returned unchanged; no rollback, retry or extra transition is performed.
pub fn record_step_with_storage(admitted: &Admitted, image: &mut MachineImage,
    storage: SlotStorage, mut cache: Option<&mut WitnessRecordCache>)
    -> Result<StrategyObservedStep, StrategyStepError>
{
    let observed = observation::record_step(admitted, image).map_err(StrategyStepError::Owner)?;
    let (before_raw, after_raw) = match &observed {
        ObservedStep::Event(record) => (&record.before, &record.after),
        ObservedStep::Terminal { before, after } => (before, after),
    };
    let before_cache = cache.as_deref_mut().map(|memo| &mut memo.before);
    let before = match materialize_with_storage(before_raw, LifetimeUnit::RecordBefore, storage, before_cache) {
        Ok(value) => value,
        Err(failure) => return Err(StrategyStepError::Materialization {
            side: WitnessSide::Before, failure, observed: Box::new(observed),
            completed_before: None,
        }),
    };
    let after_cache = cache.as_deref_mut().map(|memo| &mut memo.after);
    let after = match materialize_with_storage(after_raw, LifetimeUnit::RecordAfter, storage, after_cache) {
        Ok(value) => value,
        Err(failure) => return Err(StrategyStepError::Materialization {
            side: WitnessSide::After, failure, observed: Box::new(observed),
            completed_before: Some(Box::new(before)),
        }),
    };
    Ok(StrategyObservedStep { observed, before, after })
}

/// Yield every complete observed record/terminal boundary in order. The sink
/// may stop after a delivered boundary, preserving that exact caller-owned
/// MachineImage. No inferred work/event, hidden step, image substitution or
/// application exhaustion is introduced. Execution classification remains the
/// existing caller's obligation, independently of this source-only interface.
pub fn stream_owner_steps<F>(admitted: &Admitted, image: &mut MachineImage,
    mut cache: Option<&mut WitnessRecordCache>, mut sink: F)
    -> Result<StreamStop, StrategyStepError>
where F: FnMut(&StrategyObservedStep) -> StreamAction
{
    loop {
        let step = record_step_with_storage(admitted, image, SlotStorage::LastUse, cache.as_deref_mut())?;
        let terminal = matches!(&step.observed, ObservedStep::Terminal { .. });
        let action = sink(&step);
        // No decoded/raw record history is held across loop iterations.
        drop(step);
        if terminal { return Ok(StreamStop::Terminal); }
        if action == StreamAction::Stop { return Ok(StreamStop::Sink); }
    }
}

fn inspect_variants(materialized: &wp::StrategyMaterializedSnapshot) {
    let raw = &materialized.raw;
    let plan = &materialized.plan;
    let baseline = wp::evaluate_uncached(plan, raw).unwrap_or_else(|e| panic!("uncached {e:?}"));
    assert_eq!(materialized.evaluation.witness.decoded, baseline.decoded);
    let reclaimed = wp::evaluate_last_use(plan, raw).unwrap_or_else(|e| panic!("last-use {e:?}"));

    let live_outs = plan.live_outs.iter().copied().collect::<BTreeSet<_>>().len();
    assert_eq!(reclaimed.scheduling.reclaimed_slots, plan.nodes.len() - live_outs);
    assert_eq!(reclaimed.scheduling.retained_slots_at_publication, live_outs);
    assert!(reclaimed.scheduling.peak_live_slots <= plan.nodes.len());
    assert_eq!(reclaimed.witness.diagnostics.evaluated_nodes, plan.nodes.len());
    assert_eq!(reclaimed.witness.diagnostics.value_constructions, baseline.diagnostics.value_constructions);
    assert_eq!(reclaimed.witness.diagnostics.environment_constructions, baseline.diagnostics.environment_constructions);
    assert_eq!(reclaimed.witness.diagnostics.copied_value_nodes, baseline.diagnostics.copied_value_nodes);
    assert_eq!(reclaimed.witness.diagnostics.copied_environment_entries, baseline.diagnostics.copied_environment_entries);

    let mut cache = wp::WitnessCache::default();
    let cold = wp::evaluate_memoized(plan, raw, &mut cache, wp::SlotStorage::RetainAll)
        .unwrap_or_else(|e| panic!("cold {e:?}"));
    assert_eq!(cold.witness.decoded, baseline.decoded);
    assert!(cold.scheduling.scope_reset);
    assert_eq!(cold.scheduling.cache_hits, 0);
    assert_eq!(cold.scheduling.cache_misses, plan.nodes.len() - plan.inputs.len());
    assert_eq!(cold.witness.diagnostics, baseline.diagnostics);
    let warm = wp::evaluate_memoized(plan, raw, &mut cache, wp::SlotStorage::LastUse)
        .unwrap_or_else(|e| panic!("warm {e:?}"));

fn resolved_result(snapshot: &wp::DecodedSnapshot) -> &wp::DecodedValue {

    let wp::SlotValue::Payload(payload) = outcome else { panic!("complete outcome payload") };
    assert_eq!(payload.kind, wp::PayloadKind::Outcome);

}
fn exercise(source: CoreProgram, expected: u128, stop_and_resume: bool) {
    let (bounds, schema) = layout();
    let owner = host_owner_engine();
    let admitted = tooling(owner.admit(&source, &bounds, &schema));
    let mut image = tooling(owner.admit_run(&admitted, &RunIntC::Nat(4096),
        &RunIntC::Nat(64), &FinanceInput::Absent));
    let initial = co::capture(&admitted, &image).unwrap_or_else(|_| panic!("bound initial"));
    let mut previous = initial;
    let mut records = 0usize;
    let mut saw_pending_call = false;
    let mut final_decoded = None;
    let mut cache = wp::WitnessRecordCache::default();
    let mut sink = |step: &wp::StrategyObservedStep| {
        assert_eq!(step.before.raw, previous);
        let (raw_before, raw_after) = match &step.observed {
            co::ObservedStep::Event(record) => (&record.before, &record.after),
            co::ObservedStep::Terminal { before, after } => (before, after),
        };
        assert_eq!(&step.before.raw, raw_before);
        assert_eq!(&step.after.raw, raw_after);
        inspect_variants(&step.before);
        inspect_variants(&step.after);
        saw_pending_call |= matches!(raw_before.image.control, ControlC::Invoke { .. })
            && !raw_before.image.frames.is_empty();
        if matches!(raw_after.image.outcome, OutcomeC::Halted { .. }) {
            final_decoded = Some(step.after.evaluation.witness.decoded.clone());
        }
        previous = raw_after.clone();
        records += 1;
        if stop_and_resume && records == 1 { wp::StreamAction::Stop }
        else { wp::StreamAction::Continue }
    };
    let first = wp::stream_owner_steps(&admitted, &mut image, Some(&mut cache), &mut sink)
        .unwrap_or_else(|_| panic!("real owner stream"));
    if stop_and_resume {
        assert_eq!(first, wp::StreamStop::Sink);
        // Resume the SAME image/cache after the delivered boundary: no replay.
        let resumed = wp::stream_owner_steps(&admitted, &mut image, Some(&mut cache), &mut sink)
            .unwrap_or_else(|_| panic!("same-image continuation"));
        assert_eq!(resumed, wp::StreamStop::Terminal);
    } else { assert_eq!(first, wp::StreamStop::Terminal); }
    drop(sink);
    let returned = co::capture(&admitted, &image).unwrap_or_else(|_| panic!("bound returned"));
    assert_eq!(returned, previous);
    assert!(records > 1);
    assert!(saw_pending_call);
    assert!(matches!(returned.image.outcome, OutcomeC::Halted { .. }));
    let decoded = final_decoded.expect("actual halted boundary");
    assert_eq!(resolved_result(&decoded), &wp::DecodedValue {
        nodes: vec![wp::ValueNode::UInt(expected)], root: 0,
    });
}

    fn add(&mut self, operation: Operation, deps: Vec<Dependency>, output_type: ValueType, at: Location) -> NodeId {
        let id = NodeId(self.nodes.len());
        self.nodes.push(Node { id, location: at, operation, dependencies: deps, output_type });

    }
    fn dependency(&self, id: NodeId) -> Dependency {

    fn read(&mut self, path: InputPath, at: Location) -> Result<NodeId, WitnessFailure> {
        if let Some(id) = self.reads.get(&path) { return Ok(*id); }
        let value = dag::read(self.input, path, at)?;
        let value_type = value.source_type();

        let node = self.add(Operation::Read(id), vec![], ValueType::Source(value_type), at);

    }
    fn field(&self, name: &'static str, id: NodeId) -> FieldBinding {
        FieldBinding { field: name, dependency: self.dependency(id) }
    }
    fn bind(&mut self, source: NodeId, target: NodeId, at: Location) -> NodeId {
        self.add(Operation::Bind, vec![self.dependency(source),self.dependency(target)],
            self.nodes[target.0].output_type,at)
    }
    fn sequence(&mut self, kind: SequenceKind, item_type: ValueType, ids: Vec<NodeId>, at: Location) -> NodeId {
        let deps = ids.into_iter().map(|id| self.dependency(id)).collect();
        self.add(Operation::Sequence { kind, item_type }, deps, ValueType::Sequence(kind), at)
    }
    fn indices(&self, start: usize, len: usize, table_len: usize, at: Location) -> Result<std::ops::Range<usize>, WitnessFailure> {
        // Empty semantic slices do not dereference a cursor; retain raw start.
        if len == 0 { return Ok(0..0); }
        let end = start.checked_add(len).ok_or_else(|| fail(at, FailureKind::IndexOverflow))?;
        if end > table_len { return Err(fail(at, FailureKind::MissingInput)); }
        Ok(start..end)
    }
    fn source_sequence(&mut self, kind: SequenceKind, ty: SourceType, paths: Vec<InputPath>, at: Location) -> Result<NodeId, WitnessFailure> {
        let ids = paths.into_iter().map(|path| self.read(path, at)).collect::<Result<Vec<_>, _>>()?;
        Ok(self.sequence(kind, ValueType::Source(ty), ids, at))
    }
    fn values_slice(&mut self, slice: SliceC, at: Location) -> Result<NodeId, WitnessFailure> {
        let indices = self.indices(slice.start, slice.len, self.input.image.value_slots.len(), at)?;
        let mut ids = Vec::new();
        for index in indices {
            // Explicit Read also binds the ordered slice member, not only its target.
            let member = self.read(InputPath::ValueSlot(index), at)?;
            let value = self.value(self.input.image.value_slots[index], at)?;
            ids.push(self.bind(member,value,at));
        }
        Ok(self.sequence(SequenceKind::Values, ValueType::Value, ids, at))
    }
    fn expression_slice(&mut self, slice: SliceC, at: Location) -> Result<NodeId, WitnessFailure> {
        let indices = self.indices(slice.start, slice.len, self.input.admission.view.children.len(), at)?;
        let mut ids = Vec::new();
        for index in indices {
            let member = self.read(InputPath::Child(index), at)?;
            let expression = self.read(InputPath::Expression(self.input.admission.view.children[index]), at)?;
            ids.push(self.bind(member,expression,at));
        }
        Ok(self.sequence(SequenceKind::Expressions, ValueType::Source(SourceType::Expression), ids, at))
    }
    fn parameter_slice(&mut self, slice: SliceC, at: Location) -> Result<NodeId, WitnessFailure> {
        let indices = self.indices(slice.start, slice.len, self.input.admission.view.parameters.len(), at)?;
        self.source_sequence(SequenceKind::Parameters, SourceType::Parameter,
            indices.map(InputPath::Parameter).collect(), at)
    }
    fn arm_slice(&mut self, slice: SliceC, at: Location) -> Result<NodeId, WitnessFailure> {
        let indices = self.indices(slice.start, slice.len, self.input.admission.view.arms.len(), at)?;
        self.source_sequence(SequenceKind::Arms, SourceType::Arm, indices.map(InputPath::Arm).collect(), at)
    }

    fn value(&mut self, root: RefC, at: Location) -> Result<NodeId, WitnessFailure> {
        enum Work { Enter(RefC), Exit { reference: RefC, row: NodeId, fields: Vec<NodeId> } }
        let mut work = vec![Work::Enter(root)];
        let mut ancestors = BTreeSet::new();
        let mut results = Vec::new();
        while let Some(item) = work.pop() {
            match item {
                Work::Enter(reference) => {
                    if !ancestors.insert(reference_key(reference)) { return Err(fail(at, FailureKind::CyclicValue)); }
                    let row = self.read(match reference { RefC::Const(n) => InputPath::Constant(n), RefC::Data(n) => InputPath::Data(n) }, at)?;
                    let atom = match &self.nodes[row.0].operation {
                        Operation::Read(binding) => self.inputs[binding.0].immutable_value.clone(),
                        _ => return Err(fail(at, FailureKind::WrongOperation)),
                    };
                    let (arity, fields_slice) = shape(&atom, at)?;
                    let mut field_nodes = Vec::new();
                    let mut children = Vec::new();
                    if let Some((start, len)) = fields_slice {
                        if len != arity { return Err(fail(at, FailureKind::WrongArity { expected: arity, actual: len })); }
                        let table_len = match reference { RefC::Const(_) => self.input.admission.view.constant_fields.len(), RefC::Data(_) => self.input.image.data_fields.len() };
                        for index in self.indices(start, len, table_len, at)? {
                            let (path, child) = match reference {
                                RefC::Const(_) => (InputPath::ConstantField(index), RefC::Const(self.input.admission.view.constant_fields[index])),
                                RefC::Data(_) => (InputPath::DataField(index), self.input.image.data_fields[index]),
                            };
                            field_nodes.push(self.read(path, at)?);
                            children.push(child);
                        }
                    }
                    work.push(Work::Exit { reference, row, fields: field_nodes });
                    for child in children.into_iter().rev() { work.push(Work::Enter(child)); }
                }
                Work::Exit { reference, row, fields } => {
                    let start = results.len().checked_sub(fields.len()).ok_or_else(|| fail(at, FailureKind::IncompleteBindings))?;
                    let children = results.split_off(start);
                    let mut deps = vec![self.dependency(row)];
                    for (field, child) in fields.iter().zip(children) {
                        deps.push(self.dependency(*field)); deps.push(self.dependency(child));
                    }
                    let id = self.add(Operation::Value { source: reference, arity: fields.len() }, deps, ValueType::Value, at);
                    results.push(id);
                    ancestors.remove(&reference_key(reference));
                }
            }
        }
        if results.len() != 1 { return Err(fail(at, FailureKind::IncompleteBindings)); }
        Ok(results[0])
    }
    fn environment(&mut self, root: EnvRootC, at: Location) -> Result<NodeId, WitnessFailure> {
        let mut chain = Vec::new();
        let mut seen = BTreeSet::new();
        let mut cursor = root;
        while let Some(index) = cursor {
            if !seen.insert(index) { return Err(fail(at, FailureKind::CyclicEnvironment)); }
            let cell = *self.input.image.env.get(index).ok_or_else(|| fail(at, FailureKind::MissingInput))?;
            chain.push((index, cell)); cursor = cell.previous;
        }
        let mut previous = self.add(Operation::EmptyEnvironment, vec![], ValueType::Environment, at);
        for (index, cell) in chain.into_iter().rev() {
            let raw = self.read(InputPath::Environment(index), at)?;
            let name = self.read(InputPath::Name(cell.binder), at)?;
            let value = self.value(cell.value, at)?;
            let deps = [raw, name, value, previous].into_iter().map(|id| self.dependency(id)).collect();
            previous = self.add(Operation::Environment, deps, ValueType::Environment, at);
        }
        Ok(previous)
    }
    fn payload(&mut self, path: InputPath, kind: PayloadKind, at: Location) -> Result<NodeId, WitnessFailure> {
        let raw = self.read(path, at)?;
        let Operation::Read(binding) = self.nodes[raw.0].operation else { return Err(fail(at, FailureKind::WrongOperation)); };
        let atom = self.inputs[binding.0].immutable_value.clone();
        let mut fields = Vec::new();
        // Each match is exhaustive over current owner payload variants. Scalars,
        // status/None tags, cursors and origins also remain in the exact raw atom.
        match atom {
            InputValue::Frame(frame) => self.frame_fields(frame, &mut fields, at)?,
            InputValue::Control(control) => self.control_fields(control, &mut fields, at)?,
            InputValue::Helper(helper) => self.helper_fields(helper, &mut fields, at)?,
            InputValue::Task(task) => self.task_fields(task, &mut fields, at)?,
            InputValue::Result(result) => self.result_fields(result, &mut fields, at)?,
            InputValue::Outcome(outcome) => match outcome {
                OutcomeC::Halted { value } => self.value_field(&mut fields, "value", value, at)?,
                OutcomeC::Running | OutcomeC::Rejected { .. } | OutcomeC::Exhausted { .. } => (),
            },
            _ => return Err(fail(at, FailureKind::WrongOperation)),
        }
        let mut deps = vec![self.dependency(raw)]; deps.extend(fields.iter().map(|f| f.dependency));
        Ok(self.add(Operation::Payload { kind, fields }, deps, ValueType::Payload(kind), at))
    }
    fn source_field(&mut self, f: &mut Vec<FieldBinding>, name: &'static str, path: InputPath, at: Location) -> Result<(), WitnessFailure> {
        let id = self.read(path, at)?; f.push(self.field(name, id)); Ok(())
    }
    fn value_field(&mut self, f: &mut Vec<FieldBinding>, name: &'static str, value: RefC, at: Location) -> Result<(), WitnessFailure> {
        let id = self.value(value, at)?; f.push(self.field(name, id)); Ok(())
    }
    fn env_field(&mut self, f: &mut Vec<FieldBinding>, name: &'static str, value: EnvRootC, at: Location) -> Result<(), WitnessFailure> {
        let id = self.environment(value, at)?; f.push(self.field(name, id)); Ok(())
    }
    fn values_field(&mut self, f: &mut Vec<FieldBinding>, name: &'static str, value: SliceC, at: Location) -> Result<(), WitnessFailure> {
        let id = self.values_slice(value, at)?; f.push(self.field(name, id)); Ok(())
    }
    fn expressions_field(&mut self, f: &mut Vec<FieldBinding>, name: &'static str, value: SliceC, at: Location) -> Result<(), WitnessFailure> {
        let id = self.expression_slice(value, at)?; f.push(self.field(name, id)); Ok(())
    }
    fn arms_field(&mut self, f: &mut Vec<FieldBinding>, name: &'static str, value: SliceC, at: Location) -> Result<(), WitnessFailure> {
        let id = self.arm_slice(value, at)?; f.push(self.field(name, id)); Ok(())
    }
    fn frame_fields(&mut self, frame: FrameRowC, f: &mut Vec<FieldBinding>, at: Location) -> Result<(), WitnessFailure> {
        match frame {
            FrameRowC::LetF { binder, body, saved_env, origin } => {
                self.source_field(f,"binder",InputPath::Name(binder),at)?;
                self.source_field(f,"body",InputPath::Expression(body),at)?;
                self.env_field(f,"saved_env",saved_env,at)?;
                self.source_field(f,"origin",InputPath::Occurrence(origin),at)?;
            }
            FrameRowC::LeftF { right, saved_env, origin, .. } => {
                self.source_field(f,"right",InputPath::Expression(right),at)?;
                self.env_field(f,"saved_env",saved_env,at)?;
                self.source_field(f,"origin",InputPath::Occurrence(origin),at)?;
            }
            FrameRowC::RightF { left, origin, .. } => {
                self.value_field(f,"left",left,at)?;
                self.source_field(f,"origin",InputPath::Occurrence(origin),at)?;
            }
            FrameRowC::UnaryF { origin, .. } => self.source_field(f,"origin",InputPath::Occurrence(origin),at)?,
            FrameRowC::IfF { then_expr, else_expr, saved_env, origin } => {
                self.source_field(f,"then_expr",InputPath::Expression(then_expr),at)?;
                self.source_field(f,"else_expr",InputPath::Expression(else_expr),at)?;
                self.env_field(f,"saved_env",saved_env,at)?;
                self.source_field(f,"origin",InputPath::Occurrence(origin),at)?;
            }
            FrameRowC::MatchF { arms, saved_env, origin } => {
                self.arms_field(f,"arms",arms,at)?; self.env_field(f,"saved_env",saved_env,at)?;
                self.source_field(f,"origin",InputPath::Occurrence(origin),at)?;
            }
            FrameRowC::ArgsF { function_name, completed, rest, saved_env, origin } => {
                self.source_field(f,"function_name",InputPath::Name(function_name),at)?;
                self.values_field(f,"completed",completed,at)?;
                self.expressions_field(f,"rest",rest,at)?;
                self.env_field(f,"saved_env",saved_env,at)?;
                self.source_field(f,"origin",InputPath::Occurrence(origin),at)?;
            }
            FrameRowC::ReturnF { saved_env, origin, .. } | FrameRowC::ScopeF { saved_env, origin } => {
                self.env_field(f,"saved_env",saved_env,at)?;
                self.source_field(f,"origin",InputPath::Occurrence(origin),at)?;
            }
        }
        Ok(())
    }
    fn control_fields(&mut self, control: ControlC, f: &mut Vec<FieldBinding>, at: Location) -> Result<(), WitnessFailure> {
        match control {
            ControlC::Eval { expr } => self.source_field(f,"expr",InputPath::Expression(expr),at)?,
            ControlC::Return { value } => self.value_field(f,"value",value,at)?,
            ControlC::ScheduleNext { next, env, .. } => {
                let frame = self.payload(InputPath::ScheduledFrame,PayloadKind::Frame,at)?;
                f.push(self.field("frame",frame));
                self.source_field(f,"next",InputPath::Expression(next),at)?;
                self.env_field(f,"env",env,at)?;
            }
            ControlC::Invoke { function_name, args } => {
                self.source_field(f,"function_name",InputPath::Name(function_name),at)?;
                self.values_field(f,"args",args,at)?;
            }
            ControlC::Enter { function, args } => {
                if let Some(index) = function { self.source_field(f,"function",InputPath::Function(index),at)?; }
                self.values_field(f,"args",args,at)?;
            }
            ControlC::Bound { binding, body, .. } => {
                if let BoundBinding::Ok { env } = binding { self.env_field(f,"binding",env,at)?; }
                self.source_field(f,"body",InputPath::Expression(body),at)?;
            }
            ControlC::Cases { value, arms, saved_env } => {
                self.value_field(f,"value",value,at)?; self.arms_field(f,"arms",arms,at)?;
                self.env_field(f,"saved_env",saved_env,at)?;
            }
            ControlC::CaseResult { matching, value, body, rest, saved_env } => {
                if let CaseMatching::Matched { env } = matching { self.env_field(f,"matching",env,at)?; }
                self.value_field(f,"value",value,at)?;
                self.source_field(f,"body",InputPath::Expression(body),at)?;
                self.arms_field(f,"rest",rest,at)?; self.env_field(f,"saved_env",saved_env,at)?;
            }
            ControlC::PrimResult { result } => match result {
                PrimOutcome::Value { value } => self.value_field(f,"value",value,at)?, PrimOutcome::Error { .. } => (),
            },
            ControlC::Error { .. } => (),
        }
        Ok(())
    }
    fn helper_fields(&mut self, helper: Option<HelperControlC>, f: &mut Vec<FieldBinding>, at: Location) -> Result<(), WitnessFailure> {
        let Some(helper) = helper else { return Ok(()); };
        match helper {
            HelperControlC::Scalar { operand_a, operand_b, .. } => {
                self.value_field(f,"operand_a",operand_a,at)?;
                if let Some(value) = operand_b { self.value_field(f,"operand_b",value,at)?; }
            }
            HelperControlC::EnvGet { binder, cursor, found, .. } => {
                self.source_field(f,"binder",InputPath::Name(binder),at)?;
                self.env_field(f,"cursor",cursor,at)?;
                if let Some(value) = found { self.value_field(f,"found",value,at)?; }
            }
            HelperControlC::EnvPut { binder, value, base, new_root, .. } => {
                self.source_field(f,"binder",InputPath::Name(binder),at)?;
                self.value_field(f,"value",value,at)?;
                self.env_field(f,"base",base,at)?; self.env_field(f,"new_root",new_root,at)?;
            }
            HelperControlC::Find { function_name, found, .. } => {
                self.source_field(f,"function_name",InputPath::Name(function_name),at)?;
                if let Some(index) = found { self.source_field(f,"found",InputPath::Function(index),at)?; }
            }
            HelperControlC::Bind { function, params, args, partial, dup_cursor, .. } => {
                self.source_field(f,"function",InputPath::Function(function),at)?;
                let parameters = self.parameter_slice(params,at)?; f.push(self.field("params",parameters));
                self.values_field(f,"args",args,at)?;
                self.env_field(f,"partial",partial,at)?; self.env_field(f,"dup_cursor",dup_cursor,at)?;
            }
            HelperControlC::Append { src, value, staged, .. } => {
                self.values_field(f,"src",src,at)?; self.value_field(f,"value",value,at)?;
                if let Some(value) = staged { self.value_field(f,"staged",value,at)?; }
            }
            HelperControlC::Validate { root, .. } => self.value_field(f,"root",root,at)?,
            HelperControlC::Equal { left, right, .. } => {
                self.value_field(f,"left",left,at)?; self.value_field(f,"right",right,at)?;
            }
            HelperControlC::Match { scrutinee, arm, saved_env, partial, pat_tag, x, y, staged, phase, .. } => {
                self.value_field(f,"scrutinee",scrutinee,at)?;
                self.source_field(f,"arm",InputPath::Arm(arm),at)?;
                self.env_field(f,"saved_env",saved_env,at)?; self.env_field(f,"partial",partial,at)?;
                // Entry carries PVar/x=0 placeholders until the Arm read. They
                // are raw state, not resolved names before that read occurs.
                if phase != MatchPhase::Arm { match pat_tag {
                    PatternTag::PVar | PatternTag::PLeaf => self.source_field(f,"x",InputPath::Name(x),at)?,
                    PatternTag::PCons | PatternTag::PBranch => {
                        self.source_field(f,"x",InputPath::Name(x),at)?;
                        self.source_field(f,"y",InputPath::Name(y),at)?;
                    }
                    PatternTag::PNil | PatternTag::WildcardExtensionReserved | PatternTag::ConstructorExtensionReserved => (),
                } }
                if let Some(value) = staged { self.value_field(f,"staged",value,at)?; }
            }
        }
        Ok(())
    }
    fn task_fields(&mut self, task: HelperTaskC, f: &mut Vec<FieldBinding>, at: Location) -> Result<(), WitnessFailure> {
        match task {
            HelperTaskC::ValidateTask { r, .. } => self.value_field(f,"r",r,at)?,
            HelperTaskC::EqualTask { left, right, .. } => {
                self.value_field(f,"left",left,at)?; self.value_field(f,"right",right,at)?;
            }
            HelperTaskC::PairTask { binder, arg, .. } => {
                self.source_field(f,"binder",InputPath::Name(binder),at)?;
                self.value_field(f,"arg",arg,at)?;
            }
        }
        Ok(())
    }
    fn result_fields(&mut self, result: Option<HelperResultC>, f: &mut Vec<FieldBinding>, at: Location) -> Result<(), WitnessFailure> {
        let Some(result) = result else { return Ok(()); };
        match result {
            HelperResultC::EnvFound { value } | HelperResultC::ScalarValue { value } | HelperResultC::EqualDone { value } => self.value_field(f,"value",value,at)?,
            HelperResultC::EnvUpdated { cell } => self.env_field(f,"cell",Some(cell),at)?,
            HelperResultC::FindFound { function } => self.source_field(f,"function",InputPath::Function(function),at)?,
            HelperResultC::BindOk { env } | HelperResultC::MatchMatched { env } => self.env_field(f,"env",env,at)?,
            HelperResultC::AppendDone { seq } => self.values_field(f,"seq",seq,at)?,
            HelperResultC::Validated { .. } | HelperResultC::EnvMissing | HelperResultC::FindMissing |
            HelperResultC::BindBad | HelperResultC::ScalarError { .. } | HelperResultC::ScalarSelector { .. } |
            HelperResultC::ScalarTypeTest { .. } | HelperResultC::MatchNoMatch => (),
        }
        Ok(())
    }
}

fn reference_key(value: RefC) -> (u8, usize) { match value { RefC::Const(n) => (0,n), RefC::Data(n) => (1,n) } }
fn shape(atom: &InputValue, at: Location) -> Result<(usize, Option<(usize, usize)>), WitnessFailure> {
    Ok(match atom {
        InputValue::Constant(value) => match value {
            ConstantSnapshot::Cons { fields, .. } | ConstantSnapshot::Branch { fields, .. } => (2,Some((fields.start,fields.len))),
            ConstantSnapshot::Leaf { fields, .. } => (1,Some((fields.start,fields.len))),
            ConstantSnapshot::U128 { .. } | ConstantSnapshot::Bool { .. } | ConstantSnapshot::Nil { .. } | ConstantSnapshot::U128OutOfRange { .. } => (0,None),
        },
        InputValue::Data(value) => match value {
            DataSnapshot::Cons { fields } | DataSnapshot::Branch { fields } => (2,Some((fields.start,fields.len))),
            DataSnapshot::Leaf { fields } => (1,Some((fields.start,fields.len))),
            DataSnapshot::U128 { .. } | DataSnapshot::Bool { .. } | DataSnapshot::Nil => (0,None),
        },
        _ => return Err(fail(at, FailureKind::WrongOperation)),
    })
}

pub(super) fn build(input: &HostSnapshot, lifetime: LifetimeUnit) -> Result<WitnessPlan, WitnessFailure> {
    let mut b = Builder { input, inputs: Vec::new(), reads: BTreeMap::new(), nodes: Vec::new() };
    let at = Location::new("snapshot",None,"outputs");
    let metadata = b.read(InputPath::Metadata,at)?;
    let mut fields = Vec::new();
    let mut constants = Vec::new();
    for n in 0..input.admission.view.constants.len() { constants.push(b.value(RefC::Const(n),Location::new("constants",Some(n),"value"))?); }
    let id = b.sequence(SequenceKind::Values,ValueType::Value,constants,at); fields.push(b.field("constants",id));
    let id = b.source_sequence(SequenceKind::References,SourceType::Reference,(0..input.admission.view.constant_fields.len()).map(InputPath::ConstantField).collect(),at)?; fields.push(b.field("constant_fields",id));
    let mut data = Vec::new();
    for n in 0..input.image.data.len() { data.push(b.value(RefC::Data(n),Location::new("data",Some(n),"value"))?); }
    let id = b.sequence(SequenceKind::Values,ValueType::Value,data,at); fields.push(b.field("data",id));
    let id = b.source_sequence(SequenceKind::References,SourceType::Reference,(0..input.image.data_fields.len()).map(InputPath::DataField).collect(),at)?; fields.push(b.field("data_fields",id));
    let mut envs = Vec::new();
    for n in 0..input.image.env.len() { envs.push(b.environment(Some(n),Location::new("environment_cells",Some(n),"chain"))?); }
    let id = b.sequence(SequenceKind::Environments,ValueType::Environment,envs,at); fields.push(b.field("environment_cells",id));
    let id = b.environment(input.image.env_root,Location::new("snapshot",None,"current_environment"))?; fields.push(b.field("current_environment",id));
    let mut frames = Vec::new();
    // B stores bottom-to-top; decoded reference-facing frames are top-first.
    for n in (0..input.image.frames.len()).rev() { frames.push(b.payload(InputPath::Frame(n),PayloadKind::Frame,Location::new("frames",Some(n),"payload"))?); }
    let id = b.sequence(SequenceKind::Frames,ValueType::Payload(PayloadKind::Frame),frames,at); fields.push(b.field("frames",id));
    for (name,path,kind) in [("control",InputPath::Control,PayloadKind::Control),("helper",InputPath::Helper,PayloadKind::Helper)] {
        let id = b.payload(path,kind,Location::new("snapshot",None,name))?; fields.push(b.field(name,id));
    }
    let mut tasks = Vec::new();
    for n in 0..input.image.tasks.len() { tasks.push(b.payload(InputPath::Task(n),PayloadKind::Task,Location::new("tasks",Some(n),"payload"))?); }
    let id = b.sequence(SequenceKind::Tasks,ValueType::Payload(PayloadKind::Task),tasks,at); fields.push(b.field("tasks",id));
    let id = b.payload(InputPath::Result,PayloadKind::Result,Location::new("snapshot",None,"result"))?; fields.push(b.field("result",id));
    let id = b.values_slice(SliceC { start:0,len:input.image.value_slots.len() },Location::new("snapshot",None,"value_slots"))?; fields.push(b.field("value_slots",id));
    let id = b.payload(InputPath::Outcome,PayloadKind::Outcome,Location::new("snapshot",None,"outcome"))?; fields.push(b.field("outcome",id));
    let p = &input.admission.view;
    for (name,kind,ty,paths) in [
        ("names",SequenceKind::Names,SourceType::Name,(0..p.names.len()).map(InputPath::Name).collect()),
        ("expressions",SequenceKind::Expressions,SourceType::Expression,(0..p.expressions.len()).map(InputPath::Expression).collect()),
        ("parameters",SequenceKind::Parameters,SourceType::Parameter,(0..p.parameters.len()).map(InputPath::Parameter).collect()),
        ("arms",SequenceKind::Arms,SourceType::Arm,(0..p.arms.len()).map(InputPath::Arm).collect()),
        ("patterns",SequenceKind::Patterns,SourceType::Pattern,(0..p.patterns.len()).map(InputPath::Pattern).collect()),
        ("occurrences",SequenceKind::Occurrences,SourceType::Occurrence,(0..p.occurrences.len()).map(InputPath::Occurrence).collect()),
        ("occurrence_paths",SequenceKind::Indices,SourceType::Index,(0..p.occurrence_paths.len()).map(InputPath::OccurrencePath).collect()),
        ("children",SequenceKind::Indices,SourceType::Index,(0..p.children.len()).map(InputPath::Child).collect()),
    ] {
        let id = b.source_sequence(kind,ty,paths,at)?; fields.push(b.field(name,id));
    }
    let mut deps = vec![b.dependency(metadata)]; deps.extend(fields.iter().map(|f| f.dependency));
    let mut live_outs: Vec<_> = fields.iter().map(|f| f.dependency.node).collect();
    let output = b.add(Operation::Snapshot { fields },deps,ValueType::Snapshot,at);
    live_outs.push(output);
    let stable_topological_order = dag::order(&b.nodes)?;
    let definitions = dag::definitions(&b.nodes,&stable_topological_order,&live_outs,lifetime)?;
    Ok(WitnessPlan { format:"sherlock-host-witness-plan/1",lifetime,inputs:b.inputs,nodes:b.nodes,stable_topological_order,definitions,live_outs,output })
}

fn shifted(node: &ValueNode, offset: usize, at: Location) -> Result<ValueNode, WitnessFailure> {
    let add = |index:usize| index.checked_add(offset).ok_or_else(|| fail(at,FailureKind::IndexOverflow));
    Ok(match *node {
        ValueNode::UInt(n) => ValueNode::UInt(n), ValueNode::Bool(v) => ValueNode::Bool(v),
        ValueNode::Nil => ValueNode::Nil, ValueNode::U128OutOfRange => ValueNode::U128OutOfRange,
        ValueNode::Cons { head,tail } => ValueNode::Cons { head:add(head)?,tail:add(tail)? },
        ValueNode::Leaf { value } => ValueNode::Leaf { value:add(value)? },
        ValueNode::Branch { left,right } => ValueNode::Branch { left:add(left)?,right:add(right)? },
    })
}
pub(super) fn evaluate_value(plan: &WitnessPlan, node: &Node, source: RefC, arity: usize, args: &[&SlotValue]) -> Result<DecodedValue, WitnessFailure> {
    let at = node.location;
    let expected_len = arity.checked_mul(2).and_then(|v|v.checked_add(1)).ok_or_else(|| fail(at,FailureKind::IndexOverflow))?;
    if args.len() != expected_len { return Err(fail(at,FailureKind::IncompleteBindings)); }
    let Some(SlotValue::Source(atom)) = args.first().copied() else { return Err(fail(at,FailureKind::WrongOperation)); };
    match (source,atom) {
        (RefC::Const(_),InputValue::Constant(_)) | (RefC::Data(_),InputValue::Data(_)) => (),
        _ => return Err(fail(at,FailureKind::WrongOperation)),
    }
    let row_node = plan.nodes.get(node.dependencies[0].node.0).ok_or_else(|| fail(at,FailureKind::IncompleteBindings))?;
    let Operation::Read(binding) = row_node.operation else { return Err(fail(at,FailureKind::IncompleteBindings)); };
    let expected_path = match source { RefC::Const(n) => InputPath::Constant(n),RefC::Data(n) => InputPath::Data(n) };
    if plan.inputs.get(binding.0).map(|b| b.path) != Some(expected_path) { return Err(fail(at,FailureKind::IncompleteBindings)); }
    if shape(atom,at)?.0 != arity { return Err(fail(at,FailureKind::WrongArity { expected:shape(atom,at)?.0,actual:arity })); }
    let mut nodes = Vec::new(); let mut roots = Vec::new();
    for (index,pair) in args[1..].chunks_exact(2).enumerate() {
        let [SlotValue::Source(InputValue::Reference(reference)),SlotValue::Value(value)] = pair else { return Err(fail(at,FailureKind::WrongOperation)); };
        let child_node = &plan.nodes[node.dependencies[index*2+2].node.0];
        if !matches!(child_node.operation,Operation::Value { source: child,.. } if child == *reference) {
            return Err(fail(at,FailureKind::IncompleteBindings));
        }
        let offset = nodes.len();
        if value.root >= value.nodes.len() { return Err(fail(at,FailureKind::IncompleteBindings)); }
        for child in &value.nodes { nodes.push(shifted(child,offset,at)?); }
        roots.push(value.root.checked_add(offset).ok_or_else(|| fail(at,FailureKind::IndexOverflow))?);
    }
    let tail = match atom {
        InputValue::Constant(ConstantSnapshot::U128 { n,.. }) | InputValue::Data(DataSnapshot::U128 { n }) => ValueNode::UInt(*n),
        InputValue::Constant(ConstantSnapshot::Bool { b,.. }) | InputValue::Data(DataSnapshot::Bool { b }) => ValueNode::Bool(*b),
        InputValue::Constant(ConstantSnapshot::Nil { .. }) | InputValue::Data(DataSnapshot::Nil) => ValueNode::Nil,
        InputValue::Constant(ConstantSnapshot::U128OutOfRange { .. }) => ValueNode::U128OutOfRange,
        InputValue::Constant(ConstantSnapshot::Cons { .. }) | InputValue::Data(DataSnapshot::Cons { .. }) => ValueNode::Cons { head:roots[0],tail:roots[1] },
        InputValue::Constant(ConstantSnapshot::Leaf { .. }) | InputValue::Data(DataSnapshot::Leaf { .. }) => ValueNode::Leaf { value:roots[0] },
        InputValue::Constant(ConstantSnapshot::Branch { .. }) | InputValue::Data(DataSnapshot::Branch { .. }) => ValueNode::Branch { left:roots[0],right:roots[1] },
        _ => return Err(fail(at,FailureKind::WrongOperation)),
    };
    let sort = |index:usize| match nodes.get(index) {
        Some(ValueNode::UInt(_)) => Some(SortTag::UInt),Some(ValueNode::Bool(_)) => Some(SortTag::Bool),
        Some(ValueNode::Nil | ValueNode::Cons { .. }) => Some(SortTag::List),
        Some(ValueNode::Leaf { .. } | ValueNode::Branch { .. }) => Some(SortTag::Tree), _ => None,
    };
    let required = match tail {
        ValueNode::Cons { head,tail } => vec![(head,SortTag::UInt),(tail,SortTag::List)],
        ValueNode::Leaf { value } => vec![(value,SortTag::UInt)],
        ValueNode::Branch { left,right } => vec![(left,SortTag::Tree),(right,SortTag::Tree)],
        _ => vec![],
    };
    for (index,expected) in required {
        let actual = sort(index);
        if actual != Some(expected) { return Err(fail(at,FailureKind::WrongValueSort { expected,actual })); }
    }
    let root = nodes.len(); nodes.push(tail);
    Ok(DecodedValue { nodes,root })
}

/// Kahn's algorithm, choosing the smallest ready ID, retains duplicate edges.
/// This order is independent of hash-map iteration or previous evaluations.
pub(super) fn order(nodes: &[Node]) -> Result<Vec<NodeId>, WitnessFailure> {
    let at = Location::new("plan", None, "dependencies");
    let mut counts = vec![0usize; nodes.len()];
    let mut uses = vec![Vec::new(); nodes.len()];
    for (index, node) in nodes.iter().enumerate() {
        if node.id != NodeId(index) { return Err(fail(node.location, FailureKind::InvalidNodeId)); }
        for dep in &node.dependencies {
            let source = nodes.get(dep.node.0).ok_or_else(|| fail(node.location,
                FailureKind::MissingDependency { dependency: dep.node }))?;
            if source.output_type != dep.value_type {
                return Err(fail(node.location, FailureKind::WrongType {
                    expected: dep.value_type, actual: source.output_type,
                }));
            }
            counts[index] = counts[index].checked_add(1).ok_or_else(|| fail(at, FailureKind::IndexOverflow))?;
            uses[dep.node.0].push(NodeId(index));
        }
    }
    let mut ready = BTreeSet::new();
    for (index, count) in counts.iter().enumerate() { if *count == 0 { ready.insert(NodeId(index)); } }
    let mut result = Vec::new();
    while let Some(id) = ready.iter().next().copied() {
        ready.remove(&id);
        result.push(id);
        for next in &uses[id.0] {
            counts[next.0] -= 1;
            if counts[next.0] == 0 { ready.insert(*next); }
        }
    }
    if result.len() != nodes.len() { return Err(fail(at, FailureKind::CyclicDependencies)); }
    Ok(result)
}

pub(super) fn definitions(nodes: &[Node], order: &[NodeId], live_outs: &[NodeId], lifetime: LifetimeUnit)
    -> Result<Vec<Definition>, WitnessFailure>
{
    let mut positions = vec![0usize; nodes.len()];
    for (position, id) in order.iter().enumerate() {
        let item = positions.get_mut(id.0).ok_or_else(|| fail(Location::new("plan", None, "order"), FailureKind::InvalidNodeId))?;
        *item = position;
    }
    let mut result: Vec<_> = nodes.iter().map(|node| Definition {
        value: node.id, producer: node.id, consumers: Vec::new(),
        last_use_order: positions[node.id.0], lifetime, live_out: live_outs.contains(&node.id),
    }).collect();
    for id in order {
        for dep in &nodes[id.0].dependencies {
            result[dep.node.0].consumers.push(*id);
            result[dep.node.0].last_use_order = result[dep.node.0].last_use_order.max(positions[id.0]);
        }
    }
    for id in live_outs {
        let value = result.get_mut(id.0).ok_or_else(|| fail(Location::new("plan", None, "live_outs"), FailureKind::InvalidNodeId))?;
        // Boundary live-outs remain available through final publication.
        value.last_use_order = order.len();
    }
    Ok(result)
}

fn node_failure(node: &Node, kind: FailureKind) -> WitnessFailure {
    WitnessFailure { location: node.location, node: Some(node.id), kind }
}
fn field_values(node: &Node, fields: &[FieldBinding], args: &[&SlotValue])
    -> Result<Vec<NamedValue>, WitnessFailure>
{
    if node.dependencies.len() != fields.len() + 1 || args.len() != fields.len() + 1 {
        return Err(node_failure(node, FailureKind::IncompleteBindings));
    }
    let mut result = Vec::new();
    for (offset, field) in fields.iter().enumerate() {
        if node.dependencies[offset + 1] != field.dependency {
            return Err(node_failure(node, FailureKind::IncompleteBindings));
        }
        result.push(NamedValue { field: field.field, value: (*args[offset + 1]).clone() });
    }
    Ok(result)
}

fn operation(plan: &WitnessPlan, input: &HostSnapshot, node: &Node, args: &[&SlotValue], diagnostics: &mut HostDiagnostics)
    -> Result<SlotValue, WitnessFailure>
{
    let wrong = || node_failure(node, FailureKind::WrongOperation);
    Ok(match &node.operation {
        Operation::Read(id) => {
            if !args.is_empty() { return Err(wrong()); }
            let binding = plan.inputs.get(id.0).ok_or_else(|| node_failure(node, FailureKind::MissingInput))?;
            if binding.id != *id { return Err(node_failure(node, FailureKind::IncompleteBindings)); }
            let value = read(input, binding.path, binding.location)?;
            if value.source_type() != binding.value_type {
                return Err(node_failure(node, FailureKind::WrongType {
                    expected: ValueType::Source(binding.value_type), actual: ValueType::Source(value.source_type()),
                }));
            }
            if value != binding.immutable_value { return Err(node_failure(node, FailureKind::InputChanged)); }
            diagnostics.input_reads += 1;
            SlotValue::Source(value)
        }
        Operation::Bind => {
            if args.len() != 2 { return Err(wrong()); }
            let SlotValue::Source(_) = args[0] else { return Err(wrong()); };
            (*args[1]).clone()
        }
        Operation::Value { source, arity } => {
            let value = decode::evaluate_value(plan, node, *source, *arity, args)?;
            diagnostics.value_constructions += 1;
            let copied = value.nodes.len().checked_sub(1).ok_or_else(wrong)?;
            diagnostics.copied_value_nodes = diagnostics.copied_value_nodes.checked_add(copied)
                .ok_or_else(|| node_failure(node, FailureKind::IndexOverflow))?;
            SlotValue::Value(value)
        }
        Operation::EmptyEnvironment => {
            if !args.is_empty() { return Err(wrong()); }
            SlotValue::Environment(Vec::new())
        }
        Operation::Environment => {
            let [SlotValue::Source(InputValue::Environment(cell)), SlotValue::Source(InputValue::Name(name)), SlotValue::Value(value), SlotValue::Environment(previous)] = args else {
                return Err(wrong());
            };
            let mut entries = Vec::new();
            entries.push(EnvironmentEntry { binder: cell.binder, name: name.clone(), value: value.clone() });
            entries.extend(previous.iter().cloned());
            diagnostics.environment_constructions += 1;
            diagnostics.copied_environment_entries = diagnostics.copied_environment_entries.checked_add(previous.len())
                .ok_or_else(|| node_failure(node, FailureKind::IndexOverflow))?;
            SlotValue::Environment(entries)
        }
        Operation::Sequence { kind, item_type } => {
            for value in args { if value.value_type() != *item_type { return Err(wrong()); } }
            SlotValue::Sequence { kind: *kind, items: args.iter().map(|v| (*v).clone()).collect() }
        }
        Operation::Payload { kind, fields } => {
            let Some(SlotValue::Source(raw)) = args.first().copied() else { return Err(wrong()); };
            let expected = match kind {
                PayloadKind::Frame => SourceType::Frame, PayloadKind::Control => SourceType::Control,
                PayloadKind::Helper => SourceType::Helper, PayloadKind::Task => SourceType::Task,
                PayloadKind::Result => SourceType::Result, PayloadKind::Outcome => SourceType::Outcome,
            };
            if raw.source_type() != expected { return Err(wrong()); }
            diagnostics.payload_constructions += 1;
            SlotValue::Payload(DecodedPayload { kind: *kind, raw: raw.clone(), fields: field_values(node, fields, args)? })
        }
        Operation::Snapshot { fields } => {
            let Some(SlotValue::Source(InputValue::Metadata(metadata))) = args.first().copied() else { return Err(wrong()); };
            if fields.iter().map(|f| f.field).collect::<Vec<_>>() != OUTPUT_FIELDS {
                return Err(node_failure(node, FailureKind::IncompleteOutputs));
            }
            SlotValue::Snapshot(DecodedSnapshot { metadata: metadata.clone(), fields: field_values(node, fields, args)? })
        }
    })
}

pub(super) fn evaluate(plan: &WitnessPlan, input: &HostSnapshot)
    -> Result<WitnessEvaluation, WitnessFailure>
{
    let at = Location::new("plan", None, "structure");
    if plan.format != "sherlock-host-witness-plan/1" { return Err(fail(at, FailureKind::WrongOperation)); }
    let schedule = order(&plan.nodes)?;
    if schedule != plan.stable_topological_order { return Err(fail(at, FailureKind::UnstableOrder)); }
    if plan.definitions != definitions(&plan.nodes, &schedule, &plan.live_outs, plan.lifetime)? {
        return Err(fail(at, FailureKind::IncompleteBindings));
    }
    let output = plan.nodes.get(plan.output.0).ok_or_else(|| fail(at, FailureKind::IncompleteOutputs))?;
    let Operation::Snapshot { fields } = &output.operation else { return Err(fail(at, FailureKind::IncompleteOutputs)); };
    let mut expected_live: Vec<_> = fields.iter().map(|f| f.dependency.node).collect();
    expected_live.push(plan.output);
    if plan.live_outs != expected_live { return Err(fail(at, FailureKind::IncompleteOutputs)); }
    // Input declarations must be indexed once and have exactly one Read producer.
    let mut reads = vec![0usize; plan.inputs.len()];
    for node in &plan.nodes {
        if let Operation::Read(id) = node.operation {
            let count = reads.get_mut(id.0).ok_or_else(|| node_failure(node, FailureKind::MissingInput))?;
            *count += 1;
        }
    }
    if reads.iter().any(|count| *count != 1) { return Err(fail(at, FailureKind::IncompleteBindings)); }
    let mut slots: Vec<Option<SlotValue>> = vec![None; plan.nodes.len()];
    let mut diagnostics = HostDiagnostics::default();
    for id in schedule {
        let node = &plan.nodes[id.0];
        let mut args = Vec::new();
        for dependency in &node.dependencies {
            let value = slots.get(dependency.node.0).and_then(Option::as_ref)
                .ok_or_else(|| node_failure(node, FailureKind::MissingDependency { dependency: dependency.node }))?;
            if value.value_type() != dependency.value_type {
                return Err(node_failure(node, FailureKind::WrongType { expected: dependency.value_type, actual: value.value_type() }));
            }
            args.push(value);
        }
        let result = operation(plan, input, node, &args, &mut diagnostics)?;
        if result.value_type() != node.output_type {
            return Err(node_failure(node, FailureKind::WrongType { expected: node.output_type, actual: result.value_type() }));
        }
        slots[id.0] = Some(result);
        diagnostics.evaluated_nodes += 1;
    }
    diagnostics.retained_slots = slots.len();
    let Some(SlotValue::Snapshot(decoded)) = slots.get_mut(plan.output.0).and_then(Option::take) else {
        return Err(fail(at, FailureKind::IncompleteOutputs));
    };
    Ok(WitnessEvaluation { decoded, diagnostics })
}
