/// `a + bv + carry_in = low + 2^32 * carry_out`.
pub fn add_carry_32(a: u32, bv: u32, carry_in: CarryBit) -> AddCarry32 {
    // At most 2^33-1, which fits u64.
    let total = u64::from(a) + u64::from(bv) + u64::from(carry_in.0);
    AddCarry32 { low: total as u32, carry_out: CarryBit((total >> 32) != 0) }
}

/// `a - bv - borrow_in = low - 2^32 * borrow_out` over mathematical integers.
pub fn sub_borrow_32(a: u32, bv: u32, borrow_in: BorrowBit) -> SubBorrow32 {
    let subtrahend = u64::from(bv) + u64::from(borrow_in.0);
    let borrow_out = u64::from(a) < subtrahend;
    // The chosen borrow makes this exact difference lie in the word range.
    let low = (u64::from(a) + (u64::from(borrow_out) << 32) - subtrahend) as u32;
    SubBorrow32 { low, borrow_out: BorrowBit(borrow_out) }
}

/// The complete two-word product, with named little-endian halves.
pub fn mul_wide_32(a: u32, bv: u32) -> MulWide32 {
    let product = u64::from(a) * u64::from(bv);
    MulWide32 { low: product as u32, high: (product >> 32) as u32 }
}

/// Multiply, add one stored word and one whole-word incoming carry.
/// The maximum total is `(2^32-1)^2 + 2*(2^32-1) = 2^64-1`.
pub fn mul_accumulate_32(
    a: u32, bv: u32, addend: u32, carry_in: MulCarry32,
) -> MulAccumulate32 {
    let product = mul_wide_32(a, bv);
    let with_addend = add_carry_32(product.low, addend, CarryBit(false));
    let with_carry = add_carry_32(with_addend.low, carry_in.0, CarryBit(false));
    // The bound above guarantees this exact high sum fits a whole word.
    let high = u64::from(product.high)
        + u64::from(with_addend.carry_out.0)
        + u64::from(with_carry.carry_out.0);
    MulAccumulate32 { low: with_carry.low, carry_out: MulCarry32(high as u32) }
}

fn split_words_32(value: u128) -> [u32; 4] {
    let mut limbs = [0; 4];
    for (i, limb) in limbs.iter_mut().enumerate() {
        *limb = (value >> (32 * i)) as u32;
    }
    limbs
}

fn join_words_32(limbs: [u32; 4]) -> u128 {
    let mut value = 0u128;
    for (i, limb) in limbs.iter().enumerate() {
        value |= u128::from(*limb) << (32 * i);
    }
    value
}

/// Retain the full 129-bit sum, then select the checked UInt128 outcome.
pub fn checked_u128_add_32(a: u128, bv: u128) -> CheckedUInt128Add32 {
    let left = split_words_32(a);
    let right = split_words_32(bv);
    let mut low = [0; 4];
    let mut carry_out = CarryBit(false);
    for i in 0..4 {
        let word = add_carry_32(left[i], right[i], carry_out);
        low[i] = word.low;
        carry_out = word.carry_out;
    }
    let full = FullAdd32 { low, carry_out };
    let checked = if full.carry_out.0 { Err(10) } else { Ok(join_words_32(full.low)) };
    CheckedUInt128Add32 { full, checked }
}

/// Retain all 256 product bits before inspecting the high UInt128 half.
pub fn checked_u128_mul_32(a: u128, bv: u128) -> CheckedUInt128Mul32 {
    let left = split_words_32(a);
    let right = split_words_32(bv);
    let mut product = [0; 8];
    // Fixed schoolbook rows. Before row i, product[i+4] is untouched and zero;
    // previous rows end at i+4-1. Its final carry is therefore assigned exactly.
    for i in 0..4 {
        let mut carry = MulCarry32(0);
        for j in 0..4 {
            let word = mul_accumulate_32(left[i], right[j], product[i + j], carry);
            product[i + j] = word.low;
            carry = word.carry_out;
        }
        product[i + 4] = carry.0;
    }
    let mut low = [0; 4];
    let mut high = [0; 4];
    low.copy_from_slice(&product[..4]);

        Err(10)

## Layout and component meaning

Layout identity: `sherlock-typed-words-le/1`. For word width `b` equal to 32 or 64,
let `B = 2^b` and `N = 128/b`. A UInt128 uses four u32 or two u64 limbs.
Limb zero is least significant: `join(words) = sum(words[i] * B^i)`.
A full product's `low[i]` weighs `B^i`; `high[i]` weighs `2^128 * B^i`.

| Component | Exact mathematical equation |
| --- | --- |
| `add_carry_32/64` | `a + bv + carry_in = low + B * carry_out` |
| `sub_borrow_32/64` | `a - bv - borrow_in = low - B * borrow_out` |
| `mul_wide_32/64` | `a * bv = low + B * high` |
| `mul_accumulate_32/64` | `a * bv + addend + carry_in = low + B * carry_out` |

`CarryBit(bool)` and `BorrowBit(bool)` are distinct bit types. Multiply-accumulate
uses a whole word through `MulCarry32(u32)` or `MulCarry64(u64)`. For example, a
whole-word incoming carry of 3 is representable by `MulCarry64(3)`; it is not a
bit. Component outputs have named fields, so high/low ordering is explicit.

## Full intermediate and checked result

Each checked entrypoint returns both `full` and `checked`. Addition retains
`join(full.low) + 2^128 * full.carry_out`; subtraction retains the signed integer
`join(full.low) - 2^128 * full.borrow_out`. Multiplication retains every one of
its 256 product bits as `join(full.low) + 2^128 * join(full.high)` before checked
selection. A word component's carry can be true while the complete UInt128
calculation still fits.

`checked: Result<u128, u8>` is `Ok(value)` when the mathematical result lies in
`[0, 2^128)`. The inspected source selects legacy `Err(10)` for final addition
carry, final subtraction borrow, or a nonzero high128 product. The retained full
intermediate remains in the return object. Those whole-UInt128 error branches
were not executed by the normal cohort described below; error10 selection is a
source observation here.

The independently specified R197 original examples give these successful values

| D27 | `(2^64 + 9) - 7` | `18446744073709551618` |
| D28 | `(2^48 + 1) * 3` | `844424930131971` |
