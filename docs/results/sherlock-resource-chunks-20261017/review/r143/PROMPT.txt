/// Host-only K-A ReferenceSlice observation. It carries only fields available
/// in the existing K-A machine. Source suspension and proposed B identities,
/// layouts and helpers remain separate from this host API.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ReferenceSlice {
    pub profile_id: &'static str,
    pub start: Snapshot,
    pub records: Vec<StepRecord>,
    /// Length records.len()+1, starts zero, accumulates each actual record.charge.
    pub prefix_work: Vec<u128>,
    pub boundaries: Vec<SliceBoundary>,
    pub end: Snapshot,
    pub stop: SliceStop,
}

/// Retain at most m reference events, including free publication as an event.
/// Normal comparison budgets are 1/2/4/8. Other host usize budgets merely choose
/// grouping length; no native width or resource policy is selected here.
///
/// Zero is an explicit observation-only cut: pending input yields identical
/// start/end, no records, prefix_work=[0], and a TechnicalCut at position zero.
/// Already published outcomes, including with m=0, yield an empty terminal
/// slice with the unchanged Outcome and a Terminal boundary at position zero.
///
/// Each next step is attempted in order on the live ordinary Machine, with its
/// own fuel/cap/publication behavior. There is no stepping ahead or replay.
/// Reaching a pending terminal control at the event budget leaves publication
/// for the next group; when the budget includes publication, its zero charge
/// is retained and the group stops immediately at that first published outcome.
pub fn advance_m(machine: &mut Machine, m: usize) -> ReferenceSlice {
    let start = machine.snapshot();
    let mut records = Vec::new();
    let mut prefix_work = vec![0];

    if start.outcome == Outcome::Pending {
        for _ in 0..m {
            let Some(record) = machine.step() else {
                break;
            };
            let cumulative = prefix_work.last().copied().expect("initial zero work")
                + u128::from(record.charge);
            let terminal = record.after.outcome != Outcome::Pending;
            prefix_work.push(cumulative);
            records.push(record);
            if terminal {
                break;
            }
        }
    }

    let end = machine.snapshot();
    let (stop, kind) = if end.outcome == Outcome::Pending {
        (SliceStop::TechnicalCut, SliceBoundaryKind::TechnicalCut)
    } else {
        (SliceStop::Outcome(end.outcome.clone()), SliceBoundaryKind::Terminal)
    };
    let boundary = SliceBoundary { position: records.len(), kind };
    ReferenceSlice {
        profile_id: PROFILE,
        start,
        records,
        prefix_work,
        boundaries: vec![boundary],
        end,
        stop,
    }
}

/// Data-accounting diagnostics on supplied slices, with no rule evaluation.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ReferenceReportError {
    NoChunks,
    Profile { chunk: usize },
    Continuity { chunk: usize, difference: super::compare::ContinuityDifference },
    BetweenChunks { previous: usize, next: usize, difference: super::compare::ContinuityDifference },
    PrefixWork { chunk: usize, position: usize },
    ChargeOverflow { chunk: usize, record: usize },
    EventCountOverflow { chunk: usize },
}

impl ReferenceSlice {
    /// The same cumulative report used for multiple technical chunks.
    /// A pending or empty present slice remains a valid reportable prefix.
    pub fn resource_report(&self) -> Result<
        ReferenceResourceReport, ReferenceReportError>
    {
        cumulative_resource_report(&[self])
    }
}

/// Report one ordered K-A observation from its present technical chunks.
/// Reuses the complete typed continuity checker and exact record charges;
/// grouping boundaries contribute no event, work, helper or financial cost.
/// Prefix sums are checked independently of the caller-supplied prefix_work.
/// No transition, rule charge, source authenticity or producer is validated.
/// Reporting inherits A's unbounded structural traversal; it creates no public
/// untrusted-input resource policy and does not claim equal-fuel cross profiles.
pub fn cumulative_resource_report(chunks: &[&ReferenceSlice]) -> Result<
    ReferenceResourceReport, ReferenceReportError>
{
    use super::compare::{first_continuity_difference, TraceIdentity, TraceView};
    let Some(first) = chunks.first() else { return Err(ReferenceReportError::NoChunks); };
    let identity = TraceIdentity { profile: PROFILE, source_sha256: K_SOURCE_SHA256 };
    let mut total_work = 0u128;
    let mut total_events = 0u128;
    let mut depth = first.start.frames.len();
    for (chunk, slice) in chunks.iter().enumerate() {
        if slice.profile_id != PROFILE {
            return Err(ReferenceReportError::Profile { chunk });
        }
        let trace = TraceView { identity, initial: &slice.start, records: &slice.records };
        if let Some(difference) = first_continuity_difference(trace, &slice.end) {
            return Err(ReferenceReportError::Continuity { chunk, difference });
        }
        if chunk > 0 {
            let edge = TraceView { identity, initial: &chunks[chunk - 1].end, records: &[] };
            if let Some(difference) = first_continuity_difference(edge, &slice.start) {
                return Err(ReferenceReportError::BetweenChunks {
                    previous: chunk - 1, next: chunk, difference,
                });
            }
        }
        if slice.prefix_work.len() != slice.records.len() + 1 {
            return Err(ReferenceReportError::PrefixWork { chunk, position: slice.prefix_work.len() });
        }
        if slice.prefix_work[0] != 0 {
            return Err(ReferenceReportError::PrefixWork { chunk, position: 0 });
        }
        depth = depth.max(slice.start.frames.len()).max(slice.end.frames.len());
        let mut prefix = 0u128;
        for (record, event) in slice.records.iter().enumerate() {
            prefix = prefix.checked_add(u128::from(event.charge))
                .ok_or(ReferenceReportError::ChargeOverflow { chunk, record })?;
            if slice.prefix_work[record + 1] != prefix {
                return Err(ReferenceReportError::PrefixWork { chunk, position: record + 1 });
            }
            total_work = total_work.checked_add(u128::from(event.charge))
                .ok_or(ReferenceReportError::ChargeOverflow { chunk, record })?;
            depth = depth.max(event.before.frames.len()).max(event.after.frames.len());
        }
        total_events = total_events.checked_add(slice.records.len() as u128)
            .ok_or(ReferenceReportError::EventCountOverflow { chunk })?;
    }
    Ok(ReferenceResourceReport::observed(
        total_work, total_events, depth))
}

//! Actual forwarded-event diagnostics, SOURCE_ONLY / UNCOMPILED / UNRUN.
//!
//! This observes the existing source caller; it never evaluates, replays,
//! validates, charges, retries or mutates its engine. Sums are not RunSummary
//! assertions. Maximum depth covers only observed event before/after fields.
use std::collections::BTreeMap;
use serde::Serialize;

use super::core_action_consumer::{run_scalar_action, ConsumerConfig, ConsumerError, CoreActionRun};
use super::resource_report::Counter;
use super::shared::StepEventC;

pub const FORMAT: &str = "moriarty-sherlock-b-event-accounting/1";

/// Overflow retains the last exact prefix BEFORE the failed addition. It is
/// explicitly not the full-stream total, a saturated total or a refusal.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
#[serde(tag = "status", rename_all = "kebab-case")]
pub enum CheckedTotal {
    Exact { value: Counter },
    Overflow { exact_prefix: Counter },
}
impl CheckedTotal {
    fn zero() -> Self { Self::Exact { value: Counter(0) } }
    fn add(&mut self, delta: u128) {
        if let Self::Exact { value } = *self {
            *self = match value.0.checked_add(delta) {
                Some(sum) => Self::Exact { value: Counter(sum) },
                None => Self::Overflow { exact_prefix: value },
            };
        }
    }
    fn is_exact(self) -> bool { matches!(self, Self::Exact { .. }) }
}

/// Rust caller status only: ReturnedOk is not application acceptance.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
#[serde(rename_all = "kebab-case")]
pub enum PrefixEnd { Open, ReturnedOk, ReturnedErr }
#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
#[serde(rename_all = "kebab-case")]
pub enum ArithmeticStatus { Exact, IncompleteOverflow }

#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
pub struct EventAccountingReport {
    pub format: &'static str,
    pub prefix_end: PrefixEnd,
    pub arithmetic_status: ArithmeticStatus,
    pub event_count: CheckedTotal,
    pub work_delta_sum: CheckedTotal,
    pub helper_work_delta_sum: CheckedTotal,
    pub helper_visits_delta_sum: CheckedTotal,
    pub reference_cursor_delta_sum: CheckedTotal,
    pub maximum_observed_logical_depth: Option<usize>,
}

/// Fixed-size aggregate state. There is no event vector, rule-count map,
/// sampling, policy extent or resource input. Empty-prefix zero counts are
/// exact observations; empty-prefix depth remains None rather than inferred0.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct EventAccounting {
    event_count: CheckedTotal,
    work_delta_sum: CheckedTotal,
    helper_work_delta_sum: CheckedTotal,
    helper_visits_delta_sum: CheckedTotal,
    reference_cursor_delta_sum: CheckedTotal,
    maximum_observed_logical_depth: Option<usize>,
}
impl EventAccounting {
    pub fn new() -> Self {
        Self { event_count: CheckedTotal::zero(), work_delta_sum: CheckedTotal::zero(),
            helper_work_delta_sum: CheckedTotal::zero(), helper_visits_delta_sum: CheckedTotal::zero(),
            reference_cursor_delta_sum: CheckedTotal::zero(), maximum_observed_logical_depth: None }
    }

    /// Account every event before passing it to the caller's sink. Arithmetic
    /// overflow is diagnostic state only; every other column/depth continues.
    pub fn observe(&mut self, event: &StepEventC) {
        self.event_count.add(1);
        self.work_delta_sum.add(u128::from(event.work_delta));
        self.helper_work_delta_sum.add(u128::from(event.helper_work_delta));
        self.helper_visits_delta_sum.add(u128::from(event.helper_visits_delta));
        self.reference_cursor_delta_sum.add(u128::from(event.reference_cursor_delta));
        let event_max = event.logical_depth_before.max(event.logical_depth_after);
        self.maximum_observed_logical_depth = Some(match self.maximum_observed_logical_depth {
            Some(previous) => previous.max(event_max), None => event_max,
        });
    }

    /// A snapshot of this observed prefix, not an owner invariant/validation.
    /// The wrapper derives prefix_end from the unchanged actual Result.
    pub fn snapshot(&self, prefix_end: PrefixEnd) -> EventAccountingReport {
        let exact = [self.event_count, self.work_delta_sum, self.helper_work_delta_sum,
            self.helper_visits_delta_sum, self.reference_cursor_delta_sum]
            .into_iter().all(CheckedTotal::is_exact);
        EventAccountingReport { format: FORMAT, prefix_end,
            arithmetic_status: if exact { ArithmeticStatus::Exact } else { ArithmeticStatus::IncompleteOverflow },
            event_count: self.event_count, work_delta_sum: self.work_delta_sum,
            helper_work_delta_sum: self.helper_work_delta_sum,
            helper_visits_delta_sum: self.helper_visits_delta_sum,
            reference_cursor_delta_sum: self.reference_cursor_delta_sum,
            maximum_observed_logical_depth: self.maximum_observed_logical_depth }
    }
}

/// Original source result and separate observed-prefix accounting. On Err,
/// the original ConsumerError (including its own prefix metadata) is retained
/// untouched. On Ok, the actual RunSummary remains in run_result; no sum is
/// compared to, substituted for or asserted against that summary here.
pub struct AccountedScalarRun {
    pub run_result: Result<CoreActionRun, ConsumerError>,
    pub event_accounting: EventAccountingReport,
}

/// Additive wrapper: actual existing caller once, same sink order/delivery.
/// Neither sink panics nor engine panics are caught or translated; as before,
/// unwinding does not promise a returned report. Wrapper/accounting costs are
/// not owner work or measured physical/circuit/financial/preparation costs.
pub fn run_scalar_action_with_event_accounting(
    source: &crate::Program,
    action_index: usize,
    inputs: &BTreeMap<String, crate::Value>,
    config: &ConsumerConfig,
    sink: &mut dyn FnMut(&StepEventC),
) -> AccountedScalarRun {
    let (run_result, event_accounting) = collect_forwarded(sink, |forward| {
        run_scalar_action(source, action_index, inputs, config, forward)
    });
    AccountedScalarRun { run_result, event_accounting }
}

// One small forwarding function shared by the actual wrapper and literal-event
// unit tests. It is private: no alternative executor is registered or exposed.
fn collect_forwarded<T, E>(sink: &mut dyn FnMut(&StepEventC),
    operation: impl FnOnce(&mut dyn FnMut(&StepEventC)) -> Result<T, E>)
    -> (Result<T, E>, EventAccountingReport)
{
    let mut accounting = EventAccounting::new();
    let result = {
        let mut forward = |event: &StepEventC| {
            accounting.observe(event);
            sink(event);
        };
        operation(&mut forward)
    };
    let end = if result.is_ok() { PrefixEnd::ReturnedOk } else { PrefixEnd::ReturnedErr };
    (result, accounting.snapshot(end))
}

        .map_err(ConsumerError::SourceProjection)?;
    let (body_nodes, _) = check_generated(&mapped, &preflight, &config.host)?;

    let arguments: Result<Vec<ExprRef>, ConsumerError> = mapped.report.inputs.iter().map(|site| {
        // Preflight and check_generated already bind these same immutable names
        // to the declared scalar inputs. Keep the typed missing-input error on
        // this lookup too; do not fabricate a literal if the binding is absent.
        let value = inputs.get(&site.name).ok_or_else(|| {
            ConsumerError::HostGuard(HostGuardError::MissingInput { name: site.name.clone() })
        })?;
        let value = match value {
            crate::Value::UInt(value) => ScalarValue::UInt(value.0),
            crate::Value::Bool(value) => ScalarValue::Bool(*value),
        };
        Ok(Arc::new(Expr::Literal(value)))
    }).collect();
    let entry = Arc::new(Expr::Call(mapped.function.name.clone(), arguments?));
    check_combined_generated_ast_nodes(body_nodes, &entry, &config.host)?;

    let preparation = mapped.report;
    let canonical = ScalarProgram { functions: vec![mapped.function], entry };
    let owner_source = source_input_bridge::from_program(&canonical);
    let engine = host_owner_engine();
    let admitted = engine.admit(&owner_source, &config.bounds, &config.layout)
        .map_err(ConsumerError::Admit)?;
    let image = engine.admit_run(&admitted, &config.fuel, &config.stack, &FinanceInput::Absent)
        .map_err(ConsumerError::AdmitRun)?;

    Ok(CoreActionPrepared { preparation, admitted, image })
}

/// Project and run one currently supported scalar action through the fixed B owner.
///
/// `inputs` is the same map type as `Scenario.inputs`. Only that explicit map
/// is accepted: state, balances, liabilities and source effects are not a
/// financial input to this pure projection. Names are resolved in the real
/// adapter report's declaration order, independently of BTreeMap key order.
///
/// Preparation/guard failures and the owner's Tooling/HostViolation values
/// retain the staged security-owned error contract. If `run` fails, events
/// already forwarded to `sink` form a partial prefix and no completed result
/// is returned. The sink receives each real full StepEventC exactly once;
/// this consumer stores no event vector. Any sink retention is caller-owned.
/// Execute the existing fixed owner on the pair produced by shared preparation.
pub fn run_scalar_action(
    source: &crate::Program,
    action_index: usize,
    inputs: &BTreeMap<String, crate::Value>,
    config: &ConsumerConfig,
    sink: &mut dyn FnMut(&StepEventC),
) -> Result<CoreActionRun, ConsumerError> {
    let CoreActionPrepared { preparation, admitted, mut image } =
        prepare_scalar_action_host(source, action_index, inputs, config)?;
    let engine = host_owner_engine();
    let mut events = EventForwarding::default();
    let result = {
        let mut forwarding_sink = |event: &StepEventC| {
            events.note();
            sink(event);
        };
        engine.run(&admitted, &mut image, &mut forwarding_sink)
    };
    let summary = result.map_err(|violation| ConsumerError::Run {
        violation,
        events_forwarded: events.forwarded,
    })?;
    Ok(CoreActionRun {
        preparation,
        completed: CompletedRun {
            admitted,
            image,
            summary,
            events_forwarded: events.forwarded,
        },
    })
}

// Borrowed raw-row adapter for the expanded-13 leaves (M1). This module
// re-authors the leaves as pub fns; it assumes NO old private visibility.
pub mod pure_leaves {
    use super::{LogicalKind, RawConstRow, RawDataRow, SortTag};
    #[derive(Clone, Copy)]
    pub enum BorrowedValueRow<'a> {
        Const(&'a super::RawConstRow), Data(&'a super::RawDataRow) }
    pub fn borrowed_kind(r: BorrowedValueRow<'_>) -> LogicalKind {
        match r {
            BorrowedValueRow::Const(c) => match c {
                RawConstRow::U128 { .. } => LogicalKind::UInt,
                RawConstRow::Bool { .. } => LogicalKind::Bool,
                RawConstRow::Nil { .. } => LogicalKind::Nil,
                RawConstRow::Cons { .. } => LogicalKind::Cons,
                RawConstRow::Leaf { .. } => LogicalKind::Leaf,
                RawConstRow::Branch { .. } => LogicalKind::Branch,
                RawConstRow::U128OutOfRange { .. } => LogicalKind::Marker,
            },
            BorrowedValueRow::Data(d) => match d {
                RawDataRow::U128 { .. } => LogicalKind::UInt,
                RawDataRow::Bool { .. } => LogicalKind::Bool,
                RawDataRow::Nil => LogicalKind::Nil,
                RawDataRow::Cons { .. } => LogicalKind::Cons,
                RawDataRow::Leaf { .. } => LogicalKind::Leaf,
                RawDataRow::Branch { .. } => LogicalKind::Branch,
            },
        }
    }
    pub fn borrowed_u128(r: BorrowedValueRow<'_>) -> Option<u128> {
        match r { BorrowedValueRow::Const(RawConstRow::U128 { n, .. }) => Some(*n),
            BorrowedValueRow::Data(RawDataRow::U128 { n }) => Some(*n),
            _ => None } }
    pub fn borrowed_bool(r: BorrowedValueRow<'_>) -> Option<bool> {
        match r { BorrowedValueRow::Const(RawConstRow::Bool { b, .. }) => Some(*b),
            BorrowedValueRow::Data(RawDataRow::Bool { b }) => Some(*b),
            _ => None } }
    pub fn shallow_differ(a: BorrowedValueRow<'_>, b: BorrowedValueRow<'_>) -> bool {
        borrowed_kind(a) != borrowed_kind(b)
            || borrowed_u128(a) != borrowed_u128(b)
            || borrowed_bool(a) != borrowed_bool(b) }
    pub fn arity_of(k: LogicalKind) -> Option<usize> {
        match k { LogicalKind::UInt | LogicalKind::Bool | LogicalKind::Nil => Some(0),
            LogicalKind::Cons | LogicalKind::Branch => Some(2),
            LogicalKind::Leaf => Some(1),
            LogicalKind::Marker | LogicalKind::ConstructorExtensionReserved => None } }
    pub fn sort_of_kind(k: LogicalKind) -> Option<SortTag> {
        match k { LogicalKind::UInt => Some(SortTag::UInt),
            LogicalKind::Bool => Some(SortTag::Bool),
            LogicalKind::Nil | LogicalKind::Cons => Some(SortTag::List),
            LogicalKind::Leaf | LogicalKind::Branch => Some(SortTag::Tree),
            LogicalKind::Marker | LogicalKind::ConstructorExtensionReserved => None } }
    // Marker/reserved arity or sort = None here; the OWNER maps None to
    // HostViolation{ArithmeticUndefined} / staged PrimError as the row requires.
    #[derive(Clone, Copy, PartialEq, Eq, Debug)]
    pub enum CheckedArith { Value { n: u128 }, Overflow }
    pub fn checked_add(a: u128, b: u128) -> CheckedArith { match a.checked_add(b) { Some(n) => CheckedArith::Value { n }, None => CheckedArith::Overflow } }
    pub fn checked_sub(a: u128, b: u128) -> CheckedArith { match a.checked_sub(b) { Some(n) => CheckedArith::Value { n }, None => CheckedArith::Overflow } }
    pub fn checked_mul(a: u128, b: u128) -> CheckedArith { match a.checked_mul(b) { Some(n) => CheckedArith::Value { n }, None => CheckedArith::Overflow } }
    // Owner maps CheckedArith::Overflow only to staged PrimError{code:10}; no None-to-false and no other mapping.
}

pub fn observe_bound_image<'a>(admitted: &'a Admitted, img: &'a MachineImage)
    -> Result<HostObservation<'a>, HostViolation> {
    // AB first: a mismatch returns the SI-01 HostViolation before any read below.
    let pre = PreImage::bind(admitted, img)?;
    // reg.segBase: only the owner helper over the bound pair; Err refuses the record.
    let seg_base = implied_seg_base(&pre)?;
    let PreImage { admitted, img } = pre;
    let body: &'a AdmittedBody = &admitted.body;
    let l = &body.layout;
    let widths = l.widths.as_ref().map(|w| WidthTableObservation {
        index_widths: w.index_widths.as_slice(), counter_widths: w.counter_widths.as_slice(),
        record_widths: w.record_widths.as_slice(), _sealed: () });
    let layout = LayoutObservation { layout_version: &l.layout_version,
        field_id: l.field_id.as_ref(), widths, scalar_layout_id: l.scalar_layout_id.as_ref(),
        bounds: &l.bounds, capacities: &l.capacities,
        boundary_capacities_present: l.boundary_capacities_present, _sealed: () };
    // img.admission is the same body as `admitted` (Arc::ptr_eq above); the
    // handle itself is a host address and is not observed.
    let image = ImageObservation { run_id: &*img.run_id, control: &img.control,
        control_origin: img.control_origin, env_root: img.env_root,
        stack_limit: img.stack_limit, outcome: &img.outcome,
        initial_fuel: img.initial_fuel, remaining_work: img.remaining_work,
        consumed_work: img.consumed_work, helper_work: img.helper_work,
        helper_visits: img.helper_visits, reference_cursor: img.reference_cursor,
        alloc: &img.alloc, active: &img.active,
        helper: img.helper.as_ref(), result: img.result.as_ref(), seg_base,
        data: img.data.as_slice(), data_fields: img.data_fields.as_slice(),
        env: img.env.as_slice(), frames: img.frames.as_slice(),
        tasks: img.tasks.as_slice(), value_slots: img.value_slots.as_slice(), _sealed: () };
    Ok(HostObservation { admission: AdmissionObservation { identity: &body.identity,
        bounds: &body.bounds, view: &body.view, layout, _sealed: () }, image, _sealed: () })
}

//! First-difference and continuity positions for complete typed A observations.
//!
//! Pairwise `first_difference` compares supplied observations. The separate
//! `first_continuity_difference` checks complete state edges and total charge.
//! Neither evaluates rules, authenticates source metadata, or validates proofs.
//! For pairwise comparison the caller supplies two present traces on the declared
//! `sherlock-k-a/1` common domain. An empty record slice is a present zero-step
//! observation; it is not an absent K, Quint, Lean, Clean, or handwritten layer.
//! Missing adapters and layers must be reported separately by their consumer.
//!
//! No K execution or cross-profile clock alignment is provided. Pairwise
//! comparison reports different profile/source declarations before states. Metadata
//! equality alone does not establish that a declaration is true. This API does
//! not implement Candidate B or claim other profiles have the same semantics.
//! Structural comparison inherits A's unbounded host time and recursive data
//! traversal; no new host-resource bound or untrusted consumer is introduced.

use super::reference_types::{Snapshot, StepRecord};

/// Declared semantic clock and pinned source identity, not producer identity.
/// For the frozen K A observations use `PROFILE` and `K_SOURCE_SHA256`.
/// This comparator neither hashes sources nor parses diagnostic display text.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct TraceIdentity<'a> {
    pub profile: &'a str,
    pub source_sha256: &'a str,
}

/// Borrowed complete observations, retaining every original typed value.
/// Both an initial state and a record slice are required; absence is not a match.
#[derive(Clone, Copy, Debug)]
pub struct TraceView<'a> {
    pub identity: TraceIdentity<'a>,
    pub initial: &'a Snapshot,
    pub records: &'a [StepRecord],
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum IdentityField {
    Profile,
    SourceSha256,
}

/// Complete snapshot fields, in deterministic comparison order.
/// Program equality includes entry, every function/parameter/body, and ordering;
/// frames include their complete saved environments and values, top first.
/// Depth is represented by the complete ordered frames, not a separate counter.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum SnapshotField {
    Program,
    Control,
    Environment,
    Frames,
    StackLimit,
    RemainingFuel,
    InitialFuel,
    Outcome,
}

/// Record fields, in deterministic comparison order. Before/after fields follow
/// the ordering of `SnapshotField`.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RecordField {
    RuleStartLine,
    Charge,
    Before(SnapshotField),
    After(SnapshotField),
}

/// Position of the earliest unequal observation. Indices are zero based.
/// Retrieve raw values through the original `TraceView` at this position; no
/// display string, serialized codec, copied state, or erased outcome is returned.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum TraceDifference {
    Identity(IdentityField),
    Initial(SnapshotField),
    Record {
        index: usize,
        field: RecordField,
    },
    /// The shorter slice ended after this many equal complete records.
    Length {
        common_records: usize,
        left_records: usize,
        right_records: usize,
    },
}

/// Compare identity (profile, source), initial snapshot, then common records in
/// ascending index order (rule, charge, before, after), then record lengths.
/// Snapshot field ordering is documented on `SnapshotField`. Equality is typed
/// structural equality, including through `Arc`, rather than pointer identity.
///
/// `None` means only that both supplied observations agree completely. It says
/// nothing about validity, source authenticity, missing layers, or proofs.
#[must_use]
pub fn first_difference(
    left: TraceView<'_>,
    right: TraceView<'_>,
) -> Option<TraceDifference> {
    if left.identity.profile != right.identity.profile {
        return Some(TraceDifference::Identity(IdentityField::Profile));
    }
    if left.identity.source_sha256 != right.identity.source_sha256 {
        return Some(TraceDifference::Identity(IdentityField::SourceSha256));
    }
    if let Some(field) = snapshot_difference(left.initial, right.initial) {
        return Some(TraceDifference::Initial(field));
    }
    for (index, (left_record, right_record)) in
        left.records.iter().zip(right.records).enumerate()
    {
        let field = if left_record.rule_start_line != right_record.rule_start_line {
            Some(RecordField::RuleStartLine)
        } else if left_record.charge != right_record.charge {
            Some(RecordField::Charge)
        } else if let Some(field) =
            snapshot_difference(&left_record.before, &right_record.before)
        {
            Some(RecordField::Before(field))
        } else {
            snapshot_difference(&left_record.after, &right_record.after)
                .map(RecordField::After)
        };
        if let Some(field) = field {
            return Some(TraceDifference::Record { index, field });
        }
    }
    if left.records.len() != right.records.len() {
        return Some(TraceDifference::Length {
            common_records: left.records.len().min(right.records.len()),
            left_records: left.records.len(),
            right_records: right.records.len(),
        });
    }
    None
}

fn snapshot_difference(left: &Snapshot, right: &Snapshot) -> Option<SnapshotField> {
    if left.program.as_ref() != right.program.as_ref() {
        Some(SnapshotField::Program)
    } else if left.control != right.control {
        Some(SnapshotField::Control)
    } else if left.environment != right.environment {
        Some(SnapshotField::Environment)
    } else if left.frames != right.frames {
        Some(SnapshotField::Frames)
    } else if left.stack_limit != right.stack_limit {
        Some(SnapshotField::StackLimit)
    } else if left.remaining_fuel != right.remaining_fuel {
        Some(SnapshotField::RemainingFuel)
    } else if left.initial_fuel != right.initial_fuel {
        Some(SnapshotField::InitialFuel)
    } else if left.outcome != right.outcome {
        Some(SnapshotField::Outcome)
    } else {
        None
    }
}

/// Complete-state continuity edge. Record indices are zero based.
/// Retrieve both raw states from the original trace and returned snapshot.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ContinuityBoundary {
    /// `trace.initial` versus `trace.records[0].before`.
    InitialToFirst,
    /// `records[previous].after` versus `records[next].before`.
    BetweenRecords { previous: usize, next: usize },
    /// `records[last].after` versus the independently supplied returned state.
    LastToReturned { last: usize },
    /// A present empty record slice: `trace.initial` versus returned state.
    EmptyToReturned,
}

/// First complete-state boundary or aggregate charge-accounting mismatch.
/// Raw observations remain caller-owned; no values are erased or normalized.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ContinuityDifference {
    State { boundary: ContinuityBoundary, field: SnapshotField },
    /// Returned remaining fuel exceeds initial remaining fuel.
    UndefinedFuelDelta,
    /// Checked addition of this record's charge would exceed u128.
    ChargeOverflow { index: usize },
    ChargeSumMismatch { recorded: u128, endpoint_delta: u128 },
}

/// Check initial/first, adjacent after/before edges in ascending record order,
/// then last/returned. Empty records compare initial/returned, without inferring
/// terminality. All eight complete fields use the existing `SnapshotField` order.
/// State differences precede aggregate accounting: checked endpoint subtraction,
/// checked charge summation, then exact total equality. No counters are clamped.
///
/// `None` means these supplied boundaries and aggregate counters agree only.
/// It does not validate a transition, rule line, individual rule charge, outcome,
/// profile declaration, source authenticity, missing layer, or proof. In
/// particular, a matching pending prefix is permitted. Structural comparison
/// retains the A profile's unbounded host traversal; no new resource policy or
/// public untrusted-input admission is provided.
#[must_use]
pub fn first_continuity_difference(
    trace: TraceView<'_>,
    returned: &Snapshot,
) -> Option<ContinuityDifference> {
    if let Some((first, rest)) = trace.records.split_first() {
        if let Some(field) = snapshot_difference(trace.initial, &first.before) {
            return Some(ContinuityDifference::State {
                boundary: ContinuityBoundary::InitialToFirst, field,
            });
        }
        let mut previous = first;
        for (index, next) in rest.iter().enumerate() {
            if let Some(field) = snapshot_difference(&previous.after, &next.before) {
                return Some(ContinuityDifference::State {
                    boundary: ContinuityBoundary::BetweenRecords {
                        previous: index, next: index + 1,
                    },
                    field,
                });
            }
            previous = next;
        }
        if let Some(field) = snapshot_difference(&previous.after, returned) {
            return Some(ContinuityDifference::State {
                boundary: ContinuityBoundary::LastToReturned {
                    last: trace.records.len() - 1,
                },
                field,
            });
        }
    } else if let Some(field) = snapshot_difference(trace.initial, returned) {
        return Some(ContinuityDifference::State {
            boundary: ContinuityBoundary::EmptyToReturned, field,
        });
    }

    let Some(endpoint_delta) = trace.initial.remaining_fuel.checked_sub(returned.remaining_fuel)
    else {
        return Some(ContinuityDifference::UndefinedFuelDelta);
    };
    let mut recorded = 0u128;
    for (index, record) in trace.records.iter().enumerate() {
        let Some(sum) = recorded.checked_add(u128::from(record.charge)) else {
            return Some(ContinuityDifference::ChargeOverflow { index });
        };
        recorded = sum;
    }
    if recorded != endpoint_delta {
        return Some(ContinuityDifference::ChargeSumMismatch { recorded, endpoint_delta });
    }
    None
}

/// Structured original diagnostic payloads, never a manufactured owner outcome.
#[derive(Serialize)]
pub struct CompileFailure {
    pub stage: &'static str,
    pub actual_error: Value,
    pub evaluation_performed: bool,
    pub events_forwarded: String,
    pub completed: bool,
}
impl CompileFailure {
    pub fn input(stage: &'static str, error: impl Into<String>) -> Self {
        Self { stage, actual_error: json!({"message":error.into()}),
            evaluation_performed:false, events_forwarded:"0".into(), completed:false }
    }
    fn core(stage: &'static str, error: crate::Error) -> Self {
        Self { stage, actual_error:json!(error), evaluation_performed:false,
            events_forwarded:"0".into(), completed:false }
    }
    fn consumer(error: ConsumerError) -> Self {
        let (stage, actual_error, evaluation_performed, events_forwarded, completed) = match error {
            ConsumerError::HostGuard(error) => ("host-guard", error.owner_wire(), false, 0, false),
            ConsumerError::CoreValidation(error) => ("parse", json!(error), false, 0, false),
            ConsumerError::SourceProjection(error) => ("source-projection", json!(error), false, 0, false),
            ConsumerError::Admit(error) => ("admit", error.owner_wire(), false, 0, false),
            ConsumerError::AdmitRun(error) => ("admit-run", error.owner_wire(), false, 0, false),
            ConsumerError::Run { violation, events_forwarded } =>
                ("run", violation.owner_wire(), true, events_forwarded, false),
            ConsumerError::Observation { violation, summary, events_forwarded } =>
                ("observation", json!({"violation":violation.owner_wire(),
                    "summary":summary.owner_wire()}), true, events_forwarded, true),
        };
        Self { stage, actual_error, evaluation_performed,
            events_forwarded:events_forwarded.to_string(), completed }
    }
}

pub fn parse_config(source: &str) -> Result<ExecutionConfig, CompileFailure> {
    let value = crate::json::parse(source).map_err(|error| CompileFailure::core("config", error))?;
    let config: ExecutionConfig = serde_json::from_value(value)
        .map_err(|error| CompileFailure::input("config", error.to_string()))?;
    // Wire-only shape constraint. Existing host admission decides numerical policy.
    config.consumer().map_err(|error| CompileFailure::input("config", error))?;
    Ok(config)
}

/// SDK-free product entry point. No defaults, scenarios, fixture-name selection,
/// Core evaluation, constraint generation or native/backend dispatch occur here.
/// `observe` preserves a distinct caller execution dependency; false never runs.
pub fn compile_host(
    source: &str,
    action_name: &str,
    config_source: &str,
    target_source: &str,
    observe: bool,
) -> Result<CompileArtifact, CompileFailure> {
    let config = parse_config(config_source)?;
    let consumer = config.consumer().map_err(|error| CompileFailure::input("config", error))?;
    let target_descriptor = crate::json::parse(target_source)
        .map_err(|error| CompileFailure::core("config", error))?;
    validate_config(&consumer)
        .map_err(|error| CompileFailure::consumer(ConsumerError::HostGuard(error)))?;
    let program = crate::parse(source).map_err(|error| CompileFailure::core("parse", error))?;
    preflight_source_global(&program, &consumer).map_err(CompileFailure::consumer)?;
    let action_index = program.actions.iter().position(|action| action.name == action_name)
        .ok_or_else(|| CompileFailure::input("select", format!("unknown action: {action_name}")))?;
    let inputs = BTreeMap::<String, crate::Value>::new();
    let action = preflight_action_inputs(&program, action_index, &inputs, &consumer)
        .map_err(CompileFailure::consumer)?;
    if !action.inputs.is_empty() {
        return Err(CompileFailure {
            stage:"select", actual_error:json!({"unsupported":"parameterized-action",
                "action":action.name, "action_index":action_index.to_string(),
                "declared_inputs":action.inputs}), evaluation_performed:false,
            events_forwarded:"0".into(), completed:false,
        });
    }
    // Each branch owns exactly one admission/image pair. In observe mode the
    // pair comes only from run_scalar_action, never a separate first preparation.
    let (preparation, admission, observation) = if observe {
        let mut sink = |_event: &StepEventC| {};
        let run = run_scalar_action(&program, action_index, &inputs, &consumer, &mut sink)
            .map_err(CompileFailure::consumer)?;
        let actual = run.observe().map_err(CompileFailure::consumer)?;
        let admission = wire::admission(&actual);
        let resource = run.resource_report().map_err(CompileFailure::consumer)?;
        let observation = Observation::Observed {
            evaluation_performed:true, summary:run.completed.summary.owner_wire(),
            events_forwarded:run.completed.events_forwarded.to_string(),
            image:wire::image(&actual.image), resource_report:resource,
        };
        (run.preparation, admission, observation)
    } else {
        let prepared = prepare_scalar_action_host(&program, action_index, &inputs, &consumer)
            .map_err(CompileFailure::consumer)?;
        let actual = prepared.observe().map_err(|error| CompileFailure {
            stage:"observation", actual_error:error.owner_wire(), evaluation_performed:false,
            events_forwarded:"0".into(), completed:false,
        })?;
        let admission = wire::admission(&actual);
        (prepared.preparation, admission,
            Observation::NotRequested { evaluation_performed:false })
    };
    Ok(CompileArtifact {
        format:"sherlock-host-artifact/1",
        qualification:"host-only-scalar-value-projection; no effects/constraints/native/target qualification",
        source:SourceIdentity { language:program.language, semantics:program.semantics,
            agreement:program.name, action:action_name.to_owned(), action_index:action_index.to_string(),
            source_digest:crate::intent::hash_bytes(source.as_bytes()) },
        execution_profile:EXECUTION_PROFILE,
        input_policy:InputPolicy { kind:"parameter-free", declared:vec![], supplied:vec![] },
        config:ConfigIdentity { fields:config.clone(), file_digest:crate::intent::hash_bytes(config_source.as_bytes()) },
        target:TargetDescriptor { descriptor:target_descriptor,
            file_digest:crate::intent::hash_bytes(target_source.as_bytes()) },
        preparation, admission,
        plans:Plans {
            witness:Plan { status:"not-implemented", artifact:None, reason:"no exported WitnessPlan in this batch" },
            constraint:Plan { status:"not-implemented", artifact:None, reason:"no Sherlock constraint emission in this batch" },
            export:Plan { status:"unsupported", artifact:None, reason:"HostOnly source route has no native exporter" },
        },
        observation,
        evidence:Evidence { source_preparation:"prepared-and-admitted",
            host_observation:if observe {"observed"} else {"unrun"},
            native:"unrun", target:"unrun", acceptance:"none" },
    })
}

#!/usr/bin/env python3
"""Ordinary exact JSON data comparison against independently sealed R131 goldens.
No model, Machine, compiler, production reporting or provider imports/calls.
Object member order is not meaningful JSON; array order, type and presence are exact.
"""
import argparse
import json
from pathlib import Path

HERE = Path(__file__).resolve().parent
REPORTS = {name: HERE / ("EXPECTED-" + name.upper() + "-REPORT.json")
           for name in ("whole", "first", "second", "cumulative")}

class Difference(ValueError):
    pass

def exact(expected, actual, path=""):
    if type(expected) is not type(actual):
        raise Difference(f"{path or '/'}: expected {type(expected).__name__}, got {type(actual).__name__}")
    if isinstance(expected, dict):
        if expected.keys() != actual.keys():
            raise Difference(f"{path or '/'}: missing={sorted(expected.keys()-actual.keys())}, extra={sorted(actual.keys()-expected.keys())}")
        for key, value in expected.items():
            escaped = key.replace("~", "~0").replace("/", "~1")
            exact(value, actual[key], path + "/" + escaped)
    elif isinstance(expected, list):
        if len(expected) != len(actual):
            raise Difference(f"{path or '/'}: expected {len(expected)} entries, got {len(actual)}")
        for index, (left, right) in enumerate(zip(expected, actual)):
            exact(left, right, path + "/" + str(index))
    elif expected != actual:
        raise Difference(f"{path or '/'}: expected {expected!r}, got {actual!r}")

def compare(name, actual):
    expected = json.loads(REPORTS[name].read_text())
    exact(expected, actual)

def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("report", choices=REPORTS)
    parser.add_argument("actual", type=Path)
    args = parser.parse_args()
    try:
        compare(args.report, json.loads(args.actual.read_text()))
    except (Difference, ValueError, OSError) as error:
        parser.exit(1, str(error) + "\n")
    print("PASS: complete selected report JSON equals frozen " + args.report + " expectation")

if __name__ == "__main__":
    main()
