/// Opaque source symbol. K comparisons use canonical signed-decimal names.
/// Other spellings belong to the typed host API, outside that common domain.
pub type Name = String;
pub type Environment = BTreeMap<Name, Value>;
pub type ExprRef = Arc<Expr>;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Tree {
    Leaf(u128),
    Branch(Arc<Tree>, Arc<Tree>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Value {
    UInt(u128),
    Bool(bool),
    List(Vec<u128>),
    Tree(Tree),
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Type { UInt, Bool, List, Tree }

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum BinaryOp { Add, Sub, Mul, Eq, Lt, Cons, Branch }

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum UnaryOp { Leaf, Not }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Pattern {
    Var(Name), Nil, Cons(Name, Name), Leaf(Name), Branch(Name, Name),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arm { pub pattern: Pattern, pub body: ExprRef }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Expr {
    Literal(Value),
    Var(Name),
    Let(Name, ExprRef, ExprRef),
    Binary(BinaryOp, ExprRef, ExprRef),
    Unary(UnaryOp, ExprRef),
    If(ExprRef, ExprRef, ExprRef),
    Call(Name, Vec<ExprRef>),
    Match(ExprRef, Vec<Arm>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Parameter { pub name: Name, pub ty: Type }

#[derive(Clone, Debug, PartialEq, Eq)]

    pub name: Name,
    pub parameters: Vec<Parameter>,
    pub return_type: Type,
    pub body: ExprRef,
}

/// Function order is significant: lookup takes the first matching name.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Program { pub functions: Vec<Function>, pub entry: ExprRef }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Frame {
    Let(Name, ExprRef, Environment),
    Left(BinaryOp, ExprRef, Environment),
    Right(BinaryOp, Value),
    Unary(UnaryOp),
    If(ExprRef, ExprRef, Environment),
    Match(Vec<Arm>, Environment),
    Args(Name, Vec<Value>, Vec<ExprRef>, Environment),
    Return(Type, Environment),
    Scope(Environment),
}

    Return(Value),
    Push(Frame, Box<Control>, Environment),
    Invoke(Name, Vec<Value>),
    Enter(Option<Function>, Vec<Value>),
    Bound(Option<Environment>, Type, ExprRef),
    Cases(Value, Vec<Arm>, Environment),
    CaseResult(Option<Environment>, Value, ExprRef, Vec<Arm>, Environment),
    PrimitiveResult(Result<Value, u8>),
    Error(u8),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Outcome { Pending, Done(Value), Rejected(u8), Exhausted }

/// Complete post-startup K boundary state on the typed host common domain.
/// Frames are stored top first. `None` control represents K's `.K`.
/// Snapshots are observations; no arbitrary-state import is exposed.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Snapshot {
    pub program: Arc<Program>,
    pub control: Option<Control>,
    pub environment: Environment,
    pub frames: Vec<Frame>,
    pub stack_limit: usize,
    pub remaining_fuel: u128,
    pub initial_fuel: u128,
    pub outcome: Outcome,
}

impl Snapshot {
    pub fn depth(&self) -> usize { self.frames.len() }
    pub fn charged_steps(&self) -> u128 { self.initial_fuel - self.remaining_fuel }
}

/// A source rule transition, including free terminal publication.
/// Line numbers refer only to `K_SOURCE_SHA256`, not to executed K rules.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct StepRecord {
    pub rule_start_line: usize,
    pub charge: u8,
    pub before: Snapshot,
    pub after: Snapshot,
}

use super::reference_types::*;
use std::sync::Arc;

pub struct Machine { pub(super) state: Snapshot }

impl Machine {
    pub fn new(program: Arc<Program>, fuel: u128, stack_limit: usize) -> Self {
        Self { state: Snapshot {
            control: Some(Control::Eval(program.entry.clone())), program,
            environment: Environment::new(), frames: vec![], stack_limit,
            remaining_fuel: fuel, initial_fuel: fuel, outcome: Outcome::Pending,
        } }
    }

    pub fn snapshot(&self) -> Snapshot { self.state.clone() }

    pub fn step(&mut self) -> Option<StepRecord> {
        unimplemented!("Sherlock reference transitions")
    }

    pub fn run(&mut self) -> Snapshot {
        while self.step().is_some() {}
        self.snapshot()
    }
}

use super::*;
use std::sync::Arc;

#[test]
fn literal_publishes_at_exact_zero_remaining_fuel() {
    let program = Arc::new(Program {
        functions: vec![],
        entry: Arc::new(Expr::Literal(Value::UInt(7))),
    });
    let mut machine = Machine::new(program, 1, 8);
    let evaluated = machine.step().expect("literal step");
    assert_eq!(evaluated.rule_start_line, 102);
    assert_eq!(evaluated.charge, 1);
    assert_eq!(evaluated.after.remaining_fuel, 0);
    assert_eq!(evaluated.after.control, Some(Control::Return(Value::UInt(7))));
    let published = machine.step().expect("free publication");
    assert_eq!(published.rule_start_line, 80);
    assert_eq!(published.charge, 0);
    assert_eq!(published.after.outcome, Outcome::Done(Value::UInt(7)));
    assert!(machine.step().is_none());
}
