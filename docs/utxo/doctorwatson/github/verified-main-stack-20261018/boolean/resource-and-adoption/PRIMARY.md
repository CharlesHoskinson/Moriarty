class LiveCapture:

        self._eof_observed = None

        self.parse_errors = 0
        self.buffer = b""
        self.byte_count = 0

        return round(max(0.0, self._clock() - self._origin), 6)

    @staticmethod

        if value is None:
            return "null"
        if isinstance(value, dict):
            return "object"
        if isinstance(value, list):
            return "array"
        if isinstance(value, str):
            return "string"
        if isinstance(value, bool):
            return "boolean"
        return "number"

    def _observe_value(self, value):
        # Fixed keys and finite enums only. No arbitrary event/status/text export.
        if not isinstance(value, dict):
            self._event_counts["nonObject"] += 1
            return

        if category != "result" or len(self._result_observations) >= 2:
            return
        result = value.get("result")
        observation = {"parsedValueIndex": len(self.events) - 1,
            "observedAfterSeconds": self._elapsed(),
            "resultType": self._json_type(result) if "result" in value else "missing",
            "statusClass": "RESULT_NOT_OBJECT", "statusLiteral": None,
            "structuredOutputType": "missing"}

            status = result.get("status")

                observation["statusClass"] = "MISSING"
            elif not isinstance(status, str):
                observation["statusClass"] = "NON_STRING"
            elif status == "SUCCESS":
                observation["statusClass"] = "ALLOWLISTED"
                observation["statusLiteral"] = "SUCCESS"
            else:
                observation["statusClass"] = "OTHER_STRING"
            if "structured_output" in result:
                observation["structuredOutputType"] = self._json_type(result["structured_output"])
        self._result_observations.append(observation)

        if self._process_exit is None:
            self._process_exit = {"code": exit_code, "observedAfterSeconds": self._elapsed()}

            "eventCounts": self._event_counts,
            "resultObservations": self._result_observations,
            "resultObservationsOmitted": self._event_counts["result"] - len(self._result_observations),
            "captureParseErrors": self.parse_errors,
            "parsedTopLevelValues": len(self.events), "stdoutBytes": self.byte_count,
            "captureReachedEOF": self.complete, "eofObservedAfterSeconds": self._eof_observed,
            "processExit": self._process_exit, "actualBackendIdentityAttested": False,
            "reviewPayloadExportedHere": False, "noPrivateCaptureReread": True})

    def _line(self, line):
        if not line.strip():
            return
        try:
            self.events.append(json.loads(line.decode("utf-8")))
            self._observe_value(self.events[-1])
        except (ValueError, UnicodeError, RecursionError):
            self.parse_errors += 1

    def feed(self, chunk):
        self.complete = False
        self.byte_count += len(chunk)
        self.buffer += chunk
        while b"\n" in self.buffer:
            line, self.buffer = self.buffer.split(b"\n", 1)
            self._line(line)

    def finish(self):
        if self.buffer:
            self._line(self.buffer)
            self.buffer = b""
        self.complete = True
        if self._eof_observed is None:
            self._eof_observed = self._elapsed()


def finish_capture(capture, bundle, exit_code, unchanged):
    outcome = select_terminal_review(capture.events, schema=bundle["schema"],
        review_token=bundle["task"]["reviewToken"], terminal_exit_code=exit_code,
        input_pins_unchanged=unchanged, capture_parse_errors=capture.parse_errors)
    if not capture.complete:
        outcome = {"decision": "NO_EXPORT", "reason": "CAPTURE_NOT_COMPLETE", "review": None,
                   "receipt": dict(outcome["receipt"], decision="NO_EXPORT", reason="CAPTURE_NOT_COMPLETE")}
    return outcome


def _new_file(path, mode="wb"):
    return os.fdopen(os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600), mode)


def _write_json(path, value):
    with _new_file(path) as handle:
        handle.write((json.dumps(value, indent=2) + "\n").encode())


def _group_present(pgid):
    try:
        os.killpg(pgid, 0)
        return True
    except ProcessLookupError:
        return False


def _kill_owned_group(pgid):
    try:
        os.killpg(pgid, signal.SIGKILL)
    except ProcessLookupError:
        pass


def _run_new_process(bundle, output_dir, argv):
    capture = LiveCapture()
    deadline = time.monotonic() + bundle["task"]["wallSeconds"] + 5
    with _new_file(output_dir / "provider.private.jsonl") as private_stdout, \
            _new_file(output_dir / "stderr.private") as private_stderr:
        process = subprocess.Popen(argv, cwd=bundle["task"]["packet"], stdin=subprocess.DEVNULL,
            stdout=subprocess.PIPE, stderr=private_stderr, env=dict(os.environ), start_new_session=True)
        _write_json(output_dir / "CHILD.json", {"pid": process.pid, "ownedProcessGroup": process.pid})
        try:
            with selectors.DefaultSelector() as readiness:
                readiness.register(process.stdout, selectors.EVENT_READ)
                while time.monotonic() < deadline:
                    ready = readiness.select(min(0.2, max(0, deadline - time.monotonic())))
                    if not ready:
                        continue
                    chunk = os.read(process.stdout.fileno(), 65536)
                    if not chunk:
                        capture.finish()
                        break
                    private_stdout.write(chunk)
                    private_stdout.flush()
                    capture.feed(chunk)
            try:
                process.wait(timeout=max(0.01, deadline - time.monotonic()))
            except subprocess.TimeoutExpired:
                _kill_owned_group(process.pid)
                process.wait(timeout=1)
        finally:
            process.stdout.close()
            if process.poll() is None or _group_present(process.pid):
                _kill_owned_group(process.pid)
            if process.poll() is None:
                process.wait(timeout=1)
        capture.observe_process_exit(process.returncode)
        cleanup = {"ownedProcessGroup": process.pid, "ownedGroupRemaining": _group_present(process.pid),
                   "captureReachedEOF": capture.complete}
        return capture, process.returncode, cleanup


def run_one(task_path, task_sha256, output_dir):
    """One launch only. Existing output directory is an error, never a replay."""
    bundle = load_task(task_path, task_sha256)
    output_dir = Path(output_dir)
    if not output_dir.is_absolute():
        raise ValueError("new absolute output directory required")
    output_dir.mkdir(mode=0o700, parents=False, exist_ok=False)
    argv = build_argv(bundle, output_dir)
    started = datetime.now(timezone.utc).isoformat()
    _write_json(output_dir / "START.json", {"id": bundle["task"]["id"], "startedAt": started,
        "taskSha256": task_sha256, "inputPins": bundle["pins"], "requestedModel": bundle["task"]["model"],
        "argvWithoutPrompt": argv[:-1], "promptSha256": bundle["task"]["promptSha256"],
        "actualBackendIdentityAttested": False, "selectorSha256": SELECTOR_SHA256})
    capture, exit_code, cleanup = _run_new_process(bundle, output_dir, argv)
    _write_json(output_dir / "LIVE-METADATA.json", capture.live_metadata())
    unchanged = pins_match(bundle["pins"])
    outcome = finish_capture(capture, bundle, exit_code, unchanged)
    receipt = dict(outcome["receipt"], id=bundle["task"]["id"], taskSha256=task_sha256,
        startedAt=started, endedAt=datetime.now(timezone.utc).isoformat(), stdoutBytes=capture.byte_count,
        parsedTopLevelValues=len(capture.events), selectorSha256=SELECTOR_SHA256,
        requestedModel=bundle["task"]["model"], noPrivateCaptureReread=True,
        sourceReviewCorrectnessEstablished=False)
    if outcome["decision"] == "EXPORT":
        _write_json(output_dir / "TERMINAL-REVIEW.json", outcome["review"])
    _write_json(output_dir / "CLEANUP.json", cleanup)
    _write_json(output_dir / "TERMINAL-EXPORT-RECEIPT.json", receipt)
    return receipt


def main():

    args = parser.parse_args()

    print(json.dumps(receipt), flush=True)
    return 0 if receipt["decision"] == "EXPORT" else 2

"""Pure ordinary decoder for AGY's observed nested terminal wire path.

No file/process/network calls or recursive candidate search. The caller supplies
the terminal process status, immutable-input observation, schema and token.
This module neither attests backend identity nor evaluates review correctness.
"""
from copy import deepcopy
import json

from jsonschema import validators
from jsonschema.exceptions import SchemaError, ValidationError
from referencing import Registry
from referencing.exceptions import Unresolvable

POSITIVE_STATUSES = frozenset({"SUCCESS"})
_PARSE_FAILURE = object()


def select_terminal_review(
    events,
    *,
    schema,
    review_token,
    terminal_exit_code,
    input_pins_unchanged,
    capture_parse_errors=0,
):
    """Return EXPORT/full review or NO_EXPORT/None; never mutate inputs.

    ``events`` is an iterable of parsed top-level JSON values. The single
    terminal result must be the last parsed event. Only its exact nested
    result.structured_output dictionary is eligible. Unknown nonterminal
    dictionary events are ignored, never recursively searched.

    Input-pin truth and exit0 are externally supplied observations, not checks
    performed by this pure function. Schema validation uses the installed
    jsonschema validator for a known declared draft (2020-12 when omitted),
    with an empty, non-retrieving reference registry. Local schema references
    may resolve; missing external resources produce NO_EXPORT, never a fetch.
    """
    receipt = {
        "policy": "agy-single-nested-terminal-structured-output",
        "terminalExitCode": terminal_exit_code,
        "inputPinsUnchangedSupplied": input_pins_unchanged is True,
        "captureParseErrors": 0,
        "resultEvents": 0,
        "terminalStatusLiteral": None,
        "cliInitModelLabels": [],
        "actualBackendIdentityAttested": False,
        "noRecursiveOrTextFallback": True,
    }

    def outcome(reason, review=None):
        decision = "EXPORT" if review is not None else "NO_EXPORT"
        return {"decision": decision, "reason": reason,
                "review": deepcopy(review),
                "receipt": dict(receipt, decision=decision, reason=reason)}

    if type(capture_parse_errors) is not int or capture_parse_errors < 0:
        return outcome("INVALID_PARSE_ERROR_COUNT")
    parse_errors = capture_parse_errors
    event_not_object = False
    event_count = 0
    result_position = None
    payload = None
    for event in events:
        if event is _PARSE_FAILURE:
            parse_errors += 1
            continue
        event_count += 1
        if not isinstance(event, dict):
            event_not_object = True
            continue
        if event.get("event") == "init":
            init = event.get("init")
            if isinstance(init, dict) and isinstance(init.get("model"), str):
                label = init["model"]
                if label not in receipt["cliInitModelLabels"]:
                    receipt["cliInitModelLabels"].append(label)
        if event.get("event") == "result":
            receipt["resultEvents"] += 1
            result_position = event_count
            if receipt["resultEvents"] == 1:
                payload = event.get("result")

    receipt["captureParseErrors"] = parse_errors
    if parse_errors:
        return outcome("CAPTURE_PARSE_ERRORS")
    if event_not_object:
        return outcome("EVENT_NOT_OBJECT")
    if type(terminal_exit_code) is not int or terminal_exit_code != 0:
        return outcome("NONZERO_OR_MISSING_EXIT")
    if input_pins_unchanged is not True:
        return outcome("INPUT_PINS_NOT_UNCHANGED")
    if receipt["resultEvents"] == 0:
        return outcome("NO_RESULT_EVENT")
    if receipt["resultEvents"] != 1:
        return outcome("MULTIPLE_RESULT_EVENTS")
    if result_position != event_count:
        return outcome("RESULT_NOT_LAST")
    if not isinstance(payload, dict):
        return outcome("RESULT_PAYLOAD_NOT_OBJECT")
    if "status" not in payload:
        return outcome("MISSING_RESULT_STATUS")
    status = payload["status"]
    if not isinstance(status, str):
        return outcome("RESULT_STATUS_NOT_STRING")
    receipt["terminalStatusLiteral"] = status
    if status not in POSITIVE_STATUSES:
        return outcome("UNRECOGNIZED_RESULT_STATUS")
    if "structured_output" not in payload:
        return outcome("MISSING_STRUCTURED_OUTPUT")
    review = payload["structured_output"]
    if not isinstance(review, dict):
        return outcome("STRUCTURED_OUTPUT_NOT_OBJECT")
    if not isinstance(review_token, str) or not review_token or review.get("reviewToken") != review_token:
        return outcome("REVIEW_TOKEN_MISMATCH")

    if not isinstance(schema, (dict, bool)):
        return outcome("SCHEMA_INVALID")
    if isinstance(schema, dict) and "$schema" in schema:
        if not isinstance(schema["$schema"], str):
            return outcome("SCHEMA_INVALID")
        validator_class = validators.validator_for(schema, default=None)
        if validator_class is None:
            return outcome("SCHEMA_DIALECT_UNSUPPORTED")
    else:
        validator_class = validators.Draft202012Validator
    try:
        validator_class.check_schema(schema)
        validator_class(schema, registry=Registry()).validate(review)
    except SchemaError:
        return outcome("SCHEMA_INVALID")
    except ValidationError:
        return outcome("SCHEMA_VALIDATION_FAILED")
    except Unresolvable:
        return outcome("SCHEMA_REFERENCE_UNRESOLVED")
    return outcome("SINGLE_POSITIVE_TERMINAL_STRUCTURED_OUTPUT", review)


def decode_terminal_json_lines(lines, **conditions):
    """Consume supplied text lines once; malformed lines prevent any export.

    Blank lines are ignored. This adapter opens no files and launches no
    producer; root owns acquisition and the terminal/pin observations.
    """
    def events():
        for line in lines:
            if not isinstance(line, str):
                yield _PARSE_FAILURE
                continue
            if not line.strip():
                continue
            try:
                yield json.loads(line)
            except (ValueError, RecursionError):
                yield _PARSE_FAILURE

    return select_terminal_review(events(), **conditions)
