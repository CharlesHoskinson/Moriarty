impl ExtCapacitiesC {

        Self {
            plan_case_slots: 0,
            field_path_slots: 0,
            pattern_child_slots: 0,
            input_rows: 0,
            asset_rows: 0,
            state_rows: 0,
            account_rows: 0,
            liability_rows: 0,
            await_site_rows: 0,
            await_binder_slots: 0,
            balance_cells: 0,
            journal_rows: 0,
            pending_op_rows: 0,
            observations_per_invocation: 0,
            max_lifetime_work: 0,
            resume_rows: 0,

    pub fn effective_for_extended_domain(opt: Option<Self>) -> Self {
        opt.unwrap_or_else(Self::all_zero)
    }
}
